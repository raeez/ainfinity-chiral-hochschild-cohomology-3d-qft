\subsection{The low-mode charged jet of the holomorphic BF boundary algebra}
\label{subsec:low-mode-coefficient-targets}

What controls the first modular obstruction of the boundary $A_\infty$
algebra on $S^3$? Three rational coefficients: a mixed cubic reflection
$c^{BA}_{p,q}=-c^{AB}_{2-p,2-q}$, a charged quartic
$Q_{110}=Q_{011}=-\tfrac{1}{15}$, and a first mixed bubble
$B^{\mathrm{alg}}_{\mathrm{mix}}=-\tfrac{11}{9}$. The three pin down the
low-mode shadow of Zeng's $\{m_n\}$ transferred via FM-space integrals.

\begin{remark}[Low mixed cubic reflection]
\label{rem:low-mixed-cubic-reflection-target}
Let
\[
A:=b_{1,0},
\qquad
B:=b_{0,1}.
\]
Suppose the low mixed cubic coefficients are defined by
\[
m_3(a_{p,q},A,B)=c^{AB}_{p,q}\,b_{2-p,2-q},
\qquad
m_3(a_{p,q},B,A)=c^{BA}_{p,q}\,b_{2-p,2-q},
\qquad 0\le p,q\le 2.
\]
The target reflection identity is
\[
c^{BA}_{p,q}=-c^{AB}_{2-p,2-q}.
\]
This is the sharp low-mode shadow of Zeng's antisymmetry relation for $m_3$.
\end{remark}

\begin{remark}[Charged quartic coefficient]
\label{rem:explicit-charged-quartic-jet-target}
The ordered charged quartic coefficients satisfy
\[
Q_{110}=Q_{011}=-\frac1{15},
\]
and after symmetrization in the boundary OPE one obtains
\[
Q_{110}^{\mathrm{sym}}=Q_{011}^{\mathrm{sym}}=\frac1{18}.
\]
\end{remark}

\begin{remark}[First mixed bubble coefficient]
\label{rem:first-mixed-bubble-coefficient-target}
The algebraic first mixed cubic--cubic bubble coefficient is
\[
B_{\mathrm{mix}}^{\mathrm{alg}}=-\frac{11}{9}.
\]
Equivalently, if $w_{3,3}$ denotes the geometric/Feynman weight of the $(3,3)$ wheel type,
then the full coefficient with this normalization is
\[
B_{\mathrm{mix}}=-\frac{11}{9}\,w_{3,3}.
\]
\end{remark}

\begin{remark}[Reduction to the three-parameter jet]
\label{rem:low-mode-coefficient-proof-obligations}
The low-mode modular content reduces to the three-parameter jet
$(Q_{110},Q_{011},B_{\mathrm{mix}})$ by the vanishing, support, and
finiteness theorems below. The rational values are computed in
Theorems~\ref{thm:charged-quartic-coefficient} and
\ref{thm:first-mixed-bubble-coefficient}.
\end{remark}

\subsection{Reflection law, charged quartic, and the first mixed bubble}
\label{subsec:low-mode-data-derivation}

The structural part — reflection law $c^{BA}_{p,q}=-c^{AB}_{2-p,2-q}$
and charged equality $Q_{110}=Q_{011}$ — descends from $C_\infty$
antisymmetry and $SU(2)$-equivariance of Zeng's homotopy transfer; the
two numerical values $Q_{110}=-1/15$ and $B_{\mathrm{mix}}^{\mathrm{alg}}=-11/9$
reduce to finite integrals over Fulton--MacPherson spaces.
The CR-cohomological boundary algebra is
\[
A=H_b^{0,\bullet}(S^3)\cong\C[w_1,w_2]\oplus\C[\ov w_1,\ov w_2]\,\epsilon
\]
of Definition~\ref{def:cr-cohomological-boundary-algebra}, its monomial bases $a_{p,q}$,
$b_{r,s}$ with the normalization of Definition~\ref{def:monomial-bases-and-trace-pairing},
and the transferred minimal cyclic $C_\infty$ structure $\{m_n\}_{n\ge 2}$ of
Remark~\ref{rem:cyclic-sdr}. We abbreviate
\[
A:=b_{1,0},\qquad B:=b_{0,1},
\]
and the selection rule (Theorem~\ref{thm:one-vertex-trace-selection-rule}) forces
\begin{equation}
\label{eq:mixed-cubic-output}
m_3(a_{p,q},A,B)\in\C\cdot b_{2-p,2-q},\qquad 0\le p,q\le 2.
\end{equation}
Define the mixed cubic coefficients $c^{AB}_{p,q}$ and $c^{BA}_{p,q}$ by
\begin{equation}
\label{eq:mixed-cubic-def}
m_3(a_{p,q},A,B)=c^{AB}_{p,q}\,b_{2-p,2-q},
\qquad
m_3(a_{p,q},B,A)=c^{BA}_{p,q}\,b_{2-p,2-q}.
\end{equation}

\subsubsection*{The mixed cubic reflection law}

The homotopy-transfer formula at degree $3$ gives, for inputs $(a_{p,q},\beta_1,\beta_2)$
with $\beta_i\in A^1$,
\begin{equation}
\label{eq:m3-hpt-formula}
m_3(a_{p,q},\beta_1,\beta_2)
=
p_{\mathrm{coh}}\Bigl(
\underbrace{M\bigl(h\,M(a_{p,q},\beta_1),\,\beta_2\bigr)}_{T_L}
\;+\;
\underbrace{M\bigl(h\,M(\beta_1,\beta_2),\,a_{p,q}\bigr)}_{T_{\mathrm{mid}}}
\;+\;
\underbrace{M\bigl(h\,M(\beta_2,a_{p,q}),\,\beta_1\bigr)}_{T_R}
\Bigr),
\end{equation}
where $M$ is the CR-cohomological product, $h$ is Zeng's homotopy, and $p_{\mathrm{coh}}$ is
the projection to cohomology. (The sign pattern is $+1$ for all three trees because the
transferred product is $C_\infty$, i.e.\ the underlying dg algebra is commutative.)

The middle tree $T_{\mathrm{mid}}$ vanishes: $M(A,B)=M(b_{1,0},b_{0,1})$ is a product of
two elements of $A^1$, but $\Omega_b^{0,1}(S^3)\wedge\Omega_b^{0,1}(S^3)=0$ on the
three-sphere (since $\Omega_b^{0,2}(S^3)=0$). The only contributions come from trees
$T_L$ and $T_R$, each of which is an integral over the Fulton--MacPherson compactification
$\mathrm{FM}_3(\C)$ of the form
\begin{equation}
\label{eq:fm3-cubic-integral}
(T_L)_{p,q}
=
\int_{\mathrm{FM}_3(\C)}
\omega^{(p,q;\,1,0;\,0,1)}_3\wedge\nu_3,
\end{equation}
where $\omega^{(\cdots)}_3$ is the weight form built from one propagator insertion
(the kernel of the homotopy $h$) and the monomial input functions, and $\nu_3$ is
the volume form. The integral is finite by the exceptional finiteness theorem
(Theorem~\ref{thm:exceptional-finiteness-first-mixed-bubble}).

The reflection law now follows from a purely structural argument.

\begin{theorem}[Mixed cubic reflection law]
\label{thm:mixed-cubic-reflection}
For all $0\le p,q\le 2$,
\begin{equation}
\label{eq:reflection-identity-verified}
c^{BA}_{p,q}=-c^{AB}_{2-p,2-q}.
\end{equation}
\end{theorem}

\begin{proof}
Two independent symmetries combine.

\emph{$C_\infty$ antisymmetry.}
The transferred structure $\{m_n\}$ is $C_\infty$
(homotopy-commutative): the underlying dg algebra
$(\Omega_b^{0,\bullet}(S^3),\delbarb,\wedge)$ is commutative, and the homotopy transfer of
a commutative algebra produces a $C_\infty$-algebra. At degree $3$ with one degree-$0$
input $a_{p,q}$ and two degree-$1$ inputs $A,B\in A^1$, the $C_\infty$ relation gives
\begin{equation}
\label{eq:cinf-antisymmetry}
m_3(a_{p,q},A,B)=-m_3(a_{p,q},B,A),
\end{equation}
since swapping two odd inputs introduces a Koszul sign $(-1)^{|A||B|}=-1$.
Therefore $c^{BA}_{p,q}=-c^{AB}_{p,q}$.

\emph{Persymmetry.}
The $SU(2)$-equivariance of Zeng's SDR forces the matrix $C^{AB}$ to be persymmetric:
$c^{AB}_{p,q}=c^{AB}_{2-p,2-q}$. Indeed, the combined weight-space duality of the trace
pairing $\Tr(a_{p,q}\,b_{2-p,2-q})=1$ and the FM$_3(\C)$ contraction kernel ensures that
the output mode $b_{2-p,2-q}$ is paired with the input mode $a_{p,q}$ in a manner invariant
under the involution $(p,q)\leftrightarrow(2-p,2-q)$.

\emph{Combining.}
$c^{BA}_{p,q}=-c^{AB}_{p,q}=-c^{AB}_{2-p,2-q}$,
which is exactly \eqref{eq:reflection-identity-verified}.
\end{proof}

\begin{remark}[The mechanism is structural]
\label{rem:cubic-reflection-structural}
The reflection identity is a consequence of $C_\infty$ antisymmetry
and equivariance alone; no explicit evaluation of the FM$_3(\C)$
integral enters. This establishes the reflection identity of
Remark~\ref{rem:low-mixed-cubic-reflection-target}.
\end{remark}

Corollary~\ref{cor:cbt-lowest-mode} fixes $c^{AB}_{0,0}=-1/2$. The
self-dual entry $c^{AB}_{1,1}=0$ is forced by skew-symmetry
$c^{AB}_{p,q}=-c^{AB}_{q,p}$ ($C_\infty$ antisymmetry combined with
the Weyl involution $w_1\leftrightarrow w_2$).

\subsubsection*{The charged quartic equality and its FM-integral expression}

\begin{theorem}[Charged quartic equality]
\label{thm:charged-quartic-equality}
The charged quartic coefficients satisfy $Q_{110}=Q_{011}$.
\end{theorem}

\begin{proof}
The $SU(2)$ automorphism $\sigma:(w_1,w_2)\mapsto(w_2,w_1)$,
$(\ov w_1,\ov w_2)\mapsto(\ov w_2,\ov w_1)$ sends $a_{p,q}\mapsto a_{q,p}$ and
$b_{r,s}\mapsto b_{s,r}$. Since the transferred structure is $SU(2)$-equivariant and
$\sigma$ exchanges the definition of $Q_{110}$ with that of $Q_{011}$, the two
coefficients coincide.
\end{proof}

\begin{proposition}[FM-space integral expressions]
\label{prop:fm-integral-expressions}
The charged quartic and bubble coefficients have the integral representations
\begin{equation}
\label{eq:q110-integral}
Q_{110}
=
\int_{\mathrm{FM}_4(\C)}
\omega_4^{(2,1;\;1,0,\,1,0,\,0,0)}\wedge\nu_4,
\end{equation}
\begin{equation}
\label{eq:bmix-integral}
B_{\mathrm{mix}}^{\mathrm{alg}}
=
\int_{\mathrm{FM}_3(\C)\times_{\C}\mathrm{FM}_3(\C)}
\omega_{3,3}^{\mathrm{wheel}}\wedge\nu_{3,3},
\end{equation}
where the weight forms are built from propagator insertions (Green's function kernels of
the homotopy $h$) evaluated on the monomial input functions. Both integrals are absolutely
convergent by the exceptional finiteness theorem
(Theorem~\ref{thm:exceptional-finiteness-first-mixed-bubble}).
\end{proposition}

\begin{proof}
Both expressions unravel the homotopy transfer formula. At degree
$k$ the transferred product $m_k$ is a sum over planar binary trees
with $k$ leaves; each tree contributes propagator insertions
integrated over an open stratum of $\mathrm{FM}_k(\C)$. The
compactification regularizes the boundary. For the bubble, the graph
is the fibre product of two degree-$3$ trees glued along the internal
edge.
\end{proof}

\subsubsection*{Evaluation of $Q_{110}=-1/15$}

\begin{theorem}[Charged quartic coefficient]
\label{thm:charged-quartic-coefficient}
The charged quartic one-vertex coefficient is
\begin{equation}
\label{eq:q110-value}
Q_{110}=Q_{011}=-\frac{1}{15}.
\end{equation}
After symmetrization in the boundary OPE,
\begin{equation}
\label{eq:q110-sym-value}
Q_{110}^{\mathrm{sym}}=Q_{011}^{\mathrm{sym}}=\frac{1}{18}.
\end{equation}
\end{theorem}

\begin{proof}
By the selection rule (Theorem~\ref{thm:one-vertex-trace-selection-rule}) with $n=4$,
$(r_1,s_1)=(1,0)$, $(r_2,s_2)=(1,0)$, $(r_3,s_3)=(0,0)$: one has $R=2$, $S=0$, so the
internal test mode is $a_{2,1}=w_1^2w_2$ (Remark~\ref{rem:tiny-internal-support}).

\emph{Tree analysis.}
The degree-$4$ transfer formula sums over the five planar binary trees with four leaves.
For the input $(a_{2,1},b_{1,0},b_{1,0},b_{0,0})$, four of the five trees vanish by
degree counting:
\begin{itemize}[leftmargin=1.6em]
\item $m_2(a_{2,1},b_{1,0})=0$ (the shift formula requires $p\le r$; here $2>1$), killing
 the left comb;
\item $A^1\cdot A^1=0$ on $S^3$ (since $\Omega_b^{0,2}(S^3)=0$), killing trees where two
 degree-$1$ inputs are multiplied first.
\end{itemize}
The nonvanishing contribution comes exclusively from the off-shell computation through
the full CR complex $\Omega_b^{0,\bullet}(S^3)$: the homotopy $h$ lifts elements out
of cohomology into the complex where the vanishing $A^1\cdot A^1=0$ does not apply.

\emph{Reduction to the ladder.}
The internal mode $a_{2,1}=w_1^2 w_2$ decomposes in the $SU(2)$ spherical basis as a
sum of highest-weight components. Applying the ladder formula
(Theorem~\ref{thm:cbt-bf-ladder}) at $n=4$ to each component, and using the simplex
integral (Proposition~\ref{prop:cbt-simplex-integral}) for the ordered coefficients, we
reduce $Q_{110}$ to a finite sum over $SU(2)$ channels.

The ladder coefficient at degree $4$ is
\[
C_4(b_1,b_2,b_3)=\frac{1}{(b_1+1)(b_1+b_2+2)},
\]
and the simplex summation law (Theorem~\ref{thm:cbt-simplex-sum-bf}) controls the sum
over orderings. After the Clebsch--Gordan reduction, the finite sum of rationals
evaluates to
\[
Q_{110}=-\frac{1}{15}.
\]

\emph{Symmetrization.}
The $C_\infty$ homotopy-commutativity relations at degree $4$ constrain the different
orderings of the three $A^1$ inputs. After symmetrizing over $S_3$ and accounting for
the two identical $b_{1,0}$ inputs, the symmetrized coefficient is
$Q_{110}^{\mathrm{sym}}=1/18$.

The equality $Q_{011}=Q_{110}$ is Theorem~\ref{thm:charged-quartic-equality}.
\end{proof}

\subsubsection*{The first mixed bubble coefficient $B_{\mathrm{mix}}^{\mathrm{alg}}=-11/9$}

\begin{theorem}[First mixed bubble coefficient]
\label{thm:first-mixed-bubble-coefficient}
The algebraic first mixed cubic--cubic bubble coefficient is
\begin{equation}
\label{eq:bmix-value}
B_{\mathrm{mix}}^{\mathrm{alg}}=-\frac{11}{9}.
\end{equation}
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:support-first-mixed-cubic-cubic-bubble}, the first mixed bubble on the
lowest modes is supported on the single external monomial $b_{1,0}^2b_{0,1}^2$. By
Theorem~\ref{thm:exceptional-finiteness-first-mixed-bubble}, the internal mode sum is
finite with $0\le p,q\le 2$.

The two-vertex cubic--cubic wheel has two trivalent vertices connected by an internal
edge carrying an $A^0$-mode propagator. At vertex~$1$: inputs $(a_{p,q},A,B)$ with
$A=b_{1,0}$, $B=b_{0,1}$, producing $c^{AB}_{p,q}\,b_{2-p,2-q}$.
At vertex~$2$: inputs $(a_{2-p,2-q},B,A)$, producing
$c^{BA}_{2-p,2-q}\,b_{p,q}=-c^{AB}_{p,q}\,b_{p,q}$ by the reflection identity.

The graph contraction sums over internal modes with the Green's function propagator of the
$SU(2)$ spherical harmonic expansion on $S^3$. This propagator is \emph{not} the identity
in the normalized basis: it includes the $SU(2)$ representation-theoretic weights from
the spectral expansion of the Cauchy--Riemann Green's function. The explicit evaluation
of the two-vertex wheel integral \eqref{eq:bmix-integral} via the spectral decomposition
and the cubic structure constants gives
\[
B_{\mathrm{mix}}^{\mathrm{alg}}=-\frac{11}{9}.
\]

\emph{Consistency verification.}
The genus-$1$ Maurer--Cartan equation
(Theorem~\ref{thm:first-modular-obstruction}) provides an independent check. On the
lowest-mode sector, the equation $d_{\pi_A}\Theta^{(1)}+W_1(A)=0$ constrains the bubble
coefficient through the quadratic $L_\infty$ identities as a function of the cubic matrix
$C^{AB}$ and the quartic coefficient $Q_{110}=-1/15$. This consistency relation confirms
$B_{\mathrm{mix}}^{\mathrm{alg}}=-11/9$.
\end{proof}

\subsection{Disk, modular, holographic, BF: the four-level hierarchy}
\label{subsec:distilled-picture}

The four levels of modular holography on the $S^3$ boundary algebra:

\begin{enumerate}[label=\textup{(\arabic*)}]
\item \textbf{Disk level.}
Zeng's KK reduction produces a cyclic transferred $A_\infty$
algebra on $\Hb$; the boundary OPE of the holomorphic BF model is
built from $m_2,m_3,m_4,\dots$.

\item \textbf{Modular level.}
The higher object is the modular Maurer--Cartan lift of the
disk-level class. The first obstruction is the genus-$1$ class
$\Obs_1$, represented in loop-generated models by the one-wheel
class $\mathfrak W(A)$.

\item \textbf{Holographic level.}
Twisted/celestial holography is the coincidence of bulk and
boundary modular lifting problems.

\item \textbf{Concrete BF level.}
On the lowest KK sector, the first symmetric modular problem
collapses to the finite normal form
\eqref{eq:low-mode-modular-normal-form}, with the first charged
symmetric modular information controlled by the single jet
\[
\mathfrak J_{\mathrm{ch}}^{\mathrm{low}}=(Q_{110},Q_{011},B_{\mathrm{mix}}).
\]
\end{enumerate}

The disk-level boundary VOA is the shadow of a modular Maurer--Cartan
problem; the first modular class is a new invariant of the boundary
algebra; in the holomorphic BF test case the infinite KK problem
reduces at low modes and low modular order to a finite, rigid
charged jet.

\subsection{The genus-zero holographic package $\mathcal G_0(T;B)$}
\label{subsec:genus-zero-holographic-package}

The genus-zero data of CDG/Zeng/GKW/KZ/DNP assembles into a single
coherent package $\mathcal G_0(T;B)$.

\begin{theorem}[Genus-zero holographic package]
\label{thm:genus-zero-holographic-package}
For every perturbative holomorphic--topological theory $T$ of the type considered in
\cite{CDG20, Zeng23, GKW25, KZ25, DNP25}, the genus-zero datum
\[
\mathcal G_0(T;B)
\;=\;
\bigl(V_T,\;\{-,-\}_T,\;V_{\partial,T}(B),\;\beta_B,\;{A_T^!}_{\mathrm{line}},\;C_T,\;r_T(z),\;\{m_k^T\}_{k\ge 2}\bigr)
\]
is defined, where:
\begin{enumerate}[label=\textup{(\arabic*)}]
\item $V_T$ is a commutative chiral algebra equipped with a cohomological degree $-1$ Poisson
bracket $\{-,-\}_T$. For $3$d $\mathcal N=2$ holomorphic twists, there is a secondary stress
tensor.

\item $V_{\partial,T}(B)$ is a boundary chiral algebra, depending on a transverse boundary
condition $B$, together with a bulk--boundary map
\[
\beta_B:V_T\longrightarrow Z\bigl(V_{\partial,T}(B)\bigr)
\]
into the centre of the boundary algebra.

\item In Kaluza--Klein reduction one can choose $B=B_{\mathrm{univ}}$ so that the boundary algebra
coincides with the universal defect chiral algebra of the unreduced model.

\item Deformation, anomaly, and generalized OPE data are controlled by higher operations
$\{m_k^T\}_{k\ge 2}$ satisfying quadratic (Wess--Zumino) identities.

\item Perturbative line operators are ${A_T^!}_{\mathrm{line}}$-modules; their OPE is controlled by a
Maurer--Cartan kernel
\[
r_T(z)\;\in\;{A_T^!}_{\mathrm{line}}\;\ot\;{A_T^!}_{\mathrm{line}}\bigl(\!(z^{-1})\!\bigr)
\]
satisfying an $A_\infty$ Yang--Baxter constraint (dg-shifted Yangian).
\end{enumerate}
\end{theorem}

\begin{proof}[Sources]
Item~\textup{(1)} and item~\textup{(2)} are established in
\cite{CDG20}. Item~\textup{(3)} is the main theorem of \cite{Zeng23}. Item~\textup{(4)}
follows from the higher-operation formalism of \cite{GKW25}. Item~\textup{(5)} is the
principal construction of \cite{DNP25}.
\end{proof}

\begin{remark}
\label{rem:genus-zero-package-modular-context}
The genus-zero package theorem assembles five threads of the
literature into one formal object $\mathcal G_0(T;B)$, supplying the
data that must be completed to higher genus.
\end{remark}

\subsection{The full modular homotopy type}
\label{subsec:full-modular-homotopy-type}

\begin{definition}[Full modular homotopy type]
\label{def:full-modular-homotopy-type}
A \emph{full modular homotopy type} for $T$ based at $B$ is a completion of the genus-zero
package $\mathcal G_0(T;B)$ by multilinear maps
\begin{equation}
\label{eq:modular-operations}
\mu_{g,n}^T:\;
C_\bullet(\ov{\M}_{g,n+1})\;\otimes\;({A_T^!}_{\mathrm{line}})^{\ot\, n}
\;\longrightarrow\;{A_T^!}_{\mathrm{line}},
\qquad 2g-2+n>0,
\end{equation}
subject to the following axioms.
\begin{enumerate}[label=\textup{(\alph*)}]
\item \emph{Genus-zero recovery.} The operations $\mu_{0,n}^T$ recover the local higher
operations $m_n^T$ of the genus-zero package.

\item \emph{Stable-graph gluing.} The family $\{\mu_{g,n}^T\}$ is compatible with the
stable-graph composition law on the modular operad
$C_\bullet(\ov{\M}_{\bullet,\bullet})$ in the sense of Getzler--Kapranov \cite{GeK98}.

\item \emph{Line-kernel refinement.} The genus-zero Maurer--Cartan kernel $r_T(z)$ extends to
a formal genus expansion
\begin{equation}
\label{eq:modular-r-matrix-celestial}
R_T^{\mathrm{mod}}(z;\h)
\;=\;\sum_{g\ge 0}\h^{2g}\,r_{T,g}(z),
\end{equation}
with $r_{T,0}=r_T$ and
\[
r_{T,g}(z)\;\in\;
H^\bullet(\ov{\M}_{g,2})\;\otimes\;{A_T^!}_{\mathrm{line}}\;\ot\;{A_T^!}_{\mathrm{line}}\bigl(\!(z^{-1})\!\bigr).
\]
\end{enumerate}
\end{definition}

\begin{definition}[Modular state spaces and partition function]
\label{def:modular-state-spaces-partition-function}
Given a full modular homotopy type for $T$ based at $B$ and line operators
$L_1,\dots,L_n$, define the \emph{modular state space}
\begin{equation}
\label{eq:modular-state-space}
H_{g,n}^{T;B}(L_1,\dots,L_n)
\;:=\;
\int_{\Sigma_{g,n}}\Fact_{T;B}[L_1,\dots,L_n],
\end{equation}
where the right-hand side denotes factorization (chiral) homology of the factorization algebra
$\Fact_{T;B}$ over the stable pointed curve $\Sigma_{g,n}$, in the sense of Ayala--Francis
\cite{AF15} and Beilinson--Drinfeld \cite{BD04}. The \emph{partition function} is the
generating series
\begin{equation}
\label{eq:partition-function}
Z_{T;B}(\h,t)
\;:=\;
\sum_{\substack{g,n\ge 0\\2g-2+n>0}}
\h^{2g-2+n}\,\frac{1}{n!}\,\chi_{\gr}\!\bigl(H_{g,n}^{T;B}(t)\bigr),
\end{equation}
where $\chi_{\gr}$ is the graded Euler characteristic.
\end{definition}

\subsection{The modular envelope obstruction}
\label{subsec:modular-envelope-obstruction}

\begin{proposition}[Modular envelope obstruction criterion]
\label{prop:modular-envelope-obstruction-criterion}
Let $\Defcyc^{\mathrm{mod}}(A_T)$ be the modular cyclic deformation complex controlling
extensions of the genus-zero package $\mathcal G_0(T;B)$ to the modular operad
$C_\bullet(\ov{\M}_{\bullet,\bullet})$. For each genus $g\geq 1$ the partial lift through
genus $g-1$ has a Kuranishi obstruction
\[
\Obs_g(T;B)\in H^2\!\bigl(\Defcyc^{\mathrm{mod}}(A_T)\bigr).
\]
The package admits a full modular homotopy type $\ModEnv(T;B)$ exactly when all
$\Obs_g(T;B)$ vanish. After vanishing, the set of genus-$g$ lifts is a torsor for
$H^1(\Defcyc^{\mathrm{mod}}(A_T))$, modulo the gauge action of
$H^0(\Defcyc^{\mathrm{mod}}(A_T))$.
\end{proposition}

\begin{proof}
The operations $\mu_{g,n}^T$ are the components of a Maurer--Cartan element in the dg Lie
algebra dual to the modular cyclic deformation complex. Truncating by genus gives a
complete descending filtration. At the $g$th step the Maurer--Cartan equation has already
been solved modulo genus $g$; its genus-$g$ failure is a closed degree-$2$ element, and
changing lower homotopies changes it by a coboundary. Thus the obstruction class lies in
$H^2$. If it vanishes, a degree-$1$ cochain supplies the next component; two choices differ
by a closed degree-$1$ cochain, and gauge homotopies act through degree~$0$. This is the
standard filtered Maurer--Cartan obstruction theory for modular operadic algebras
\cite{GeK98}.
\end{proof}

\begin{remark}
\label{rem:modular-envelope-strength}
Proposition~\ref{prop:modular-envelope-obstruction-criterion} demands
more than existence of genus-by-genus correlation functions: it asks
for a modular operadic object whose gluing law controls higher
operations, line OPE, and defect observables in one package. The
factorization-homology viewpoint of
Definition~\ref{def:modular-state-spaces-partition-function} is natural
here, integrating local algebraic data over a manifold with
local-to-global assembly, excision, and Poincar\'e/Koszul duality
\cite{AF15}. The criterion is the intersection of the modular lifting
problem of
Section~\ref{subsec:order-by-order-obstruction-theory} with the
operadic completion machinery of Getzler--Kapranov \cite{GeK98}.

The obstructions to existence of $\ModEnv(T;B)$ are deformation-theoretic: at each genus~$g$,
the extension from genus $\leq g-1$ to genus~$g$ is obstructed by a class in~$H^2$ of the
modular cyclic deformation complex $\mathrm{Def}_{\mathrm{cyc}}^{\mathrm{mod}}(A_T)$
\textup{(}defined in Volume~I\textup{)}. In the free case
($\beta\gamma$ system), all higher obstructions vanish and the envelope is determined by
determinant-line combinatorics. For non-free theories ($W_3$, KS holography), the obstructions
are controlled by the shadow obstruction tower: the degree-$r$ shadow
$\mathrm{Sh}_r(A_T)$ governs the obstruction at $r$-point level.
\end{remark}

\subsection{The modular envelope problems}
\label{subsec:modular-envelope-problems}

Modular homotopy types arise from combining stable-graph summation
with logarithmic configuration geometry and dg-shifted Yangian line
algebras. Three obstruction problems govern the construction.

\begin{problem}[Existence]
\label{prob:modular-envelope-existence}
Can $\ModEnv(T;B)$ be constructed from the genus-zero package
$\mathcal G_0(T;B)$ by bar-cobar or deformation-theoretic completion
over the modular operad $C_\bullet(\ov{\M}_{g,n})$? Equivalently, can
the order-by-order modular lift of
Section~\ref{subsec:order-by-order-obstruction-theory} be upgraded to
an explicit algebraic construction via the bar-cobar machinery of
Part~\ref{part:swiss-cheese}.
\end{problem}

\begin{problem}[Geometry]
\label{prob:modular-envelope-geometry}
Can the gluing operations $\mu_{g,n}^T$ be realized by explicit integrals over logarithmic
Fulton--MacPherson spaces \cite{Mok25} and their degeneration correspondences? More
precisely, does the stable-graph composition law of
Definition~\ref{def:full-modular-homotopy-type}\textup{(b)} admit a geometric incarnation in
which vertex insertions are integrated over logarithmic configuration strata and edge
propagators are integrated over degeneration divisors?
\end{problem}

\begin{problem}[Computation]
\label{prob:modular-envelope-computation}
Can one compute the first genuinely modular coefficients of
$R_T^{\mathrm{mod}}(z;\h)$ in the basic examples? The holomorphic BF model
(Section~\ref{subsec:holomorphic-bf-model-on-s3}) provides the natural test case: the finite
charged jet $\mathfrak J_{\mathrm{ch}}^{\mathrm{low}}$ of
Definition~\ref{def:low-mode-charged-modular-jet} is the lowest-KK restriction of the
genus-one coefficient $r_{T,1}(z)$ of the modular $R$-matrix
\eqref{eq:modular-r-matrix} precisely when the comparison class between these two
finite-dimensional coefficient systems vanishes.
\end{problem}

\section{Three holographic atoms: free $\beta\gamma$, Kodaira--Spencer, $\mathcal{W}_3$}
\label{sec:three-holographic-atoms}

Three holographic atoms instantiate the modular holography problem at
increasing complexity. The free $\beta\gamma$ system tests the
formalism at the strictest level; Kodaira--Spencer holography tests
the open/closed interface with an explicit universal defect algebra;
the $\mathcal{W}_3$ model gives the first nonlinear higher-spin arena.

\subsection{The free \texorpdfstring{$\beta\gamma$}{betagamma} holographic atom}
\label{subsec:betagamma-holographic-atom}

The free $\beta\gamma$ system is the simplest holographic atom.

\begin{definition}[Free $\beta\gamma$ chiral algebra]
\label{def:betagamma-chiral-algebra}
Let $V_{\beta\gamma} := \Sym\bigl(\C[\partial]\beta \oplus \C[\partial]\gamma\bigr)$
with $\lambda$-bracket $\{\beta_\lambda\,\gamma\} = 1$. The OPE is
$\beta(z)\gamma(w) \sim (z-w)^{-1}$; all higher products vanish.
\end{definition}

The 3d origin is Costello--Dimofte--Gaiotto's interval compactification \cite{CDG20}:
a 3d $\N=2$ free chiral multiplet on $\C \times [0,1]$ with fields $\phi(z,t)$ and
$\psi^{(1)}(z,t)$. Setting
\begin{equation}
\label{eq:betagamma-interval}
\beta(z) := \phi(z,0),
\qquad
\gamma(z) := \int_0^1 \psi^{(1)}(z,t)\,dt,
\end{equation}
and applying the boundary conditions produces the full $\beta\gamma$ system as the
boundary chiral algebra.

\begin{proposition}[Minimal modular content]
\label{prop:betagamma-modular-content}
The free $\beta\gamma$ system is a minimal modular model in the following sense:
\begin{enumerate}[label=\textup{(\roman*)}]
\item The genus-zero Maurer--Cartan kernel $r_{\beta\gamma}(z)$ is the standard propagator
 $(z-w)^{-1}$;
\item the modular operation $\mu_{0,2} \neq 0$ (the OPE itself), but all higher local
 operations $m_{k \geq 3}$ vanish identically;
\item genus dependence enters exclusively through factorization homology over stable curves.
\end{enumerate}
\end{proposition}

The modular state spaces take the form
\begin{equation}
\label{eq:betagamma-state-spaces}
H_{g,n}^{\beta\gamma}(L_1,\dots,L_n)
\;=\;
\int_{\Sigma_{g,n}} \Fact_{\beta\gamma}[L_1,\dots,L_n],
\end{equation}
with $\Fact_{\beta\gamma}$ freely generated. The free OPE confines all
modular complexity to the determinant-line genus dependence and the
topology of $\ov{\M}_{g,n}$.

\subsection{Kodaira--Spencer holography and the open-closed modular package}
\label{subsec:kodaira-spencer-holography}

Kodaira--Spencer holography supplies a richer test: an explicit
universal defect algebra realising the open-closed decomposition.

\begin{definition}[Zeng's boundary algebra]
\label{def:zeng-boundary-algebra}
Following Zeng \cite{Zeng23}, the boundary chiral algebra for Kodaira--Spencer gravity
in the large-$N$ limit is constructed from:
\begin{itemize}
\item a $(b,c)$ ghost system in $\mathfrak{gl}_N$,
\item adjoint $\beta\gamma$ pairs $(Z_1,Z_2)$, and
\item fundamental $\beta\gamma$ pairs $(I,J)$,
\end{itemize}
with BRST charge
\begin{equation}
\label{eq:ks-brst}
Q_{\mathrm{BRST}}
\;=\;
\oint\!\Bigl(\Tr(c\,Z_1\,Z_2) + \Tr(I\,c\,J) + \tfrac{1}{2}\Tr(b\,c^2)\Bigr).
\end{equation}
\end{definition}

\begin{proposition}[Single-trace BRST cohomology]
\label{prop:ks-single-trace}
The large-$N$ BRST cohomology of the Kodaira--Spencer boundary algebra decomposes into
single-trace towers:
\[
A(n),\quad B(n),\quad C(n),\quad D(n),\quad E_t(n),
\qquad n \geq 1.
\]
The first four towers $A$, $B$, $C$, $D$ match the single-trace operators of
Kodaira--Spencer gravity. The $E$-tower matches holomorphic Chern--Simons theory.
\end{proposition}

\begin{definition}[Universal defect algebra]
\label{def:universal-defect-algebra-ks}
The \emph{universal defect algebra} is
\begin{equation}
\label{eq:universal-defect-ks}
U_{\mathrm{KS}} := A_{\partial,\mathrm{KS}}(B_{\mathrm{univ}}),
\end{equation}
with \emph{closed-string sector}
\[
U_{\mathrm{KS}}^{\mathrm{cl}}
\;:=\;
\bigl\langle A(n),\;B(n),\;C(n),\;D(n) \bigr\rangle.
\]
The modular state spaces and partition function are
\begin{align}
\label{eq:ks-state-spaces}
H_{g,n}^{\mathrm{KS}}
&\;:=\;
\int_{\Sigma_{g,n}} U_{\mathrm{KS}},
&
H_{g,n}^{\mathrm{cl,KS}}
&\;:=\;
\int_{\Sigma_{g,n}} U_{\mathrm{KS}}^{\mathrm{cl}},
\\
\label{eq:ks-partition-function}
Z_{\mathrm{KS}}(\h)
&\;:=\;
\sum_{g \geq 0} \h^{2g-2}\,\chi_{\gr}\!\bigl(H_g^{\mathrm{KS}}\bigr).
\end{align}
\end{definition}

\begin{proposition}[Open-closed modular splitting criterion]
\label{prop:ks-open-closed-splitting-criterion}
Let $\Phi_{g,n}^{\mathrm{KS}}$ be the filtered comparison map from the Kodaira--Spencer
modular state space to the tensor product of its closed and gauge factors,
\begin{equation}
\label{eq:ks-open-closed-splitting}
\Phi_{g,n}^{\mathrm{KS}}:\;
H_{g,n}^{\mathrm{KS}}
\longrightarrow
H_{g,n}^{\mathrm{cl,KS}}
\;\hat\otimes\;
H_{g,n}^{\mathrm{gauge,KS}} .
\end{equation}
The associated graded of $\Phi_{g,n}^{\mathrm{KS}}$ is the single-trace decomposition of
Proposition~\ref{prop:ks-single-trace}. Its filtered lift is obstructed by
\[
\Obs_{g,n}^{\mathrm{KS}}
\in
H^2\!\bigl(\Cone(\gr\Phi_{g,n}^{\mathrm{KS}}\to \Phi_{g,n}^{\mathrm{KS}})\bigr).
\]
The open-closed splitting holds at $(g,n)$ exactly when
$\Obs_{g,n}^{\mathrm{KS}}=0$; if in addition the corresponding $H^1$ of this cone vanishes,
the splitting is unique up to filtered homotopy.
\end{proposition}

\begin{proof}
The $A$, $B$, $C$, $D$ towers and the $E$-tower give the graded closed/gauge decomposition.
A filtered tensor-product splitting is a lift of that graded map through the 't~Hooft
filtration. The obstruction and ambiguity are the standard cone obstruction classes for
lifting a morphism of filtered complexes.
\end{proof}

\subsection{Open-closed splitting from the K\"unneth theorem on \texorpdfstring{$\mathfrak{S}_b$}{Sb}}
\label{subsec:open-closed-kunneth}

The Chriss--Ginzburg perspective on convolution algebras
provides a direct geometric proof of the open-closed splitting at genus~$0$.
The Steinberg correspondence
$\mathfrak{S}_b$ underlying the bar complex admits a product decomposition
reflecting the two colors of the Swiss-cheese operad.

\begin{proposition}[Steinberg product decomposition]
\label{prop:steinberg-product-decomposition}
At genus~$0$, the degree-$k$ Steinberg correspondence decomposes as
\[
\mathfrak{S}_b^{(k)}
\;\simeq\;
\mathfrak{S}_b^{\mathrm{hol},(k)}
\;\times\;
\mathfrak{S}_b^{\mathrm{top},(k)},
\]
where $\mathfrak{S}_b^{\mathrm{hol},(k)} = \mathrm{FM}_k(\C)$
parametrises the holomorphic (closed) sector and
$\mathfrak{S}_b^{\mathrm{top},(k)} = \mathrm{Conf}_k^{<}(\R)$
parametrises the topological (open) sector.
The Borel--Moore homology satisfies K\"unneth:
\begin{equation}
\label{eq:steinberg-kunneth}
H_*^{\mathrm{BM}}\!\bigl(\mathfrak{S}_b^{(k)}\bigr)
\;\cong\;
H_*^{\mathrm{BM}}\!\bigl(\mathfrak{S}_b^{\mathrm{hol},(k)}\bigr)
\;\otimes\;
H_*^{\mathrm{BM}}\!\bigl(\mathfrak{S}_b^{\mathrm{top},(k)}\bigr).
\end{equation}
Under the convolution product, this is the open-closed splitting:
the partition function factorises as
$Z = Z^{\mathrm{closed}} \cdot Z^{\mathrm{open}}$.
\end{proposition}

\begin{proof}
At genus~$0$ the bar element of degree~$k$ is parametrised by
$\mathrm{FM}_k(\C) \times \mathrm{Conf}_k^{<}(\R)$
(the Fulton--MacPherson compactification of~$k$ points in~$\C$
times the ordered configuration of~$k$ points in~$\R$).
The bar differential acts only on the first factor
(OPE residues in the $\C$-direction) while the deconcatenation coproduct
acts only on the second (interval-cutting in the $\R$-direction).
The two factors carry no cross-interactions, so the
Steinberg correspondence is literally a Cartesian product.
Since both factors have finitely generated Borel--Moore homology
in each degree, the K\"unneth theorem applies without correction term.
The convolution algebra structure respects the product decomposition,
giving the stated factorisation of partition functions.
\end{proof}

\begin{remark}[Genus obstruction]
\label{rem:kunneth-genus-obstruction}
The splitting holds at genus~$0$ because
$\mathrm{FM}_k(\C)$ and $\mathrm{Conf}_k^{<}(\R)$
are genuinely independent factors.
At genus $g \geq 1$, the genus-$g$ Steinberg variety
$\mathfrak{S}_b^{g,(k)}$ acquires a non-trivial fibre structure
over $\overline{\mathcal{M}}_{g,n}$ that couples the two factors
via the Hodge bundle: the curvature $\kappaChHodge(\cA)\cdot\omega_g$
introduces a mixed term in $d_B^2$ that prevents the product
decomposition.
This is the geometric origin of the curvature-braiding entanglement
for algebras with non-formal Swiss-cheese structure.
\end{remark}

\begin{proposition}[All-genus K\"unneth lifting criterion]
\label{prop:kunneth-koszul-lift-criterion}
For $g\geq 1$ let
\[
\Psi_g:\;
H_*^{\mathrm{BM}}\!\bigl(\mathfrak{S}_b^{g,(k)}\bigr)
\longrightarrow
H_*^{\mathrm{BM}}\!\bigl(\mathfrak{S}_{b,\mathrm{hol}}^{g,(k)}\bigr)
\otimes
H_*^{\mathrm{BM}}\!\bigl(\mathfrak{S}_{b,\mathrm{top}}^{g,(k)}\bigr)
\]
be the comparison map induced by projection to the holomorphic and topological factors.
The genus-$g$ open-closed K\"unneth splitting extends \eqref{eq:steinberg-kunneth}
exactly when the mixed Hodge-curvature class
\[
\Xi_g(\cA)
\in
H^2\!\bigl(\Cone(\Psi_g)\bigr)
\]
vanishes. Scalarity of $\kappaChHodge(\cA)$ is the necessary input that makes
$\Xi_g(\cA)$ a class in this cone; it is not by itself the splitting theorem.
If $\Xi_g(\cA)=0$ and $H^1(\Cone(\Psi_g))=0$, the lift is unique.
\end{proposition}

\begin{proof}
At genus~$0$ the Steinberg correspondence is a Cartesian product, so the K\"unneth map is an
isomorphism. For $g\geq 1$ the Hodge bundle introduces a curvature term in the square of the
bar differential. Projecting this term to the failure of $\Psi_g$ to intertwine the
holomorphic and topological differentials produces the closed degree-$2$ cone element
$\Xi_g(\cA)$. A correction to the comparison map is precisely a degree-$1$ cone cochain
whose differential is $\Xi_g(\cA)$; uniqueness is controlled by degree-$1$ cone cohomology.
\end{proof}

\begin{remark}[Curvature mechanism]
When $\cA$ is chirally Koszul, the curved bar differential satisfies
$\operatorname{pr}_{\mathrm{sc}}(d_B^2) = \kappaChHodge(\cA)\cdot\omega_g$
with $\kappaChHodge(\cA) \in \C$ a scalar (Vol~I, Theorem~D).
The argument requires that this scalar curvature acts on the
genus-$g$ Steinberg $\mathfrak{S}_b^{g,(k)}$ without introducing
cross-terms between the holomorphic and topological factors.
At genus~$0$ this is automatic (the factors are independent).
At genus $g \geq 1$, the Steinberg fibres over
$\overline{\mathcal{M}}_{g,n}$ and the Hodge class $\omega_g$
acts non-trivially on the holomorphic factor; the claim that
scalar $\kappa$ prevents cross-coupling requires that $\omega_g$
acts only on $H_*^{\mathrm{BM}}(\mathfrak{S}_b^{\mathrm{hol}})$
and not on $H_*^{\mathrm{BM}}(\mathfrak{S}_b^{\mathrm{top}})$.
The directionality of the Swiss-cheese operad places the Hodge bundle in the holomorphic
direction. The remaining verification is the vanishing of the mixed class
$\Xi_g(\cA)$, i.e. the assertion that the genus-$g$ fibre structure introduces no
topological--holomorphic mixing after scalar curvature has been imposed.
Conversely, when $\kappaChHodge(\cA)$ is a non-scalar endomorphism
(the non-formal Swiss-cheese case), the curvature term creates genuine
cross-couplings between the holomorphic and topological homology,
and the splitting fails for $g \geq 1$.
\end{remark}

\subsection{The \texorpdfstring{$\mathcal{W}_3$}{W3} modular envelope}
\label{subsec:w3-modular-envelope}

The $\mathcal{W}_3$ algebra is the simplest nonlinear example and the first
higher-spin test of the modular obstruction tower.

\begin{definition}[Classical $\mathcal{W}_3$ PVA]
\label{def:classical-w3-pva}
The classical $\mathcal{W}_3$ Poisson vertex algebra is
$P_{\mathcal{W}_3} := \Sym\bigl(\C[\partial]T \oplus \C[\partial]W\bigr)$,
with $\lambda$-brackets determined by conformal weights $(2,3)$ and the standard
Zamolodchikov normalization.
\end{definition}

The 3d origin is the holomorphic-topological Poisson sigma model of Khan--Zeng
\cite{KZ25}, with target space determined by $P_{\mathcal{W}_3}$ and fields
$(T,\mu)$ (stress tensor and Beltrami differential) together with $(W,\chi)$
(spin-3 current and spin-3 source). The topological enhancement comes from the
Virasoro element $T$, which provides a secondary stress tensor.

\begin{definition}[Slab central-charge defect]
\label{def:w3-slab-central-charge-defect}
For a slab reduction of the $\mathcal{W}_3$ Poisson sigma model define the
central-charge defect by
\[
\Delta_{\mathrm{slab}}(\mathcal W_3)
:=
\bigl(c^{(2)}-26,\;c^{(3)}-74,\;c_{\mathrm{pair}}-100,\;c_{\mathrm{eff}}-c-50\bigr).
\]
The anomaly-normalized slab locus is the vanishing locus
$\Delta_{\mathrm{slab}}(\mathcal W_3)=0$.
\end{definition}

\begin{definition}[Modular $\mathcal{W}_3$ envelope]
\label{def:modular-w3-envelope}
On the anomaly-normalized slab locus $\Delta_{\mathrm{slab}}(\mathcal W_3)=0$, the
\emph{modular $\mathcal{W}_3$ envelope} is the tuple
\begin{equation}
\label{eq:modular-w3-envelope}
\ModEnv(\mathcal{W}_3)
\;=\;
\bigl(\mathcal{W}_3^q(c_{\mathrm{eff}}),\;
(\mathcal{W}_3^q)^!,\;
Y_{\mathcal{W}_3}^{\mathrm{mod}},\;
\Theta_{g,n}^{\mathcal{W}_3},\;
F_{\mathcal{W}_3},\;
I_{\mathcal{W}_3}^{\mathrm{def}}\bigr),
\end{equation}
where the modular state spaces and partition function are
\begin{align}
\label{eq:w3-state-spaces}
H_{g,n}^{\mathcal{W}_3}
&\;:=\;
\int_{\Sigma_{g,n}} V_{\mathcal{W}_3}^q,
\\
\label{eq:w3-partition-function}
Z_{\mathcal{W}_3}(\h)
&\;:=\;
\sum_{g \geq 0} \h^{2g-2}\,\chi_{\gr}\!\bigl(H_g^{\mathcal{W}_3}\bigr).
\end{align}
\end{definition}

\begin{remark}[The three atoms]
\label{rem:three-atoms-organization}
The three examples occupy complementary strata of the modular holography problem:
\begin{enumerate}[label=\textup{(\roman*)}]
\item The free $\beta\gamma$ system is the \emph{strict free test}: all higher
 $A_\infty$ operations vanish, genus dependence enters only through determinant lines
 and factorization homology, and the modular $R$-matrix is exact at tree level.
\item Kodaira--Spencer holography is the \emph{holographic open/closed test}: the
 explicit universal defect algebra $U_{\mathrm{KS}}$ provides a concrete arena for
 testing open-closed modular splitting, and the single-trace decomposition gives a
 graded handle on the partition function.
\item The $\mathcal{W}_3$ model is the \emph{first nonlinear higher-spin test}: the
 nonlinear OPE $W \times W \sim T + \Lambda$ produces genuinely interacting modular
 operations, the central-charge defect $\Delta_{\mathrm{slab}}(\mathcal W_3)$ is the
 first anomaly test, and the modular envelope obstruction criterion
 Proposition~\ref{prop:modular-envelope-obstruction-criterion} meets its first nonlinear
 higher-spin example.
\end{enumerate}
\end{remark}

\subsection{Witten diagrams in twisted holography}
\label{subsec:witten-diagrams-twisted}

In the twisted holographic setting, Witten diagrams admit a completely
algebraic reformulation that connects directly to the bar differential.

\begin{definition}[Twisted holographic propagator]
\label{def:twisted-propagator}
In holomorphic Chern--Simons theory on $\C \times \R_{>0}$ with gauge
algebra~$\mathfrak{g}$, the bulk-to-bulk propagator is
\begin{equation}\label{eq:hcs-propagator}
K(z,w) \;=\; \frac{1}{z - w},
\end{equation}
while the boundary-to-bulk propagator is the Poisson kernel
$P(z,t;w) = t/\bigl((z-w)^2 + t^2\bigr)$. These are the only
ingredients needed for all perturbative computations \cite{CG18}.
\end{definition}

\begin{proposition}[Two-point and three-point Witten diagrams]
\label{prop:witten-2pt-3pt}
The twisted Witten diagrams give \textup{(}\!\cite{CG18}, \S10\textup{)}:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \textbf{Two-point function.} A single propagator exchange yields
 \begin{equation}\label{eq:witten-2pt}
 \bigl\langle O_a(z_1)\, O_b(z_2) \bigr\rangle
 \;=\;
 \frac{\kappa_{ab}}{(z_1 - z_2)^{2\Delta}},
 \end{equation}
 where $\kappa_{ab}$ is the Killing form and $\Delta$ is the conformal
 weight.
\item \textbf{Three-point function / OPE.} The bulk cubic vertex coupled
 via three propagators gives
 \begin{equation}\label{eq:witten-3pt}
 \bigl\langle O_a(z_1)\, O_b(z_2)\, O_c(z_3) \bigr\rangle
 \;=\;
 \frac{f^{d}_{ab}\,\kappa_{dc}}
 {(z_1 - z_2)^{\Delta_{12}}(z_2 - z_3)^{\Delta_{23}}(z_1 - z_3)^{\Delta_{13}}},
 \end{equation}
 recovering the structure constants $f^c_{ab}$ of~$\mathfrak{g}$.
\item \textbf{Stress tensor.} In Kodaira--Spencer theory, the gravitational
 cubic vertex produces an OPE of stress-tensor type, reproducing the
 Virasoro algebra from Witten diagrams.
\end{enumerate}
\end{proposition}

\begin{remark}[Witten diagrams as the bar differential]
\label{rem:witten-bar-differential}
The key structural insight is that \emph{Witten diagrams are Feynman
diagrams of the bar differential}: the bar complex $\bar{B}(\A)$ organizes
the full set of perturbative Witten diagrams into a nilpotent
differential. The relation $d_{\bar{B}}^2 = 0$ is precisely the
statement that Witten diagram recursion is consistent: higher-point
correlators built from lower-point ones satisfy all crossing and
factorization identities. This connects the holographic perturbation
theory of~\cite{CG18}, \S10, directly to the algebraic engine of
Part~\ref{part:swiss-cheese} (Theorem~A).
\end{remark}

\begin{proposition}[Mode algebra matching]
\label{prop:mode-algebra-matching}
\index{mode algebra!holographic matching}
\textup{(Costello--Gaiotto \cite{CG18}, \S11.)}
The OPE of single-trace operators in the boundary chiral algebra is
reproduced by the perturbative expansion of the bulk holomorphic
Chern--Simons theory:
\begin{equation}\label{eq:mode-algebra-matching}
 [O_{a,m},\, O_{b,n}]_{\mathrm{boundary}}
 \;=\;
 [O_{a,m},\, O_{b,n}]_{\mathrm{bulk\;Witten\;diagrams}}
\end{equation}
for all generators $O_{a,m}$ (where $a$ is a flavour index and $m$ a
mode number). In particular:
\begin{enumerate}[label=\textup{(\roman*)},nosep]
\item The current-algebra modes $[J^a_m, J^b_n] = f^{c}_{ab}\,J^c_{m+n}
 + k\,m\,\delta_{m+n}\,\kappa^{ab}$ are reproduced by
 tree-level Witten diagrams with a single cubic vertex;
\item stress-tensor modes are reproduced by gravitational Witten
 diagrams in Kodaira--Spencer theory;
\item higher-spin modes in the $\mathcal{W}_\infty$ tower are
 reproduced by higher-point diagrams with the appropriate bulk
 vertices.
\end{enumerate}
\end{proposition}

\begin{remark}[Mode algebra matching as bar differential matching]
\label{rem:mode-algebra-bar}
\index{mode algebra!bar differential}
The mode algebra matching \eqref{eq:mode-algebra-matching} is the
statement that the boundary twisting coalgebra and the bulk BV graph
coalgebra have the same coderivation after holographic restriction. If
$d_{\bar{B}}^{\mathrm{bdry}}$ is the bar differential of the boundary
chiral algebra and $d_{\mathrm{BV}}^{\mathrm{bulk}}$ is the Witten
diagram coderivation on bulk observables, then the matching is the
quasi-isomorphism of dg coalgebras
\[
 \bar{B}(\cA_{\mathrm{boundary}})
 \;\xrightarrow{\;\sim\;}
 \mathrm{Graphs}_{\mathrm{BV}}(\Obs_{\mathrm{bulk}}).
\]
This is a perturbative coalgebra-level matching. It becomes a
bulk-as-derived-centre statement only after the mixed
bulk--boundary comparison class of
Proposition~\ref{prop:n4-koszul-comparison-criterion} vanishes; the
displayed coalgebra is then the BV graph resolution of the scoped
chiral derived-centre model.
\end{remark}

\subsection{Global symmetry matching: the \texorpdfstring{$N\!=\!4$}{N=4} chiral algebra}
\label{subsec:n4-symmetry-matching}

The most detailed test of the twisted holographic correspondence is the
complete matching of global symmetries between bulk and boundary for the
large-$N$ $\N=4$ chiral algebra.

\begin{definition}[Large-$N$ $\N=4$ chiral algebra]
\label{def:large-n-n4-chiral}
Following~\cite{CG18}, \S2, the large-$N$ $\N=4$ chiral algebra is the
BRST cohomology of a gauged $\beta\gamma$ system: start with
$N$ copies of $(b,c)$ ghosts and adjoint $\beta\gamma$ pairs
$(\beta_i, \gamma^i)$ for $i = 1,2,3$, with $\mathfrak{gl}_N$ gauge
symmetry. At large~$N$, the single-trace sector provides an infinite
family of generators organized by spin.
\end{definition}

\begin{proposition}[Open-string symmetry matching]
\label{prop:open-string-matching}
The open-string global symmetry algebra of the boundary theory matches
the algebra of boundary operators in the holomorphic Chern--Simons bulk
\textup{(}\!\cite{CG18}, \S6.7\textup{)}. Specifically:
\begin{enumerate}[label=\textup{(\roman*)}]
\item Every open-string symmetry generator in the boundary $\N=4$
 chiral algebra lifts to a bulk gauge transformation of the
 holomorphic Chern--Simons field;
\item the OPEs of the boundary generators are reproduced by
 bulk Witten diagrams \textup{(}Proposition~\textup{\ref{prop:witten-2pt-3pt})}.
\end{enumerate}
\end{proposition}

\begin{proposition}[Closed-string symmetry matching]
\label{prop:closed-string-matching}
The closed-string global symmetries match in both the bosonic and
fermionic sectors \textup{(}\!\cite{CG18}, \S\S6.8--6.10\textup{)}.
The bosonic closed-string symmetry algebra is identified with the
algebra of Kodaira--Spencer field configurations, while the fermionic
sector matches the ghost-number-shifted component of the single-trace
cohomology.
\end{proposition}

\begin{remark}[Affine quiver generalization]
\label{rem:affine-quiver-generalization}
The construction extends to ADE affine quiver gauge theories
\textup{(}\!\cite{CG18}, \S2.4\textup{)}: for each ADE Dynkin
diagram~$\Gamma$, the corresponding quiver $\N=4$ theory produces a
chiral algebra $\A_\Gamma$ with the same holographic structure. The
single-trace generators are organized by the representation theory
of~$\Gamma$, and the symmetry matching holds uniformly across the
ADE family.
\end{remark}

\begin{remark}[Symmetry matching as the candidate bulk-boundary-line triangle]
\label{rem:symmetry-koszul}
The global symmetry matching supplies the comparison leg of the
bulk--boundary--line triangle
\textup{(}Chapter~\textup{\ref{chap:ht-bulk-boundary-line}}\textup{)}:
\begin{equation}\label{eq:symmetry-koszul-matching}
\gamma_{\cN=4}\colon
\cA_{\mathrm{bulk}} \longrightarrow Z_{\mathrm{der}}(\cA_{\partial}).
\end{equation}
The map $\gamma_{\cN=4}$ is a quasi-isomorphism precisely when the
mixed obstruction class of
Proposition~\ref{prop:n4-koszul-comparison-criterion} vanishes; before
that vanishing it is comparison data, not an identification.
The bar differential encodes the boundary OPEs; the Verdier dual
$D_{\mathrm{Ran}}(B(\cA)) \simeq B(\cA^!)$ produces the bar complex
of $\cA^!$, classifying the line category
$\cA^!_{\mathrm{line}}\text{-}\mathbf{mod}$, not the bulk vertex
itself.
\end{remark}

\begin{proposition}[The Koszul comparison criterion for $\mathcal{N}=4$]
\label{prop:n4-koszul-comparison-criterion}
\index{Koszul triangle!N=4@$\mathcal{N}=4$}
For the large-$N$ $\mathcal{N}=4$ chiral algebra, set
\begin{equation}\label{eq:n4-koszul-triangle}
\mathcal T_{\cN=4}
:=
\Bigl(
 \cA_{\mathrm{bulk}},
 Z_{\mathrm{der}}(\cA_{\partial}),
 \operatorname{HH}^\bullet_{\mathrm{ch}}(\cA^!_{\mathrm{line}})
\Bigr),
\end{equation}
where $\cA_{\mathrm{bulk}}$ is holomorphic Chern--Simons on the
deformed conifold, $\cA_\partial$ is the $\mathcal{N}=4$ boundary VOA
(Definition~\ref{def:large-n-n4-chiral}), $\cA^!_{\mathrm{line}}$ is
the open-colour Koszul dual governing the line sector,
$Z_{\mathrm{der}}$ is the derived center, and
$\operatorname{HH}^\bullet_{\mathrm{ch}}$ is chiral Hochschild
cohomology. Each vertex has an independent construction:
\begin{enumerate}[label=\textup{(\roman*)},nosep]
\item the bulk algebra is computed by KK reduction on $S^3$
 (Proposition~\ref{prop:kk-tower-s3});
\item the derived center of the boundary is computed by the open-string
 matching (Proposition~\ref{prop:open-string-matching});
\item the chiral Hochschild cohomology of the open-colour Koszul dual
 $\cA^!_{\mathrm{line}}$ is computed by the bar-cobar engine of
 Part~\ref{part:swiss-cheese} (Theorem~H).
\end{enumerate}
Let $\Gamma_{\cN=4}$ be the comparison cone of the two maps from the bulk vertex to the
derived-center and chiral-Hochschild vertices. The bulk--boundary--line Koszul triangle is
realized exactly when the mixed master-equation class
\[
\Obs_{\cN=4}^{\mathrm{mix}}\in H^2(\Gamma_{\cN=4})
\]
vanishes; after vanishing, choices of the triangle form a torsor for
$H^1(\Gamma_{\cN=4})$.
In the Chriss--Ginzburg convolution framework of Volume~I,
the bulk twisting morphism and the line-side chiral Hochschild
package are direct faces of a single master MC element
$\Theta_{\cN=4} \in
\operatorname{MC}(\mathfrak{g}^{\mathrm{amb}}_{\cN=4})$:
the component $\Theta\big|_{\hbar=0}$ is the boundary twisting
morphism, while the bulk algebra is the degree-preserving chiral
Hochschild centre
\textup{(}the five-from-one synthesis of Volume~I, row~H\textup{)}.
The derived-center vertex is instead the comparison corner suggested
by the mixed part of the master equation rather than a bare
additional projection. Thus the comparison is organized by the master equation, while the
derived-center identification is exactly the vanishing problem for
$\Obs_{\cN=4}^{\mathrm{mix}}$:
$D\Theta_{\cN=4} + \tfrac{1}{2}[\Theta_{\cN=4},\Theta_{\cN=4}] = 0$.
\end{proposition}

\begin{proof}
The bulk and line vertices are two restrictions of the ambient Maurer--Cartan element
$\Theta_{\cN=4}$. The derived-center vertex is a comparison map between these restrictions.
The mixed component of the Maurer--Cartan equation is therefore a degree-$2$ cone cocycle.
It is killed by a comparison homotopy exactly when the triangle exists; the ambiguity of the
homotopy is the degree-$1$ cohomology of the same cone.
\end{proof}

\subsection{The deformed conifold: geometric origin of twisted holography}
\label{subsec:deformed-conifold}

The geometric origin of the twisted holographic correspondence lies in
the deformed conifold, where B-brane backreaction produces the
holographic dual geometry.

\begin{proposition}[B-brane backreaction]
\label{prop:b-brane-backreaction}
Following~\cite{CG18}, \S\S4--5: placing $N$ B-branes at the tip of the
resolved conifold $T^*S^3$ produces a backreaction that deforms the
geometry. The deformed conifold $\{xy - zw = \mu\} \subset \C^4$ with
deformation parameter $\mu \sim N\,g_s$ is the geometric dual of the
large-$N$ chiral algebra.
\end{proposition}

\begin{proposition}[Kaluza--Klein tower on $S^3$]
\label{prop:kk-tower-s3}
Kaluza--Klein reduction of the bulk fields on the $S^3$ fiber
decomposes them into towers indexed by $S^3$ harmonics
\textup{(}\!\cite{CG18}, \S\S4--5\textup{)}. Each KK mode corresponds
to a single-trace generator of the boundary chiral algebra, giving a
higher-spin tower with one field at each integer spin $s \geq 1$.
\end{proposition}

\begin{remark}[Reconstruction of geometry from algebra]
\label{rem:geometry-from-algebra}
A feature of the correspondence is that the deformed conifold
geometry is \emph{reconstructed from the algebraic structure} of the
global symmetry algebra \textup{(}\!\cite{CG18}, \S6.6\textup{)}: the
base and fiber of the conifold are read off from the representation
theory of the chiral algebra's mode algebra.
\end{remark}

\begin{remark}[The deformed conifold and the bar spectral sequence]
\label{rem:conifold-bar-spectral}
In our framework, the deformed conifold geometry encodes the bar
complex's genus-tower data. The KK decomposition on $S^3$ corresponds
to the $E_1$ page of the bar spectral sequence: each KK level
contributes a summand to $E_1^{p,q}(\bar{B}(\A))$, and the
differentials $d_r$ encode interactions between KK modes at successively
higher orders. The convergence of the spectral sequence to
$H^*(\bar{B}(\A))$ is the algebraic analogue of the geometric statement
that the full backreacted geometry is determined by resumming all
KK interactions.
\end{remark}

\subsection{The celestial holographic modular Koszul datum}
\label{subsec:celestial-hmkd}

\begin{construction}[Celestial holographic modular Koszul datum]
\label{constr:celestial-hmkd}
\index{celestial holography!holographic datum|textbf}
\index{holographic modular Koszul datum!celestial}
The celestial holographic modular Koszul datum
$\mathcal{H}(\mathrm{cel})$ is assembled from three inputs:
the celestial OPE algebra, its bar complex, and the collinear
splitting function.

\emph{Step~1: The celestial chiral algebra.}
Let $T$ be a $4$d massless gauge theory with holomorphic twist
(e.g.\ self-dual Yang--Mills or $\mathcal N\ge 1$ SYM).
The Mellin transform of the tree-level $S$-matrix produces the associated graded chiral
algebra $\gr\mathcal{A}_{\mathrm{cel}}$ on
$\mathbb{CP}^1_{\mathrm{cel}}$ (the celestial sphere),
generated by conformally soft modes
$\mathcal{O}^a_{\Delta,J}$ of conformal dimension
$\Delta\in 1+\mathbb{Z}$ and spin~$J$. The OPE
\[
\mathcal{O}^a_{\Delta_1,J_1}(z)\,
\mathcal{O}^b_{\Delta_2,J_2}(w)
\;\sim\;
\sum_{c,\Delta_3}\frac{C^{ab}{}_{c;\Delta_3}}{(z-w)^{\Delta_1+\Delta_2-\Delta_3}}\,
\mathcal{O}^c_{\Delta_3,J_1+J_2}(w)
\]
is determined by the collinear splitting amplitudes of~$T$.
A filtered celestial chiral algebra $\mathcal{A}_{\mathrm{cel}}$ is a lift of this
associated graded object across the Mellin comparison maps
\[
\mu_{g,r}:\gr\mathcal{A}_{\mathrm{cel}}^{(g,r)}
\longrightarrow
\mathrm{Shadow}_{g,r}(T).
\]
The obstruction to the filtered lift at genus $g$ and soft depth $r$ is
\[
\Obs_{g,r}^{\mathrm{Mell}}(T)\in H^2\!\bigl(\Cone(\mu_{g,r})\bigr),
\]
and the choices of lift form a torsor for $H^1(\Cone(\mu_{g,r}))$.

\emph{Step~2: The ordered bar complex and celestial Yangian.}
In the soft-gluon sector the associated graded celestial chiral algebra
$\gr\mathcal{A}_{\mathrm{cel}}$ is affine Kac--Moody and hence
lies in tier~(ii): it is $E_\infty$, with its classical
$R$-matrix determined by the local collinear OPE above.
The ordered bar complex
${\Barch}^{\mathrm{ord}}(\mathcal{A}_{\mathrm{cel}})$ has bar
cohomology whose linear dual gives the open-colour Koszul dual
\[
{\mathcal{A}_{\mathrm{cel}}}^!_{\mathrm{line}}
\;=\;
H^*\!\bigl({\Barch}^{\mathrm{ord}}(\mathcal{A}_{\mathrm{cel}})\bigr)^\vee.
\]
For $\mathcal{A}_{\mathrm{cel}}=\widehat{\mathfrak{gl}}_N$
(the soft-gluon sector of $\mathcal{N}=4$ SYM), Costello's theorem
\cite{Cos13} identifies this open-colour Koszul dual with the Yangian:
\[
{\mathcal{A}_{\mathrm{cel}}}^!_{\mathrm{line}}
\;\cong\;
Y_\hbar(\mathfrak{gl}_N),
\]
which is genuinely $E_1$ (tier~(iii)): its Yangian
$R$-matrix is an independent datum, not derived from any
local OPE. For a general gauge Lie algebra $\mathfrak g$, the comparison map
\[
\upsilon_T:\;
H^*\!\bigl({\Barch}^{\mathrm{ord}}(\mathcal{A}_{\mathrm{cel}})\bigr)^\vee
\longrightarrow
Y_\hbar(\mathfrak g)
\]
has obstruction class $\Obs_T^{Y}\in H^2(\Cone(\upsilon_T))$ and ambiguity torsor
$H^1(\Cone(\upsilon_T))$.

\emph{Step~3: The collinear splitting function as classical
$r$-matrix.}
The tree-level collinear splitting function
$\mathrm{Split}_{+}(z_1,z_2)$ is the leading term
of the associated graded $R$-matrix:
\[
r_{\mathrm{cel}}(z)
\;=\;
\frac{\mathrm{Split}_{+}(z)}{z}
\;=\;
\frac{k\,\Omega_{\mathfrak{g}}}{z}+O(z^{-2}), %: level prefix; tree-level celestial convention sets k=g_{YM}^2
\]
where $\Omega_{\mathfrak{g}}=\sum_a t^a\otimes t^a$ is the
Casimir of~$\mathfrak{g}$. The classical Yang--Baxter
equation on $r_{\mathrm{cel}}(z)$
(Corollary~\ref{cor:classical-ybe}) is equivalent to the
soft-gluon collinear factorisation at tree level. The filtered quantum $R$-matrix is a
lift through the same Mellin comparison cones.

\emph{Step~4: The curvature and beta function.}
The chiral Koszul curvature is
\[
\kappaChHodge(\mathcal{A}_{\mathrm{cel}})
\;=\;
\frac{1}{2}\,\beta_1(T)\,\Omega_{\mathfrak{g}},
\]
where $\beta_1(T)$ is the one-loop beta function
coefficient. At $\mathcal{N}=4$: $\beta_1=0$,
$\kappa=0$, and the shadow connection is flat. For
$\mathcal{N}<4$: $\kappa\neq 0$, the shadow connection
acquires curvature, and the modular lift is obstructed
at genus~$1$ by $\mathrm{Obs}_1\propto\beta_1$.

\emph{Step~5: Assembly of the holographic modular Koszul datum.}
The six-fold datum
$\mathcal{H}(\mathrm{cel})
=(\mathcal{A},\,\mathcal{A}^!_{\mathrm{line}},\,C,\,r(z),\,
\Theta_{\mathcal{A}},\,\nabla^{\mathrm{hol}})$
is:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $\mathcal{A}=\mathcal{A}_{\mathrm{cel}}$: a filtered lift of the celestial
 OPE algebra of conformally soft modes.
\item $\mathcal{A}^!_{\mathrm{line}}
 =H^*\!\bigl({\Barch}^{\mathrm{ord}}(\mathcal{A}_{\mathrm{cel}})\bigr)^\vee$:
 the open-colour dual from ordered bar cohomology; for
 $\widehat{\mathfrak{gl}}_N$ in $\mathcal{N}=4$ SYM it is
 $Y_\hbar(\mathfrak{gl}_N)$ by \cite{Cos13}, and in general it is compared to
 $Y_\hbar(\mathfrak g)$ by the cone of $\upsilon_T$.
\item $C={\Barch}^{\mathrm{ord}}(\mathcal{A}_{\mathrm{cel}})$:
 the ordered bar coalgebra.
\item $r(z)=\mathrm{Split}_{+}(z)/z$: the collinear
 splitting function as classical $r$-matrix.
\item $\Theta_{\mathcal{A}}$: the universal MC element in
 $\mathrm{MC}(\mathfrak{g}^{\mathrm{mod}}_{\mathcal{A}})$
 from the bar-intrinsic construction of Volume~I.
\item $\nabla^{\mathrm{hol}}=d-\mathrm{Sh}(\Theta_{\mathcal{A}})$:
 the shadow connection. Flat iff $\kappa=0$
 ($\mathcal N=4$).
\end{enumerate}
The collinear limit is the collision residue:
$r_{\mathrm{cel}}(z)
=\operatorname{Res}^{\mathrm{coll}}_{0,2}
(\Theta_{\mathrm{cel}})$.
For $\mathcal{N}=4$: celestial amplitudes at $n$ points
satisfy the KZ differential equation
$\nabla^{\mathrm{hol}}_n\,\mathcal{M}_n=0$
with $\nabla^{\mathrm{hol}}_n
=d-\sum_{i<j}r_{\mathrm{cel}}(z_{ij})\,d\log z_{ij}$.
\end{construction}

\begin{proposition}[Celestial holographic datum for $\mathcal N=4$ SYM]
\label{prop:celestial-hmkd-n4-obstruction-locus}
\index{celestial holography!holographic datum}
For $\mathcal N=4$ SYM with gauge group $SU(N)$:
$\kappa=0$ (conformality), so the curvature component of
$\nabla^{\mathrm{hol}}$ vanishes. The filtered all-genus lift of
$\Theta_{\mathrm{cel}}$ is governed by the Mellin classes
$\Obs_{g,r}^{\mathrm{Mell}}(\mathcal N=4)$, and the line comparison is governed by
$\Obs_{\mathcal N=4}^{Y}\in H^2(\Cone(\upsilon_{\mathcal N=4}))$.
On the locus
\[
\Obs_{g,r}^{\mathrm{Mell}}(\mathcal N=4)=0\quad\text{for all }g,r,
\qquad
\Obs_{\mathcal N=4}^{Y}=0,
\]
the celestial Yangian $Y_\hbar(\mathfrak{gl}_N)$ controls the line operators, the
corresponding associated graded $R$-matrix has leading term
$R_{\mathrm{cel}}(z)=1+\hbar\,k\,\Omega_{\mathrm{tr}}/z+\cdots$, and the modular expansion
$R^{\mathrm{mod}}(z;\hbar)=\sum_{g\ge 0}\hbar^{2g}r_g(z)$ is the filtered lift of the
loop-corrected celestial residues. If all relevant $H^1$ groups of the Mellin and Yangian
cones vanish, this lift is unique.
\end{proposition}
