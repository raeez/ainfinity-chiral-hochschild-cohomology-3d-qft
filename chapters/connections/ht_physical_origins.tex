\section{Physical origins}
\label{sec:physical-origins-overview}
\label{chap:physical-origins}
\label{rem:physical-origins}

String amplitudes, gauge theory partition functions, and conformal
blocks each produce bar-cobar structures: the generating function of
$n$-point functions is the bar complex; the factorization of
amplitudes under degeneration is the coproduct. The physical interfaces
below are formulated as comparison problems with named algebraic input
data.

\subsection{4d/2d correspondence algebras}
\label{sec:4d2d}

\subsubsection{The 4d/2d correspondence (BLLPRR)}

\begin{construction}[4d/2d correspondence (BLLPRR)]\label{constr:4d2d}
Starting from a 4d $\mathcal{N} = 2$ superconformal field theory $\mathcal{T}_{4d}$:
\begin{enumerate}[label=(\roman*)]
\item Select the \emph{Schur supercharge} $\mathbb{Q} = Q_1^- + \widetilde{S}^{1\dot{-}}$, a specific nilpotent supercharge in the $\mathcal{N}=2$ superconformal algebra ($\mathbb{Q}^2 = 0$).
\item Pass to the cohomology $H^*(\mathbb{Q})$ of local operators inserted at a fixed plane $\mathbb{R}^2 \subset \mathbb{R}^4$.
\item The resulting algebra $\mathcal{A}[\mathcal{T}_{4d}] = H^*(\mathbb{Q}, \text{local ops})$ carries the structure of a 2d vertex algebra (chiral algebra).
\end{enumerate}

This construction, due to Beem--Lemos--Liendo--Peelaers--Rastelli--van~Rees~\cite{beem-et-al}, produces vertex algebras from 4d SCFTs.
It is distinct from the $\Omega$-background/Nekrasov partition-function
approach used in the AGT correspondence below.
\end{construction}

\begin{theorem}[4d/2d is \texorpdfstring{$\Einf$}{E-infinity}-chiral]\label{thm:4d2d-einf}
The 2d chiral algebra $\mathcal{A}[\mathcal{T}_{4d}]$ obtained from the 4d/2d
correspondence is an $\Einf$-chiral algebra (vertex algebra).

The OPE of the resulting chiral algebra is generically \emph{non-commutative}: for instance, the 4d/2d image of $SU(2)$ $\mathcal{N}=4$ SYM is the small $\mathcal{N}=4$ superconformal vertex algebra at $c = -9$, which contains $\widehat{\mathfrak{sl}}_2$ at level $k = -3/2$ (an admissible, non-integral level; determined by $c = 3k/(k+2)$) as its R-symmetry current subalgebra. The $\Einf$ structure arises from the full vertex algebra axioms (locality, associativity of OPE), not from commutativity.
\end{theorem}

\begin{example}[Class \texorpdfstring{$\mathcal{S}$}{S}]\label{ex:class-s}
For class $\mathcal{S}$ theories of type $A_{n-1}$ on a Riemann surface $C$:
\[
\mathcal{A}[\mathcal{T}_{A_{n-1}}(C)] = \mathcal{W}^{-n}(\mathfrak{sl}_n)
\]
the W-algebra associated to $\mathfrak{sl}_n$ at the critical level $k = -n = -h^\vee$. At this special level the Sugawara construction degenerates and $\mathcal{W}^{-h^\vee}(\mathfrak{sl}_n)$ coincides with the Feigin--Frenkel center $\mathfrak{z}(\widehat{\mathfrak{g}})$, a commutative vertex algebra (see Feigin--Frenkel \cite{FF} for the center isomorphism and Arakawa \cite{Ara15} for related rationality results). The dependence on the Riemann surface $C$ and its pants decomposition appears at the level of conformal blocks and modules, not the algebra itself.
\end{example}


\subsection{Non-commutative Chern--Simons theory}
\label{sec:nc-chern-simons}

\subsubsection{Chern--Simons as source of chiral algebras}

\begin{construction}[CS boundary algebra]\label{constr:cs-boundary}
3d Chern--Simons theory with gauge group $G$ at level $k$ on $M^3 = \Sigma \times \bR_+$
produces:
\begin{enumerate}[label=(\roman*)]
\item On the boundary $\Sigma$: WZW model at level $k$.
\item The chiral algebra is the affine Kac--Moody algebra $\hat{\mathfrak{g}}_k$.
\end{enumerate}
\end{construction}

\begin{remark}[Non-commutative Chern--Simons comparison problem]\label{rem:nc-cs-comparison}
Noncommutative Chern--Simons theory on $\bR^3_\theta$ produces
$\Eone$-chiral algebras:
\begin{enumerate}[label=(\roman*)]
\item The boundary theory is a non-commutative WZW model.
\item The OPE is R-twisted with $R$ depending on the non-commutativity parameter $\theta$.
\item At $\theta = 0$, the structure reduces to the standard $\Einf$-chiral Kac--Moody.
\end{enumerate}
\end{remark}

\begin{remark}[Seiberg--Witten input]
The $R$-twisted OPE structure is consistent with Definition~\ref*{V1-def:chiral-ass-operad}; deriving it from non-commutative CS requires the Seiberg--Witten deformation quantization map~\cite{SW99}.
\end{remark}


\subsubsection{\texorpdfstring{Noncommutative geometry and associative / topological $\mathsf{E}_1$ structures}{Noncommutative geometry and associative / topological E1 structures}}

\begin{construction}[NC torus as associative deformation]
\label{constr:nc-torus-e1}
\index{noncommutative torus!topological $E_1$ structure@topological $\mathsf{E}_1$ structure}
The noncommutative torus $T^2_\theta$ ($\theta \in \mathbb{R} \setminus \mathbb{Q}$), with Moyal star product $f \star_\theta g$ deforming $\{x,y\}=\theta$, carries a topological $\Eone$-algebra structure ($VU = e^{2\pi i\theta}UV$, associative but noncommutative). This is \emph{not} a Beilinson--Drinfeld $\Eone$-chiral algebra on a curve.
\end{construction}

\begin{remark}[Koszul duality as Morita equivalence]
\label{rem:nc-torus-koszul-morita}
\index{Morita equivalence!Koszul duality interpretation}
The noncommutative-torus comparison asks for a topological
$\Eone$-Koszul duality model whose duality map exchanges the deformation
parameter:
\[
\mathcal{A}_\theta^! \;\cong\; \mathcal{A}_{1/\theta}.
\]
This coincides with Rieffel's Morita equivalence $T^2_\theta \sim_{\mathrm{Morita}} T^2_{1/\theta}$
for irrational $\theta$, reinterpreted as $\Eone$-Koszul duality.
The bar complex $\barB(\mathcal{A}_\theta)$ computes the cyclic homology
$\mathrm{HC}_*(\mathcal{A}_\theta)$, which by Connes' theorem equals the de~Rham
cohomology $H^*_{\mathrm{dR}}(T^2)$, independent of $\theta$ (topological invariance
of cyclic homology for smooth deformations).
\end{remark}

\begin{remark}[Morita input]
Rieffel Morita equivalence and Connes' $\mathrm{HC}_*(A_\theta)$ computation are established. The comparison consists of constructing the topological $\Eone$ bar complex for $\mathcal{A}_\theta$ and proving that its cohomology recovers the Koszul dual at $1/\theta$.
\end{remark}


\subsection{Gauge theory and D-branes}
\label{sec:gauge-d-branes}
\subsubsection{D-brane vertex algebras}

\begin{construction}[Open string VOA]\label{constr:open-string-voa}
Open strings on D-branes in type IIB produce boundary chiral algebras depending on the brane configuration.
\end{construction}

\begin{remark}[D-brane algebras are \texorpdfstring{$\Eone$}{E1}]\label{rem:dbrane-e1}
Open string vertex algebras on non-trivial D-brane configurations are generically $\Eone$-chiral: Chan--Paton factors give matrix-valued OPE ($\Phi^{ij}(z)\Psi^{k\ell}(w) \sim \delta^{jk}\cdots$), breaking $\Einf$-commutativity.
\end{remark}

\begin{remark}[String-field-theory input]
Matrix-valued OPE from Chan--Paton factors is compatible with $\Eone$-chiral structure (Definition~\ref*{V1-def:chiral-ass-operad}); deriving it from open string field theory requires the cubic string-field-theory vertex of~\cite{Zwi93}.
\end{remark}

\subsubsection{The deformed Steinberg correspondence: noncommutativity, Morita, and Chan--Paton}
\label{sec:deformed-steinberg}

\providecommand{\FM}{\mathrm{FM}}
\providecommand{\Conf}{\mathrm{Conf}}
\providecommand{\BM}{\mathrm{BM}}
\providecommand{\HC}{\mathrm{HC}}
\providecommand{\dR}{\mathrm{dR}}

\begin{remark}[Three deformations of the Steinberg correspondence]
\label{rem:three-deformations}
\index{Steinberg correspondence!deformed}
\index{noncommutative Chern--Simons!Steinberg perspective}
\index{Chan--Paton factors!Steinberg perspective}
The Steinberg correspondence for a commutative chiral algebra $\cA$ on a curve
lives over the product $\FM_k(\bC) \times \Conf_k(\bR)$ of holomorphic and
topological configuration spaces. The bar differential inhabits the
$\bC$-factor; the deconcatenation coproduct inhabits the $\bR$-factor.
The comparison problems in
Remarks~\ref{rem:nc-cs-comparison}, \ref{rem:nc-torus-koszul-morita},
and \ref{rem:dbrane-e1} are three independent deformations of this
correspondence, each touching a different factor.

\begin{enumerate}[label=(\Roman*)]
\item \emph{Noncommutative Chern--Simons} (Remark~\ref{rem:nc-cs-comparison}).
 Deform $\bC$ to the Moyal plane $\bC_\theta$.
 The Fulton--MacPherson compactification $\FM_k(\bC)$ becomes the
 noncommutative FM space $\FM_k(\bC_\theta)$, whose boundary strata carry
 $R$-twisted factorization data.
 The topological factor $\Conf_k(\bR)$ is unchanged: the boundary
 algebra remains $\Eone$, but the holomorphic OPE acquires Moyal phases
 $e^{i\theta\, \partial_{z_i} \wedge \partial_{z_j}}$.
 The resulting bar differential $d_B^\theta$ is the NC deformation
 of the standard bar differential.

\item \emph{Koszul--Morita duality} (Remark~\ref{rem:nc-torus-koszul-morita}).
 For the NC torus $\cA_\theta$, the Steinberg self-intersection
 $\cA_\theta \times_{\cA_\theta\text{-bimod}} \cA_\theta$
 is Morita-equivalent to $\cA_{1/\theta}$ (Rieffel's theorem),
 recast as the Steinberg involution $\theta \mapsto 1/\theta$.
 This is the \emph{same} involution that Koszul duality implements on
 $\Eone$-algebras: dualise the bar complex, invert the parameter.
 The three identifications (Koszul dual, Steinberg involution,
 Morita equivalence) collapse to one.

\item \emph{Chan--Paton enrichment} (Remark~\ref{rem:dbrane-e1}).
 Tensor the correspondence with $\End(V)$ for a Chan--Paton bundle $V$.
 The convolution algebra becomes matrix-valued.
 Commutativity breaks because $\End(V)$ is noncommutative: the
 $\Einf$-structure on the holomorphic factor cannot descend through a
 noncommutative tensor factor, leaving only $\Eone$.
 The topological factor enforces left-to-right ordering of matrix
 multiplication along $\bR$.
\end{enumerate}
In all three cases, the $(-1)$-shifted symplectic structure on the
Steinberg variety $\mathfrak{S}_b$ (Volume~I, Theorem~C, the Lagrangian splitting of genus-$g$ cohomology) persists through the
deformation. This is the geometric reason that each deformation produces a
well-defined bar complex: the symplectic pairing controls the signs in the
$\Ainf$-relations, and its deformation-invariance ensures the relations
remain consistent.
\end{remark}

\begin{proposition}[NC cyclic invariance from Steinberg deformation-invariance]
\label{prop:nc-cyclic-steinberg}
\index{cyclic homology!noncommutative torus}
\index{Steinberg variety!deformation invariance}
For the noncommutative torus $\cA_\theta$ with irrational $\theta$,
the bar complex $\barB(\cA_\theta)$ computes cyclic homology:
\[
 H_\bullet\bigl(\barB(\cA_\theta)\bigr) \;\cong\; \HC_*(\cA_\theta).
\]
By Connes' theorem, $\HC_*(\cA_\theta) \cong H^*_\dR(T^2)$ independent
of $\theta$. From the Steinberg perspective: the Borel--Moore homology
$H_*^\BM(\mathfrak{S}_b^\theta)$ of the NC Steinberg variety is
deformation-invariant because the $(-1)$-shifted symplectic structure
on $\mathfrak{S}_b^\theta$ is itself deformation-invariant.
\end{proposition}

\begin{proof}
The bar complex $\barB(\cA_\theta)$ is, by construction, the chain complex
computing the derived tensor product
$\cA_\theta \otimes_{\cA_\theta \otimes \cA_\theta^{\mathrm{op}}}^{\bL}
\cA_\theta$, i.e.\ cyclic homology $\HC_*(\cA_\theta)$.
Connes proved $\HC_*(\cA_\theta) \cong H^*_\dR(T^2)$ for irrational $\theta$
by exhibiting an explicit trace on $\cA_\theta$ whose induced map on cyclic
homology is an isomorphism independent of $\theta$.

For the Steinberg interpretation: the NC Steinberg variety
$\mathfrak{S}_b^\theta = T^*_{\cA_\theta} (\cA_\theta \otimes
\cA_\theta^{\mathrm{op}})$ carries a $(-1)$-shifted symplectic form
$\omega_{\mathfrak{S}}^\theta$ obtained by cotangent lifting.
Kontsevich's formality theorem for NC deformations guarantees
that the $L_\infty$-quasi-isomorphism type of the polyvector fields
$\mathrm{T}_{\mathrm{poly}}$ is deformation-invariant,
hence $\omega_{\mathfrak{S}}^\theta$ is cohomologous to
$\omega_{\mathfrak{S}}^0$ for all $\theta$.
The BM homology of the Steinberg is the convolution algebra, which computes
the bar homology. Since the symplectic form controlling the bar signs
is deformation-invariant, so is the homology.
\end{proof}

\begin{proposition}[Chan--Paton breaking: $\Einf \to \Eone$ via matrix ordering]
\label{prop:chan-paton-breaking}
\index{Chan--Paton factors!E-infinity breaking@$\mathsf{E}_\infty$ breaking}
\index{Steinberg variety!Chan--Paton enrichment}
Let $V$ be a finite-dimensional vector space (the Chan--Paton bundle) and let
$\mathfrak{S}_b$ be the Steinberg variety of a commutative chiral algebra
$\cA$. The Chan--Paton--enriched Steinberg variety
$\mathfrak{S}_b^{\mathrm{CP}} = \mathfrak{S}_b \times \End(V)$
has convolution algebra
\[
 H_*^\BM\bigl(\mathfrak{S}_b^{\mathrm{CP}}\bigr)
 \;\cong\;
 H_*^\BM(\mathfrak{S}_b) \otimes \End(V),
\]
where the tensor product uses \emph{ordered} (left-to-right) multiplication
in $\End(V)$ along the $\bR$-direction. This convolution algebra is $\Eone$
but not $\Einf$: the matrix factor forces an ordering on compositions that
the holomorphic factor alone does not require.
\end{proposition}

\begin{proof}
Convolution on $\mathfrak{S}_b$ composes correspondences over the base.
Tensoring with $\End(V)$ replaces scalar-valued correspondences by
matrix-valued ones: a correspondence of degree $k$ now carries a $k$-fold
composition $M_1 \cdots M_k$ in $\End(V)$. Since matrix multiplication
is associative but not commutative, the $k$-fold composition depends on
the linear order of the factors. This order is supplied by $\Conf_k(\bR)$
in the topological direction.

Concretely, the $\Einf$-structure on $H_*^\BM(\mathfrak{S}_b)$ uses the
fact that the convolution product is independent of the ordering of
configuration points in $\FM_k(\bC)$ (holomorphic commutativity).
After tensoring with $\End(V)$, the product of two elements
$(\alpha \otimes M)$ and $(\beta \otimes N)$ satisfies
$(\alpha \otimes M) \cdot (\beta \otimes N)
= (\alpha \cdot \beta) \otimes MN
\neq (\alpha \cdot \beta) \otimes NM$
in general. The symmetric group action on $\FM_k(\bC)$ permutes the
$\End(V)$-labels, changing $M_1 \cdots M_k$ to $M_{\sigma(1)} \cdots
M_{\sigma(k)}$, which is generically different. Hence only the identity
permutation contributes, reducing $\Einf$ to $\Eone$.

The $(-1)$-shifted symplectic form on $\mathfrak{S}_b$ extends to
$\mathfrak{S}_b^{\mathrm{CP}}$ by taking the trace pairing on $\End(V)$:
$\omega^{\mathrm{CP}} = \omega_{\mathfrak{S}} \otimes \mathrm{tr}$.
This remains nondegenerate (since $\mathrm{tr}$ is nondegenerate on
$\End(V)$), so the bar complex of the enriched algebra is well-defined.
\end{proof}

\begin{remark}[Ordered multiplication and the two-colour architecture]
\label{rem:ordered-mult-two-colour}
The ordered multiplication in the $\bR$-direction exhibited above is
the $\Eone$ structure of $\SCchtop$. In the bar-cobar framework, this
$\Eone$ colour produces the \emph{ordered} bar complex
$\barB^{\mathrm{ord}}(\cA)$ on $\Conf_k^<(\bR)$, whose linear dual
is the line-operator Koszul dual~$\cA^!_{\mathrm{line}}$. On the
chirally Koszul locus, $\cA^!_{\mathrm{line}}$ carries the dg-shifted
Yangian package (Theorem~\ref{thm:yangian-recognition}; see
Part~\textup{VII}, \S\ref{ch:ordered-associative-chiral-kd} for the full
construction). The Chan--Paton mechanism thus provides the physical
origin of the open-colour Koszul duality: the matrix ordering in
$\End(V)$ forces the passage from the symmetric bar
$\barB^{\Sigma}(\cA)$ to the ordered bar
$\barB^{\mathrm{ord}}(\cA)$, and this passage is precisely the
two-colour architecture of the $\SCchtop$-algebra datum.
\end{remark}

\begin{remark}[Two inequivalent chiral-Yangian definitions, and which
one the HT landscape produces]
\label{rem:ht-yangian-two-definitions}
\index{Yangian!two chiral definitions}
Two inequivalent definitions of ``chiral Yangian'' appear across
this volume and must be distinguished whenever the recognition
Theorem~\ref{thm:yangian-recognition} is invoked. The weaker
notion (Definition~\ref{V1-def:e1-chiral-yangian}) asks only for an
$\Eone$-factorization algebra equipped with an RTT-style spectral
R-matrix $R(z)$ satisfying the quantum Yang--Baxter equation; the
stronger notion (Definition~\ref{def:chiral-yangian-datum}) asks
for the full datum $(\cA, S(z), \Delta^{\mathrm{ch}}, \{m_k\})$
plus the modular Maurer--Cartan tower
$D_\cA \Theta_\cA + \tfrac{1}{2}[\Theta_\cA,\Theta_\cA]=0$.
Their equivalence on the chirally Koszul locus is the content
of Theorem~\ref{thm:yangian-recognition}; off that locus the
equivalence requires an independent construction of the modular MC
element.
The dg-shifted Yangian package produced by the Chan--Paton
boundary mechanism above and by the Paquette--Williams boundary
VOAs below lands in the \emph{stronger} datum, because the bar
coalgebra $\barB^{\mathrm{ord}}(\cA_{\mathrm{bdy}})$ on
$\Conf_k^<(\bR)$ carries the modular MC element $\Theta^{\mathrm{oc}}$
of Theorem~\ref*{V1-thm:thqg-swiss-cheese}. Recognition theorems
whose hypotheses give only RTT + factorization (e.g.\ direct
boundary quantization of HCS without the modular MC tower) land
only in the weaker notion; an independent construction of the
modular MC element is required before the two can be identified.
The obstruction-theoretic comparison is stated in
Theorem~\ref{thm:strong-weak-chiral-yangian-comparison-obstruction}.
\end{remark}

\subsubsection{\texorpdfstring{D-brane categories from $\Eone$-module Koszul duality}{D-brane categories from E1-module Koszul duality}}

\begin{remark}[HMS from \texorpdfstring{$\Eone$}{E1}-module Koszul duality]
\label{rem:hms-koszul}
\index{homological mirror symmetry!Koszul duality interpretation}
\index{D-brane!module Koszul duality}
In string theory, D-branes on a target space $X$ form the derived category
$D^b(\mathrm{Coh}(X))$ (B-branes) or the Fukaya category $\mathrm{Fuk}(X)$
(A-branes). For a chiral algebra $\mathcal{A}$ on a curve $\Sigma$, the
module category $\mathrm{Mod}(\mathcal{A})$ is the mathematical incarnation
of the category of branes.

Under $\Eone$-Koszul duality, the module bar-cobar functor
$\Phi\colon \mathrm{Mod}(\mathcal{A}) \xrightarrow{\sim}
\mathrm{CoMod}(\mathcal{A}^!)$
of Theorem~\ref*{V1-thm:e1-module-koszul-duality} exchanges module and comodule
categories. When $X$ is a Calabi--Yau surface and $\mathcal{A}$ is the chiral
algebra of the sigma model on $X$, the comparison identifies this
functor with homological mirror symmetry (Kontsevich):
\[
D^b\bigl(\mathrm{Coh}(X)\bigr)
\;\simeq\;
D^b\bigl(\mathrm{Fuk}(X^\vee)\bigr)
\]
for a mirror pair $(X, X^\vee)$. In this interpretation, the bar-cobar
adjunction is the proposed passage from B-branes to A-branes: the bar construction
$\barB(\mathcal{A})$ computes the Ext-algebra of the diagonal brane,
and the cobar construction recovers the endomorphism algebra of the dual
collection on $X^\vee$. The required input is the identification of
sigma-model chiral algebra modules with the A- and B-brane categories.
\end{remark}

\begin{remark}[Brane-category input]
Theorem~\ref*{V1-thm:e1-module-koszul-duality} stands independently; the HMS interpretation requires identifying chiral algebra modules with brane categories (boundary CFT input) and passing from the abstract functor $\Phi$ to derived category equivalences (tilting generation).
\end{remark}

\section{Holomorphic-topological boundary conditions and 4d origins}
\label{sec:ht-boundary-physical-origins}

\index{holomorphic-topological theory|textbf}

The physical-origin lane of the E$_1$ core contains those chiral
algebras for which a holomorphic-topological field theory in the sense
of Costello--Li, or an explicit open/closed comparison datum, has been
constructed.  The standard families below enter through such named
realisations; an abstract chiral algebra in the E$_1$ core does not
thereby acquire a physical HT origin.
The bar-cobar adjunction classifies \emph{twisting morphisms}
between a boundary algebra and its Koszul dual (Volume~I, Theorem~A);
the algebraic closed-sector vertex is the chiral derived center
$\mathcal{Z}^{\mathrm{der}}_{\mathrm{ch}}(\cA)
\simeq C^\bullet_{\mathrm{ch}}(\cA,\cA)$.  Its physical bulk reading
requires the chosen HT open/closed comparison
(Volume~I, \S\ref*{V1-sec:thqg-open-closed-realization}).

\subsection{Mathematical relationships between frameworks}
\label{sec:precise-relationships-frameworks}

\subsubsection{From 4D gauge theory to 2D chiral algebras}

\begin{theorem}[Costello--Li dimensional reduction]\label{thm:costello-li-reduction}
\index{Costello--Li construction}
\cite{CL16} Consider 4D $\mathcal{N}=2$ super Yang--Mills with gauge group $G$
on $\mathbb{C}^2$. After holomorphic-topological twist:
\begin{enumerate}
\item Fields become $\bar{\partial}$-closed differential forms with values in
 $\mathfrak{g} \otimes \mathcal{O}_{\mathbb{C}^2}$

\item The action becomes BV-BRST exact up to a topological term: $S = \{Q_{\text{BRST}}, \Psi\} + S_{\mathrm{top}}$

\item Replacing $\mathbb{C}^2$ by $\mathbb{C} \times \mathbb{C}^*$ and taking $S^1$-equivariant reduction along $|w| = 1 \subset \mathbb{C}^*$
 produces a factorization algebra on $\mathbb{C}$

\item The resulting 2D theory has structure of a \emph{factorization algebra}
 $\mathcal{F}_G$, which is not a priori a chiral algebra
\end{enumerate}
\end{theorem}

\begin{remark}[Factorization algebras versus chiral algebras]\label{rem:fact-vs-chiral}
The distinction is as follows (see \BDref{§2.3, §3.2}):

\emph{Factorization algebra} $\mathcal{F}$: assigns $\mathcal{F}(U)$ to every open set $U \subset X$, with multiplication maps $\mathcal{F}(U) \otimes \mathcal{F}(V) \to \mathcal{F}(U \sqcup V)$ for disjoint $U, V$ satisfying associativity (the factorization property over multiple disjoint opens). Example: observables in any QFT.

\emph{Chiral algebra} $\mathcal{A}$: assigns a $\mathcal{D}_X$-module $\mathcal{A}_x$ at each point $x \in X$, with chiral product $j_*j^*(\mathcal{A} \boxtimes \mathcal{A}) \to \Delta_!\mathcal{A}$ having prescribed pole structure along the diagonal. Example: vertex algebras, affine Kac--Moody algebras (Virasoro action is additional structure, not part of the definition).

\emph{Relationship} \cite{BD04,CG17}.
\[\text{Chiral algebras} \xleftrightarrow{\sim} \text{Factorization algebras on curves}\]
is an equivalence (Beilinson--Drinfeld \cite[Theorem~3.4.9]{BD04}). In higher dimensions, chiral algebras embed into CG-style factorization algebras as those with $\mathcal{D}$-module structure and holomorphic dependence.
\end{remark}

\begin{proposition}[Costello--Li chiral upgrade criterion]\label{prop:CL-produces-chiral}
The Costello--Li factorization algebra upgrades to a Beilinson--Drinfeld
chiral algebra after the following three obstruction classes vanish:
\begin{enumerate}
\item the holomorphicity obstruction of the twist on the Coulomb branch;
\item the conformal-preservation obstruction for the dimensional reduction;
\item the central-charge and anomaly obstruction of the boundary theory.
\end{enumerate}

For $\mathcal{N}=4$ SYM, this is established:
Theorem~\ref{thm:cl-n4-chirality} proves that
$\mathcal{F}_{G,\mathrm{Ad}}$ is genuinely chiral, producing the
affine Kac--Moody algebra $\widehat{\mathfrak{g}}_k$ at level~$k$
determined by the gauge coupling. For $\mathcal{N}=2$ SYM with
general matter content, the obstruction classes are displayed in
Proposition~\ref{prop:cl-general-chirality-obstructions}.
\end{proposition}

\begin{remark}[Chirality inputs]
Chirality requires: (1) the twist preserves a holomorphic structure on the Coulomb branch \cite{Gai19}; (2) $T(z)$ survives the twist with vanishing anomalies; (3) the factorization algebra extends to a $\mathcal{D}$-module on $\mathrm{Ran}(X)$ (automatic for chiral algebras by \BDref{Chapter 3}, requires verification for twisted gauge theories).
\end{remark}

\begin{remark}[General matter obstruction]
Chirality is proved for $\mathcal{N}=4$ SYM (Theorem~\ref{thm:cl-n4-chirality}). See \cite{CDG20,GKW25,Zeng23} for related work. The analytic HS-sewing lane of MC5 is proved at all genera in Volume~I; the genuswise chain-level BV/BRST/bar identification is proved on the strict ML pro/$J$-adic/filtered-completed surface, and $D^{\mathrm{co}}$-level BV=bar is proved for all shadow classes. For general twisted gauge theories the remaining input is the vanishing of the three chirality obstruction classes in Proposition~\ref{prop:CL-produces-chiral}.
\end{remark}

\subsubsection{Costello--Li conditions: from factorization to chiral}

\begin{theorem}[CL chirality for \texorpdfstring{$\mathcal{N}=4$}{N=4}]
\label{thm:cl-n4-chirality}
\index{Costello--Li construction!chirality criterion}
\cite{CL16}
Let $G$ be a reductive group and consider 4d $\mathcal{N}=2$ gauge theory
with gauge group $G$ and matter in a representation $R$ on $\Sigma \times \mathbb{C}$.
After holomorphic twist, the theory produces a factorization algebra
$\mathcal{F}_{G,R}$ on $\Sigma$. When the theory has $\mathcal{N}=4$
supersymmetry (equivalently, $R = \mathrm{Ad}(G)$), the factorization algebra
is strictly chiral: $\mathcal{F}_{G,\mathrm{Ad}}$ satisfies the chiral algebra
axioms of Beilinson--Drinfeld. The resulting chiral algebra is the affine
Kac--Moody algebra $\widehat{\fg}_k$ at level $k$ determined by the gauge coupling.
\end{theorem}

\begin{proposition}[General-matter chirality obstruction classes]
\label{prop:cl-general-chirality-obstructions}
\index{Costello--Li construction!$\mathcal{N}=2$ conditions}
For $\mathcal{N}=2$ gauge theory with gauge group $G$ and matter content $R$,
the obstruction data for the chiral upgrade of
$\mathcal{F}_{G,R}$ are:
\begin{enumerate}[label=\textup{(\roman*)}]
\item the self-duality defect of $R \to R^*$ as a $G$-representation;
\item the one-loop anomaly class $c_1(R) \in H^2(\Sigma, \mathbb{Z})$;
\item the balancing class $\chi(\Sigma, R)$ in each genus.
\end{enumerate}
On the common vanishing locus, the resulting chiral algebra
$\mathcal{A}_{\mathrm{CL}}(G, R)$
on $\Sigma$ has:
\begin{itemize}
\item Central charge $c = \dim(G) \cdot (1 - h^\vee/k) + \tfrac{1}{2}\dim(R)$,
 where $k$ is the level and $h^\vee$ is the dual Coxeter number.
\item Koszul dual $\mathcal{A}_{\mathrm{CL}}(G, R)^! = \mathcal{A}_{\mathrm{CL}}(G^\vee, R^\vee)$,
 where $G^\vee$ is the Langlands dual group and $R^\vee$ the dual representation.
\item Bar complex $\barB(\mathcal{A}_{\mathrm{CL}})$ computing the derived moduli
 of solutions to the BPS equations on $\Sigma \times \mathbb{C}$.
\end{itemize}
For $G = \mathrm{GL}(N)$ and $R = \mathrm{Ad}$, the chiral algebra is the
$\mathcal{W}_N$-algebra, and the instanton partition function on $\mathbb{C}^2$
equals the genus-$0$ bar complex evaluation; this is the AGT correspondence
(Theorem~\ref{thm:agt}).
\end{proposition}

\begin{remark}[Adjoint locus]
The three obstruction classes vanish for $\mathcal{N}=4$; the general
matter calculation is the direct verification of the same three classes.
\end{remark}

\subsubsection{Paquette--Williams boundary vertex algebras}

\begin{theorem}[Paquette--Williams 2022]\label{thm:paquette-williams-boundaries}
\cite{PW22} Consider holomorphic-topological 4D $\mathcal{N}=2$ gauge theory on
$\mathbb{C}^2$ with boundary at $z_2 = 0$. Then:
\begin{enumerate}
\item Boundary conditions $\mathcal{B}$ correspond to Lagrangian submanifolds in
 Coulomb branch $\mathcal{M}_C$

\item The boundary supports a \emph{vertex operator algebra} (VOA) $V_{\mathcal{B}}$

\item This VOA is the quantization of the symplectic reduction of $\mathcal{M}_C$
 at the boundary

\item The VOA $V_{\mathcal{B}}$ has a \emph{chiral envelope} $\mathcal{A}_{\mathcal{B}}$,
 which is a chiral algebra in the BD sense
\end{enumerate}
\end{theorem}

\begin{remark}[Bar-cobar duality for boundary VOAs]\label{rem:PW-connection}
Paquette--Williams produce vertex algebras, which are algebraic objects (Frenkel--Ben-Zvi
\cite{FBZ04}). The bar-cobar framework of Volume~I applies to their \emph{chiral envelopes}, which are
geometric objects ($\mathcal{D}$-modules).

\begin{center}
\begin{tikzcd}
\text{Boundary VOA } V_{\mathcal{B}} \ar[r, "\text{chiral envelope}"] \ar[d, "\text{PW}"]
& \text{Chiral algebra } \mathcal{A}_{\mathcal{B}} \ar[d, "\text{bar-cobar}"] \\
\text{4D HT gauge theory} \ar[r, "\text{CL reduction}"]
& \text{2D factorization algebra}
\end{tikzcd}
\end{center}

The vertex algebra $V_{\mathcal{B}}$ contains the algebraic information (OPE, modes,
etc.), while the chiral algebra $\mathcal{A}_{\mathcal{B}}$ contains the geometric
information (configuration spaces, $\mathcal{D}$-modules, sheaf cohomology).

Bar-cobar duality (Volume~I) applies to $\mathcal{A}_{\mathcal{B}}$: the bar construction $\bar{B}(\mathcal{A}_{\mathcal{B}})$ \emph{is} the factorization coalgebra on $\Ran(X)$ that encodes PW boundary data (Convention~\ref*{V1-conv:bar-coalgebra-identity}). \end{remark}

\subsection{From 4d SYM to holomorphic Chern--Simons}

\subsubsection{The holomorphic twist and localization}

\begin{definition}[Holomorphic twist of 4d \texorpdfstring{$\mathcal{N}=4$}{N=4} SYM]
Start with $\mathcal{N}=4$ super Yang--Mills in 4d with gauge group $G$, viewed as an $\mathcal{N}=2$ theory by choosing a preferred $\mathcal{N}=2$ subalgebra (Costello~\cite{costello-renormalization}). The field content before twisting:
\begin{itemize}
\item Vector multiplet: $A_\mu$ (gauge field), $\Phi^I$ (6 scalars), $\lambda,
\bar{\lambda}$ (fermions)
\item Supersymmetry: 16 supercharges transforming under $\text{Spin}(6)_R$
\end{itemize}

The \emph{holomorphic twist} (not to be confused with the fully topological Kapustin--Witten A-twist of $\mathcal{N}=4$):
\begin{enumerate}
\item Decompose $\mathbb{R}^4 = \mathbb{C} \times \mathbb{C}$ with coordinates
$(z, w)$
\item Twist the Lorentz group: $\text{SO}(4) \to \text{SO}(2)_{\text{hol}} \times
\text{SO}(2)_{\text{top}}$
\item Mix with R-symmetry to make a supercharge $Q$ scalar
\item Result: Theory is holomorphic in $z$, topological in $w$
\end{enumerate}
\end{definition}

\begin{theorem}[Perturbative localization to holomorphic data;
\ClaimStatusConditional; licensing $\alpha+\gamma$ via the
Costello--Li holomorphic-twist datum, BV gauge fixing, and
renormalisation scheme]\label{thm:ht-localization}
Fix a Costello--Li holomorphic twist, a BV gauge fixing, and a
renormalisation scheme.  In the formal neighbourhood of the \(Q\)-fixed
locus, the perturbative BV theory is equivalent to the derived moduli
problem
\[
\mathbf{R}\!\operatorname{Crit}_Q
\simeq
\bigl[\bar{\partial}_A=0,\;\mu_{\mathrm{hol}}=0\bigr]/G,
\]
and its Feynman expansion is computed by observables on this derived
holomorphic moduli problem.  This is the localization statement used
below; it is not an ordinary convergent measure-theoretic path integral.

where $\bar{\partial}_A = 0$ means:
\begin{itemize}
\item $A$ is a holomorphic connection
\item Scalars $\Phi$ satisfy holomorphic moment map equation
\end{itemize}

Its underived truncation is the moduli space of holomorphic
\(G\)-bundles with Higgs field.
\end{theorem}

\begin{proof}
The \(Q\)-exact part of the twisted BV action supplies the gauge-fixed
elliptic complex.  The linearised \(Q\)-fixed equations give
\[
\mathcal{M}_Q =
\{F_{A}^{(0,2)} = 0,\;
F_A^{(1,1)} + [\Phi, \Phi^*] = 0,\;
\bar{\partial}_A \Phi = 0\},
\]
with derived structure furnished by the BRST/BV complex.  The
perturbative localization theorem is the statement that the resulting
factorization algebra is computed from this derived elliptic moduli
problem after the chosen renormalisation data are installed.
\end{proof}

\subsubsection{Holomorphic Chern--Simons as effective theory}

\begin{definition}[Holomorphic Chern--Simons action]
\index{holomorphic Chern--Simons}
On a Calabi--Yau 3-fold $X$ with holomorphic volume form $\Omega \in \Omega^{3,0}(X)$, the
holomorphic Chern--Simons action is:
\[S_{\text{HCS}}[A] = \frac{1}{2}\int_{X} \Omega \wedge \operatorname{Tr}\left(A
\wedge \bar{\partial} A + \frac{2}{3} A \wedge A \wedge A\right)\]

where $A \in \Omega^{0,1}(X, \mathfrak{g}_{\mathbb{C}})$. Since $A \in \Omega^{0,1}$ and $\bar\partial A \in \Omega^{0,2}$, the forms $A \wedge \bar\partial A \in \Omega^{0,3}$ and $A^3 \in \Omega^{0,3}$, so $\Omega \wedge \text{Tr}(\cdots)$ is a $(3,3)$-form, hence integrable over~$X$. The factor $\frac{1}{2}$ follows the convention of Costello~\cite{costello-renormalization}.
\end{definition}

\begin{theorem}[HCS from dimensional reduction]\label{thm:hcs-dimensional-reduction}
Holomorphic Chern--Simons \cite{costello-renormalization} arises from the holomorphic twist of 4d SYM by:
\begin{enumerate}
\item The holomorphic twist on $\mathbb{C}^2$ yields a theory holomorphic in $z$, topological in~$w$
\item Compactify the topological direction $w$ on $S^1$ to produce a 3d theory on $\mathbb{C} \times \mathbb{R}$
\item This 3d theory is holomorphic Chern--Simons
\end{enumerate}

The 4d holomorphic twist yields a holomorphic-topological theory on $\mathbb{C}^2$. Compactifying one direction produces a 3d theory; the holomorphic volume form on the resulting CY 3-fold $X$ is a $(3,0)$-form $\Omega \in \Omega^{3,0}(X)$. Locally, taking $X = \mathbb{C}^3$ with coordinates $(z, w, u)$:
\[\Omega = dz \wedge dw \wedge du\]
\end{theorem}

\begin{remark}[Cross-volume: the CY-geometric sources of the HT landscape]

\index{Calabi--Yau!compactification origin of 3d HT theories}

\subsubsection{HT boundary conditions}

\begin{definition}[Holomorphic-topological boundary condition]
For HCS on $X$ with boundary compactification $\bar{X}$ and boundary divisor
$D = \bar{X} \setminus X$:

A \emph{holomorphic-topological boundary condition} specifies:
\begin{enumerate}
\item Extension: Fields extend to $\bar{X}$ as holomorphic sections
\item Vanishing order: $A \in H^0(\bar{X}, \Omega^{0,1}(\mathcal{O}(-D)))$ (simple zero at $D$)
\item Behavior: As approaching $D$, $A \sim (\text{distance}) \cdot (\text{smooth})$
\end{enumerate}
\end{definition}

\begin{theorem}[Boundary chiral algebra from supplied HT boundary data;
\ClaimStatusConditional; licensing $\alpha+\beta+\gamma$ via a chosen
boundary chart, renormalised boundary factorization algebra,
bulk-to-boundary OPE comparison, level normalisation, and the
Costello--Dimofte--Gaiotto comparison]
\label{thm:ht-boundary-chiral-algebra}
Fix an HCS boundary condition \(B\) along \(D\), and fix a holomorphic
boundary chart \(C_D\subset D\) with coordinate \(z\).  Suppose that
the renormalised boundary factorization algebra
\(\Obs^\partial_B\), the bulk-to-boundary OPE comparison, and the
level normalisation are supplied, and that the
factorization-to-chiral comparison of Costello--Dimofte--Gaiotto
applies on \(C_D\)~\cite{CDG20}.  Then \(\Obs^\partial_B|_{C_D}\) is the
factorization algebra underlying a chiral algebra
\(\mathcal{A}_{\text{bdy}}\) whose:
\begin{enumerate}
\item \emph{Generators.} Boundary local operators $\mathcal{O}(z)$ for $z \in C_D$
\item \emph{OPE.} Determined by the renormalised boundary
 factorization product of the gauge-fixed bulk theory with boundary
 insertions
\[\mathcal{O}_1(z) \mathcal{O}_2(w) = \sum_k C_{12}^k(z-w) \mathcal{O}_k(w)\]
\item \emph{Factorization.} Extends to factorization algebra on $D$
\end{enumerate}
\end{theorem}

\begin{proof}
The hypotheses give a renormalised factorization algebra
\(\Obs^\partial_B\) on the chosen boundary chart.  Its costalks are the
boundary local-operator spaces, and its factorization products give the
operator products for pairwise disjoint configurations.  Expanding the
two-point product along the diagonal in the coordinate \(z\) gives the
displayed OPE coefficients; the central term is read only after the
level normalisation has been fixed.  The supplied
factorization-to-chiral comparison then converts
\(\Obs^\partial_B|_{C_D}\) into the chiral algebra
\(\mathcal{A}_{\text{bdy}}\).  Thus the assertion is a consequence of
the stated boundary factorization and comparison data; an
unrenormalised analytic HCS path integral is not being invoked.
\end{proof}

\subsection{Open-closed correspondence and bar-cobar duality}

\subsubsection{Open and closed sectors}

\begin{theorem}[Open-string bar identification]
\label{thm:open-string-bar}
\index{open-closed correspondence!open sector}
Let $\mathcal{A}_{\mathrm{bdy}}$ be the boundary chiral algebra of
a holomorphic-topological theory with boundary condition supporting
a Kac--Moody or W-algebra vertex algebra. Then the bar complex
$\barB^{\mathrm{ch}}(\mathcal{A}_{\mathrm{bdy}})$ computes the
open-string state space: there is a quasi-isomorphism
\[
\barB^{\mathrm{ch}}(\mathcal{A}_{\mathrm{bdy}})
\simeq
C^{\frac{\infty}{2}+\bullet}(\mathcal{A}_{\mathrm{bdy}})
\]
to the semi-infinite (BRST) complex of~$\mathcal{A}_{\mathrm{bdy}}$.
Verdier duality on $\operatorname{Ran}(X)$ converts the
bar coalgebra $\barB^{\mathrm{ch}}(\cA_{\mathrm{bdy}})$ into a
factorization algebra whose cohomology, on the Koszul locus, is the
Koszul dual~$\cA_{\mathrm{bdy}}^!$
\textup{(}Theorem~\textup{\ref*{V1-thm:verdier-bar-cobar})}.
The coalgebra encodes boundary data; its Verdier dual algebra
encodes the dual boundary condition~$\cA_{\mathrm{bdy}}^!$.
The bulk observables are the chiral derived center
$Z^{\mathrm{der}}_{\mathrm{ch}}(\cA_{\mathrm{bdy}})$, a distinct object
(see Remark~\ref{rem:bar-not-bulk}).
\end{theorem}

\begin{proof}
For Kac--Moody boundary conditions
$\mathcal{A}_{\mathrm{bdy}} = \widehat{\fg}_k$ ($k \neq -h^\vee$),
Theorem~\ref*{V1-thm:bar-semi-infinite-km} provides the quasi-isomorphism
$\barB^{\mathrm{ch}}(\widehat{\fg}_k) \simeq C^{\frac{\infty}{2}+\bullet}(\widehat{\fg}_k)$
via the PBW filtration and Arkhipov's functor.
For W-algebra boundary conditions
$\mathcal{A}_{\mathrm{bdy}} = \mathcal{W}_k(\fg)$,
Theorem~\ref*{V1-thm:bar-semi-infinite-w} gives the analogous
identification through Drinfeld--Sokolov reduction.
In both cases, the semi-infinite complex is the mathematical
formulation of the open-string BRST complex: its cohomology
computes the physical state space of open strings ending on
the holomorphic-topological boundary condition.

For the Verdier duality statement:
Theorem~\ref*{V1-thm:verdier-bar-cobar} shows that $\mathbb{D}_{\operatorname{Ran}}$
maps the bar coalgebra to a factorization algebra whose cohomology
on the Koszul locus is $\cA_{\mathrm{bdy}}^!$.
Verdier duality converts the coalgebra to an algebra; this is
distinct from the cobar functor $\Omega$ (the left adjoint to~$\barB$,
which recovers~$\cA$ by bar-cobar inversion).
\end{proof}

\begin{remark}[Closed-string cobar comparison problem]
\label{rem:closed-string-cobar}
\index{open-closed correspondence!closed sector}
Let $\mathcal{C}_{\mathrm{bulk}}$ be the bulk chiral coalgebra of
a holomorphic-topological string theory. Then:
\begin{enumerate}
\item \emph{Closed strings.} The cobar complex
 $\Omega^{\mathrm{ch}}(\mathcal{C}_{\mathrm{bulk}})$ computes the
 closed-string field theory state space in the sense of
 Zwiebach~\cite{Zwi93}.
\item \emph{Open-closed duality.} The bar-cobar adjunction
 supplies the algebraic comparison map
 $\barB(\mathcal{A}_{\mathrm{bdy}}) \to
 \Omega(\mathcal{C}_{\mathrm{bulk}})$. This is a map between twisting data
 (the bar complex classifying couplings) and a cobar reconstruction
 (recovering the closed-string algebra from its coalgebra); the bulk
 \emph{observables} are the chiral derived center
 $C^\bullet_{\mathrm{ch}}(\cA_{\mathrm{bdy}},\cA_{\mathrm{bdy}})$,
 a separate object.
\end{enumerate}
\end{remark}

\begin{remark}[Open and closed inputs]
Open sector: boundary vertex operators are elements of $\mathcal{A}_{\mathrm{bdy}}$, and off-shell amplitudes lie in $\barB^{\mathrm{ch}}(\mathcal{A}_{\mathrm{bdy}})$. Closed sector: bulk local operators are classified by the chiral derived center $C^\bullet_{\mathrm{ch}}(\cA_{\mathrm{bdy}},\cA_{\mathrm{bdy}})$ (Volume~I, Theorem~\ref*{V1-thm:thqg-swiss-cheese}); the field-theoretic input is the identification of the closed-string state space with the cobar of a bulk coalgebra.
\end{remark}

\begin{remark}[Closed-string input]
Open-string side proved (Theorems~\ref*{V1-thm:bar-semi-infinite-km}, \ref*{V1-thm:bar-semi-infinite-w}). MC4 is proved in Volume~I and the analytic lane of MC5 is proved at all genera; the genuswise chain-level BV/BRST/bar identification is proved on the strict ML completed/pro surface, and $D^{\mathrm{co}}$-level BV=bar is proved for all shadow classes. The remaining closed-string input is the field-theoretic state-space identification, not the Vol~I MC5 comparison.
\end{remark}

\begin{remark}[Two-colour decomposition of the open sector]
\label{rem:open-sector-two-colour}
The open sector has two algebraically distinct aspects,
corresponding to the two colours of $\SCchtop$.
\begin{enumerate}[label=(\roman*)]
\item \emph{Chiral bar} (closed colour). The symmetric bar complex
 $\barB^{\Sigma}(\cA_{\mathrm{bdy}})$ is parametrised by the
 Fulton--MacPherson space $\FM_k(\C)$ and uses all OPE products with
 $\Sigma_k$-coinvariants. Its Verdier dual yields the chiral Koszul
 dual~$\cA^!_{\mathrm{bdy}}$, the algebra of the dual boundary
 condition. This is the bar complex of Volume~I, Theorem~A.
\item \emph{Ordered bar} (open colour). The ordered bar complex
 $\barB^{\mathrm{ord}}(\cA_{\mathrm{bdy}})$ is parametrised by the
 ordered configuration space $\Conf_k^<(\bR)$ and retains the linear
 ordering from the topological direction. Its linear dual yields the
 line-operator Koszul dual~$\cA^!_{\mathrm{line}}$, and on the chirally
 Koszul locus this carries the dg-shifted Yangian package
 (Theorem~\ref{thm:yangian-recognition}).
\end{enumerate}
The descent from the ordered bar to the symmetric bar is
$R$-matrix twisted: $\barB^{\Sigma}_n \simeq
(\barB^{\mathrm{ord}}_n)^{R\text{-}\Sigma_n}$. For vertex
algebras with OPE poles (all standard examples: $\widehat{\fg}_k$,
$\operatorname{Vir}_c$, $\cH_k$), the $R$-matrix is nontrivial,
so the ordered bar carries strictly more information than the
symmetric bar. The full two-colour Koszul duality theorem
(Theorem~\ref{thm:two-color-master}) and the ordered
associative chiral Koszul duality
(\S\ref{ch:ordered-associative-chiral-kd}) develop this architecture in
detail.
\end{remark}

\subsection{$\mathcal{W}$-algebras from Hitchin moduli}

\subsubsection{The Higgs branch and Hitchin system}

\begin{definition}[Hitchin moduli space]
For a Riemann surface $\Sigma_g$ of genus $g$ and gauge group $G$, the Hitchin
moduli space is:
\[\mathcal{M}_{\text{Hit}}(\Sigma_g, G) = \{(E, \Phi) : \bar{\partial}_E \Phi = 0\} /
\sim\]

where $E \to \Sigma_g$ is a holomorphic $G$-bundle, $\Phi \in H^0(\Sigma_g, \operatorname{End}(E) \otimes K_{\Sigma_g})$ is the Higgs field, and $\sim$ denotes gauge equivalence.
\end{definition}

\begin{proposition}[Virasoro from \texorpdfstring{$SL_2$}{SL_2}-Hitchin]\label{prop:virasoro-from-hitchin}
For $G = SL_2$, the chiral algebra of local operators on the Hitchin moduli space $\mathcal{M}_{\mathrm{Hit}}(\Sigma_g, SL_2)$ is the Virasoro algebra:
\[\mathcal{A}_{\mathrm{local}}(\mathcal{M}_{\mathrm{Hit}}(\Sigma_g, SL_2)) \cong \mathrm{Vir}_c.\]
\end{proposition}

\begin{proof}
The Hitchin system for $SL_2$ is the cotangent bundle $T^*\mathrm{Bun}_{SL_2}(\Sigma_g)$; quantization of the coordinate ring via the Beilinson--Drinfeld construction \cite{BD04} produces the Virasoro vertex algebra at the level determined by the symplectic structure.
\end{proof}

\begin{remark}[Hitchin-to-\texorpdfstring{$\mathcal{W}$}{W} comparison problem]
\label{rem:w-algebra-hitchin}
For general $G$, the comparison map from quantized local operators on
$\mathcal{M}_{\mathrm{Hit}}(\Sigma_g, G)$ to the Drinfeld--Sokolov
$\mathcal{W}$-algebra has target
\[
\mathcal{W}(G),
\]
the W-algebra associated to~$G$. For $G = SL_N$ this target is the
$\mathcal{W}_N$ algebra with generators
$T(z)$, $W^{(3)}(z)$, \ldots, $W^{(N)}(z)$
of conformal weights $2, 3, \ldots, N$.
\end{remark}

\begin{remark}[AGT localization input]
The $SL_2$ case is Beilinson--Drinfeld \cite{BD04}. For general
$G$, the comparison uses AGT localization together with the
Schiffmann--Vasserot \cite{SV13}, Maulik--Okounkov \cite{MO19}, and
Arakawa \cite{Ara07} constructions.
\end{remark}

\begin{remark}[AGT localization dictionary]
Place 4d $\mathcal{N}=2$ gauge theory with gauge group $G$ on
$\mathbb{R}^2_\epsilon \times \Sigma_g$ with $\Omega$-background
parameters $(\epsilon_1, \epsilon_2)$.  Equivariant localization
computes the Nekrasov partition function by fixed-point data on the
instanton/Hitchin moduli problem.  Under the AGT comparison hypotheses,
and in the Nekrasov--Shatashvili limit $\epsilon_2 \to 0$, this
equivariant partition function matches the $\mathcal{W}(G)$ conformal
block on $\Sigma_g$:
\[Z_{4d}^{\Omega}\big|_{\epsilon_2 \to 0}
   \;\sim_{\mathrm{AGT}}\; Z_{\mathcal{W}}[\Sigma_g].\]
The operator correspondence (4d line operators $\leftrightarrow$ 2d
vertex operators \cite{DNP25}, 4d surface operators $\leftrightarrow$
2d extended operators) is part of this comparison datum; it does not
turn the unregularised 4d path integral into a literal integral over
\(\mathcal{M}_{\mathrm{Hit}}\).
\end{remark}

\subsubsection{Bar-cobar for $\mathcal{W}$-algebras}

\begin{theorem}[Genus-graded bar complex]
\label{thm:genus-graded-bar}
The full bar complex has a genus-graded deformation
\[\bar{B}^{\mathrm{ch}}_\hbar(\mathcal{A}) = \bar{B}^{\mathrm{ch}}(\mathcal{A})[[\hbar]]\]
with differential
\[d_\hbar = d_0 + \hbar d_1 + \hbar^2 d_2 + \cdots\]
where $d_g$ denotes the genus-$g$ component of $\Dg{\bullet}$, i.e.\ the genus-$g$ contribution from the modular operad decomposition.
The nilpotency $(d_\hbar)^2 = 0$ holds as a formal power series identity.
\end{theorem}

\begin{proof}

$\bar{B}^{\mathrm{full}}(\mathcal{A}) = \bigoplus_g \bar{B}^{(g)}(\mathcal{A})$

the modular operad structure. Setting $\hbar^g$ as the genus-counting
parameter, the relation $(d_\hbar)^2 = 0$ is the genus-by-genus
identity $\sum_{g_1 + g_2 = g} d_{g_1} \circ d_{g_2} = 0$
for each $g$, which is proved in Theorem~\ref*{V1-thm:genus-induction-strict}.
\end{proof}

In the Costello--Li framework, $\hbar = \epsilon_1$ and $d_k$ is the $k$-loop correction; this identification requires gauge-theoretic input.
\end{remark}

\begin{remark}[Evidence for items \textup{(2)} and \textup{(3)} at generic~$k$]

\end{remark}

\section{Holomorphic-topological field theories and BV-BRST structure}
\label{sec:bv-ht-physics}

\subsection{Gaiotto's framework: from 4d to 2d}

\begin{definition}[Holomorphic-topological (HT) theory]
A holomorphic-topological field theory on a complex surface $\Sigma$ is holomorphic in one complex direction and topological in the other. Fields are sections of $\mathcal{O}$-modules satisfying $\bar{\partial}_{\bar{z}} \phi = 0$ (holomorphic) and $Q_{\text{top}} \phi = 0$ (topologically BRST-closed).
\end{definition}

\begin{theorem}[HT theory from 4d \texorpdfstring{$\mathcal{N}=4$}{N=4} SYM;
\ClaimStatusConditional; licensing $\alpha+\gamma$ via the Costello
holomorphic-twist datum, BV gauge fixing, dimensional reduction, and
$\hypKMHTBV$]
\label{thm:ht-from-n4-sym}
Starting with 4d $\mathcal{N}=4$ super Yang--Mills with gauge group
$G$, the Costello holomorphic twist~\cite{costello-gaiotto}, followed
by BV gauge fixing and dimensional reduction to a Riemann surface
\(\Sigma\), produces the perturbative holomorphic BF theory whose
derived classical moduli problem is the \(\bar\partial\)-connection
stack on \(\Sigma\).  This is the dimensional reduction of
holomorphic Chern--Simons to \(\Sigma\).

The action is:
\[S_{\text{BF}} = \int_{\Sigma} \operatorname{Tr}\!\left(B \wedge \bar{\partial} \mathbf{A} + B \wedge \mathbf{A} \wedge \mathbf{A}\right)\]

where $\mathbf{A} = c + A + \cdots$ denotes the full BV field with $c \in \Omega^{0,0}(\Sigma, \mathfrak{g})$ (ghost) and $A \in \Omega^{0,1}(\Sigma, \mathfrak{g})$ (connection), and $B \in \Omega^{1,0}(\Sigma, \mathfrak{g})$ is the adjoint-valued Lagrange multiplier.
On a Riemann surface, $\Omega^{0,2}(\Sigma) = 0$, so the classical fields
$A \in \Omega^{0,1}$ satisfy $\bar{\partial}A = 0$ and $A \wedge A = 0$ identically. The non-trivial
content of the action arises from the BV extension: the ghost component $c \in \Omega^{0,0}$ has
$\bar{\partial}c \in \Omega^{0,1}$, giving $B \wedge \bar{\partial}c \in \Omega^{1,1}$ (the volume form).
The full holomorphic Chern--Simons action $\int \Omega \wedge \operatorname{CS}(\mathcal{A})$ lives on a
complex $3$-fold with $(3,0)$-form $\Omega$; on $\Sigma$ only the holomorphic BF action above survives the dimensional reduction.
\end{theorem}

\begin{proof}[From Costello--Gaiotto]
\emph{Step~1: The Holomorphic Twist.}

Start with $\mathcal{N}=4$ SYM on $\mathbb{R}^4 \cong \mathbb{C}^2$. The field
content includes the gauge field $A_\mu$, scalars $\Phi^I$ in the adjoint ($I=1,\ldots,6$), and fermions $\psi, \bar{\psi}$. The twist redefines the Lorentz group action by mixing with R-symmetry:
\[\text{Spin}(4) \times \text{Spin}(6)_R \to \text{Spin}(4)_{\text{new}}\]

After twisting, some bosons become fermions (ghosts), some fermions become bosons (matter), and a nilpotent supercharge $Q$ becomes scalar.

\emph{Step~2: Perturbative BV localization.}

The twisted action has $Q^2 = 0$ and:
\[S_{\text{twisted}} = \{Q, \Lambda\} + S_0\]

After BV gauge fixing, the perturbative factorization algebra is
computed in the formal neighbourhood of the \(Q\)-fixed locus; this is
not a convergent ordinary integral:
\[Z_{\mathrm{pert}}
  \rightsquigarrow
  \operatorname{Obs}\bigl(\widehat{\mathcal{M}}_{Q}\bigr).\]

The Q-fixed locus consists of holomorphic data:
\[\bar{\partial}_A = 0, \qquad \bar{\partial}_A \Phi = 0\]

These are the equations for a holomorphic bundle with holomorphic Higgs field (the Hitchin equations in the holomorphic sector).

\emph{Step~3: Volume Form.}

The holomorphic volume form $\Omega$ arises from the topological twist. On
$\mathbb{C}^2$ with holomorphic coordinates $(z,w)$:
\[\Omega = dz \wedge dw\]

On the deformed conifold $Y_\mu = \{u_1 w_2 - u_2 w_1 = \mu\} \subset \mathbb{C}^4$ (a Calabi--Yau 3-fold), $\Omega$ is the unique holomorphic $(3,0)$-form obtained via the Poincar\'{e} residue:
\[\Omega = \operatorname{Res}_{Y_\mu}\frac{du_1 \wedge du_2 \wedge dw_1 \wedge dw_2}{u_1 w_2 - u_2 w_1 - \mu}\]
\end{proof}

\subsection{Boundary conditions and chiral algebras}

\begin{theorem}[Boundary chiral algebra;
\ClaimStatusConditional; licensing $\alpha+\beta+\gamma$ via a chosen
HCS boundary condition, bulk-to-boundary OPE comparison, level
normalisation, and $\hypKMHTBV$]
\label{thm:boundary-chiral-algebra-bv}
For a holomorphic Chern--Simons boundary condition whose boundary
factorisation algebra, bulk-to-boundary OPE comparison, and level
normalisation are supplied, the boundary local operators form a chiral
algebra $\mathcal{A}_{\text{bdy}}$. Its generators are boundary local
operators, its OPE is the renormalised bulk-to-boundary OPE, and its
central term is determined by the chosen HCS level.
\end{theorem}

\begin{example}[Kac--Moody from HCS]
Holomorphic Chern--Simons with gauge group $G$ at level $k$ produces:
\[\mathcal{A}_{\text{bdy}} = \widehat{\mathfrak{g}}_k\]

the affine Kac--Moody algebra at level $k$.

The currents:
\[J^a(z) = \text{Tr}(T^a A(z))\]

satisfy:
\[J^a(z) J^b(w) \sim \frac{k \delta^{ab}}{(z-w)^2} +

\subsection{The holomorphic-topological boundary condition}

\begin{definition}[HT boundary condition]
For holomorphic Chern--Simons on a surface $\Sigma$ with boundary $\partial \Sigma$,
the \emph{holomorphic-topological boundary condition} requires:
\begin{enumerate}
\item Fields extend holomorphically to a compactification $\bar{\Sigma}$
\item At the boundary divisor $D = \bar{\Sigma} \setminus \Sigma$, fields have
a simple zero: $A \in \Omega^{0,*}(\bar{\Sigma}, \mathcal{O}_{\bar{\Sigma}}(-D))$
\item The volume form $\Omega$ has compatible pole: $\Omega \in K_{\bar{\Sigma}}(kD)$
for cubic interaction $(k=3)$
\end{enumerate}
\end{definition}

\begin{remark}[HT boundary bar-cobar comparison problem]
\label{rem:bar-cobar-ht-boundary}
The holomorphic-topological boundary condition realizes bar-cobar duality:
\begin{itemize}
\item \emph{Bar side.} Fields with prescribed asymptotics near $D$ (logarithmic
forms)
\item \emph{Cobar side.} Distributional fields on $\Sigma$ (delta functions at $D$)
\item \emph{Duality.} Perfect pairing via residue-distribution integral
\end{itemize}
\end{remark}

\begin{remark}[Geometric realization input]
The key is the volume form behavior. If $\Omega$ has a pole of order $k$ along
$D$:
\[\Omega \sim \frac{dz \wedge dw}{(z-w)^k}\]

then the action term:
\[\int_{\Sigma} \Omega \cdot A^k\]

is finite if and only if $A$ vanishes to order 1 along $D$.

This pole-zero compatibility is exactly the bar-cobar relationship: the logarithmic form $\eta = d\log(z-w)$ (bar) has a logarithmic singularity whose residue is the distribution $\delta(z-w)$ (cobar).

The holomorphic-topological boundary condition enforces this duality at the
geometric level.
\end{remark}

\begin{remark}[Extended-field-theory input]
The geometric realization of bar-cobar duality via
holomorphic-topological boundary conditions requires the extended
field-theory construction of the logarithmic and distributional
boundary conditions. The comparison map pairs logarithmic forms with
distributional fields by the residue integral across the boundary
divisor~$D$.
\end{remark}
