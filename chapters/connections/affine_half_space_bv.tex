\chapter{The affine half-space BV theorem and its transport to Virasoro and \texorpdfstring{$W_3$}{W3}}
\label{chap:affine-half-space-bv}

\begin{remark}[Position in the theorem graph]
Half-space BV quantization of the affine Khan--Zeng Poisson--vertex sigma model on $\C \times \R_{\ge 0}$ produces the affine Kac--Moody boundary VOA; the genus-$0$ projection of $\Theta^{\mathrm{oc}}$ is one-loop exact. The affine Poisson--vertex sigma model on $\C \times \R_{\ge 0}$ is mixed BF/Chern--Simons theory in the chiral boundary condition, and the half-space quantization input is the Garner--Raghavendran--Williams holomorphic--topological finiteness theorem in this one-real-topological-direction gauge. For a finite-type freely generated polynomial PVA, Construction~\ref{cons:kz-bv-fields-action} gives the shifted-cotangent BV action and Lemma~\ref{lem:kz-cme-pva-jacobi} identifies the classical master equation with the PVA Hamiltonian condition. The all-loop statement is Proposition~\ref{prop:kz-all-loop-counterterm-recursion} under Assumption~\ref{assum:kz-analytic-sdr-package}.
\end{remark}

For a semisimple Lie algebra $\mathfrak g$, the affine Poisson--vertex sigma model of
Khan--Zeng on the half-space $X = \mathbb C \times \mathbb R_{\ge 0}$ with Neumann boundary
condition admits a renormalized BV quantization to all orders in $\hbar$ in the product HT gauge; the affine
mixed BF/Chern--Simons presentation is one-loop exact. The resulting boundary vertex algebra is the affine
Kac--Moody vertex algebra $V^{K_{\mathrm{eff}}}(\mathfrak g)$ at the shifted formal invariant form
$K_{\mathrm{eff}}(k,\hbar)=k+\hbar\,\delta_{\mathfrak g}$, with one scalar component on each simple summand
of $\mathfrak g$. Principal Drinfeld--Sokolov reduction transports this single affine formal modulus to the
universal Virasoro and $W_3$ families away from the critical denominators.

\section{The question in first principles form}

\begin{problem}[The half-space BV problem]
Let $V$ be a freely generated Poisson vertex algebra, and let
$X = \mathbb C_z \times \mathbb R_{\ge 0,t}$.
The three-dimensional Poisson--vertex sigma model attaches to $V$ a mixed holomorphic-topological
BV theory on $X$ whose Neumann boundary condition keeps the $\Phi$-fields free and sets the
$\eta$-fields to vanish at the boundary. The sought deformation quantization of $V$ is the boundary
vertex algebra of the quantized half-space theory.

The theorem to prove is the existence of a renormalized solution of
the quantum master equation on the half-space, together with control
of the boundary anomaly classes at all loop orders, from the PVA
Hamiltonian bidifferential itself.
\end{problem}

\begin{remark}[Why affine is the first theorem-bearing case]
The affine Poisson vertex algebra is the first case where the nonlinear Poisson--vertex sigma model
reduces to a chiral deformation of mixed BF theory, hence to a class admitting rigorous BV
quantization. The half-space theorem sought by Khan--Zeng is already available in this affine case, once the fields and boundary condition are
identified correctly.
\end{remark}

\section{The affine Poisson--vertex sigma model from first principles}

\begin{setup}[Affine PVA and half-space fields]
Let $\mathfrak g$ be a finite-dimensional semisimple Lie algebra over $\mathbb C$, with structure constants
$f^{ab}{}_c$ and fixed nondegenerate invariant forms $\kappa_s$ on its simple summands. Write
$k\,\kappa^{ab}$ for the invariant form whose scalar on the summand $\mathfrak g_s$ is $k_s$.
The affine Poisson vertex algebra is
\[
V_{\mathrm{aff}}(\mathfrak g,k) = \mathrm{Sym}\bigl(\mathbb C[\partial] \otimes \mathfrak g\bigr)
\]
with generators $B^a$ and $\lambda$-bracket
\[
\{ B^a{}_{\lambda} B^b \} = f^{ab}{}_c B^c + k\,\kappa^{ab}\lambda.
\]
The corresponding three-dimensional Poisson--vertex sigma model on
$X = \mathbb C \times \mathbb R_{\ge 0}$ has fields
\[
B = B^a t_a \in \Omega^{0,0}(X) \otimes \mathfrak g,
\qquad
A = A_t\,dt + A_{\bar z}\,d\bar z \in \Omega^{0,1}_{\mathrm{mix}}(X) \otimes \mathfrak g,
\]
with classical action
\begin{equation}
\label{eq:affine-kz-action}
S_{\mathrm{aff}}(B,A)
=
\int_X \! dz\,
\left(
\kappa_{ab} B^a (d_t + \bar\partial) A^b
+ \frac{1}{2} f_{abc} B^a A^b A^c
+ \frac{k}{2}\kappa_{ab} A^a \partial A^b
\right).
\end{equation}
The Neumann boundary condition is
\begin{equation}
\label{eq:kz-neumann}
i^*A = 0,
\qquad
B\ \text{free on }\partial X,
\end{equation}
where $i\colon \partial X \hookrightarrow X$ is the boundary inclusion.
\end{setup}

\begin{remark}[Affine as Chern--Simons away from degenerate level]
When the invariant form $k\,\kappa$ is nondegenerate, the field redefinition
\[
\widetilde A = A + (k\,\kappa)^{-1}(B)\,dz
\]
transforms the bulk action to analytically continued Chern--Simons theory.
This is the classical reason the affine example is fully topological once the inner Sugawara/Virasoro
field is present.
\end{remark}

\section{Identification with mixed BF/Chern--Simons}

\begin{proposition}[Dictionary with mixed BF theory]\label{prop:affine-mixed-bf-dictionary}
Let
\[
\alpha := A \in \Omega^{0,\bullet}(\mathbb C)\widehat\otimes \Omega^{\bullet}(\mathbb R_{\ge 0})[1] \otimes \mathfrak g,
\qquad
\beta := B\,dz \in \Omega^{1,\bullet}(\mathbb C)\widehat\otimes \Omega^{\bullet}(\mathbb R_{\ge 0}) \otimes \mathfrak g.
\]
Then the affine action \eqref{eq:affine-kz-action} is exactly
\begin{equation}
\label{eq:smix-plus-ics}
S_{\mathrm{aff}} = S_{\mathrm{mix}} + I_{\mathrm{CS}}^{k\kappa},
\end{equation}
where
\begin{equation}
\label{eq:smix-def}
S_{\mathrm{mix}}(\alpha,\beta)
=
\int_X \langle \beta,(\bar\partial+d_t)\alpha\rangle
+ \frac{1}{2}\int_X \langle \beta,[\alpha,\alpha]\rangle,
\end{equation}
and
\begin{equation}
\label{eq:ics-def}
I_{\mathrm{CS}}^{q}(\alpha)
=
\frac{1}{2}\int_X q(\alpha,\partial\alpha)
\end{equation}
for an invariant symmetric form $q$ on $\mathfrak g$.
In particular, the affine Poisson--vertex sigma model is a chiral deformation of mixed BF theory.
\end{proposition}

\begin{proof}
The field $B$ is a scalar, so $\beta = B\,dz$ has bidegree $(1,\bullet)$ in the mixed holomorphic-topological
splitting. The kinetic term in \eqref{eq:affine-kz-action} becomes
\[
\int_X \kappa_{ab} B^a (d_t+\bar\partial)A^b\,dz
=
\int_X \langle \beta,(\bar\partial+d_t)\alpha\rangle.
\]
Likewise the cubic term rewrites as
\[
\frac{1}{2}\int_X f_{abc} B^a A^b A^c\,dz
=
\frac{1}{2}\int_X \langle \beta,[\alpha,\alpha]\rangle.
\]
Finally,
\[
\frac{k}{2}\int_X \kappa_{ab} A^a \partial A^b\,dz
=
\frac{k}{2}\int_X \langle \alpha,\partial\alpha\rangle
=
 I_{\mathrm{CS}}^{k\kappa}(\alpha).
\]
This is the mixed BF theory with its chiral Chern--Simons deformation.
\end{proof}

\begin{remark}[Combinatorial consequence]
The chiral nature of the deformation is decisive:
every propagator runs from an $\alpha$-leg to a $\beta$-leg, and every chiral interaction
has at most one $\beta$-leg, so the only connected Feynman graphs are trees,
single vertices, or one-loop graphs. This is the combinatorial source of one-loop exactness.
\end{remark}

\section{The boundary condition as a derived Lagrangian boundary condition}

\begin{definition}[Chiral boundary condition in mixed BF language]
Let $\sigma\colon \mathbb R \to \mathbb R$ be reflection, $\sigma(t)=-t$.
Write $A^{\mathrm{odd}} \subset \Omega^{0,\bullet}(\mathbb C)\widehat\otimes\Omega^{\bullet}(\mathbb R)$
for the $(-1)$-eigenspace under pullback by $\sigma$.
Concretely, its $0$-form part consists of functions $f(t)$ with $f(0)=0$, and its $1$-form part consists
of one-forms $g(t)\,dt$ whose pullback to the boundary vanishes because $i^*(dt)=0$.
The chiral boundary condition is the restriction
\[
\alpha \in A^{\mathrm{odd}}\otimes \mathfrak g[1],
\qquad
\beta\ \text{free}.
\]
\end{definition}

\begin{proposition}[Equivalence of boundary conditions]\label{prop:boundary-condition-equivalence}
The Neumann boundary condition \eqref{eq:kz-neumann} for the affine Poisson--vertex sigma model is
implemented, in the mixed BF/Chern--Simons presentation, by the chiral boundary condition above.
More precisely, the odd-reflection model is a strict model for the derived fiber product imposing
$i^*\alpha=0$.
\end{proposition}

\begin{proof}
If $\alpha \in A^{\mathrm{odd}}\otimes \mathfrak g[1]$, then its $0$-form part is an odd function of $t$ and hence
vanishes at $t=0$; its $dt$-part survives as an odd coefficient times $dt$, but its pullback to the boundary
also vanishes because $i^*(dt)=0$. Therefore $i^*\alpha=0$, which is exactly the Khan--Zeng condition
$i^*A=0$.

Conversely, the BV formalism does not impose boundary conditions by naive degreewise kernels alone; one must
produce a Lagrangian subcomplex modelling the derived fiber product of fields with the zero section of the boundary
restriction map. The odd-reflection model does exactly this. Because the restriction map to the boundary is
levelwise surjective, the strict fiber product coincides with the derived one. Hence the chiral boundary condition
is a BV-correct model of the Neumann boundary condition.
\end{proof}

\begin{remark}[First-principles content]
The half-space problem for the affine Poisson--vertex sigma model is not a new analytic problem:
it is the already-solved half-space problem for mixed BF/Chern--Simons, expressed in different fields.
\end{remark}

\section{The all-loops half-space BV theorem in the affine case}

\begin{maintheorem}[Affine half-space BV quantization]\label{thm:affine-half-space-bv}
Let $\mathfrak g$ be semisimple and let $k$ be the tuple of affine scalar levels, equivalently the invariant
form $k\,\kappa\in\mathrm{Sym}^2(\mathfrak g^*)^{\mathfrak g}$ used in the affine $\lambda$-bracket.
Consider the affine Poisson--vertex sigma model on
$X=\mathbb C\times\mathbb R_{\ge 0}$ with Neumann boundary condition, in the mixed BF/Chern--Simons
presentation of Proposition~\ref{prop:affine-mixed-bf-dictionary} and the chiral boundary condition of
Proposition~\ref{prop:boundary-condition-equivalence}.
Then the following hold.
\begin{enumerate}
 \item There exists a renormalized perturbative BV quantization solving the quantum master equation to all orders in $\hbar$.
 \item The quantization is one-loop exact: the renormalized effective action receives no contributions from loop order $\ge 2$.
 \item The only quantum correction is a local one-loop functional proportional to
 \begin{equation}
 \label{eq:one-loop-boundary-shift-functional}
 \hbar\int_X \operatorname{Tr}^{\mathrm{ad}}_{\mathfrak g}(\alpha\,\partial\alpha).
 \end{equation}
 \item Consequently the quantum theory is equivalent to the affine theory at the shifted invariant form
 \begin{equation}
 \label{eq:keff-def}
 K_{\mathrm{eff}}(k,\hbar) = k + \hbar\,\delta_{\mathfrak g},
 \end{equation}
 where $\delta_{\mathfrak g}\in\mathrm{Sym}^2(\mathfrak g^*)^{\mathfrak g}$ has one scalar component on each
 simple summand of $\mathfrak g$.
\end{enumerate}
\end{maintheorem}

\begin{proof}
By Proposition \ref{prop:affine-mixed-bf-dictionary}, the affine theory is a chiral deformation of mixed BF
by the Chern--Simons functional $I_{\mathrm{CS}}^{k\kappa}$. By Proposition \ref{prop:boundary-condition-equivalence},
its Neumann boundary condition is precisely the chiral boundary condition in the mixed BF presentation.

The Garner--Raghavendran--Williams holomorphic--topological finiteness theorem in one real topological
direction applies to this mixed BF/Chern--Simons theory with chiral boundary condition~\cite{GRW21}. In this
cotangent BF presentation every connected graph contributing to the effective interaction is a tree or a
one-loop graph, because propagators pair $\alpha$ with $\beta$ and every interaction has at most one
$\beta$-leg. Thus the affine deformation has a renormalized solution of the QME and no loop contribution beyond
one loop, proving items (1) and (2). The same local computation identifies the one-loop correction with a
functional proportional to \eqref{eq:one-loop-boundary-shift-functional}, giving item (3).

For semisimple $\mathfrak g$, translation-invariant cotangent quantizations of mixed BF theory form a torsor for
$\hbar\,\mathrm{Sym}^2(\mathfrak g)^{\mathfrak g}$. Since $\mathfrak g$ is semisimple, this space is one-dimensional on each
simple summand and generated by the invariant bilinear form. Thus the one-loop correction has no freedom other
than a level shift, yielding \eqref{eq:keff-def}. Since there are no loops beyond one, the formal level has no
higher $\hbar$-terms. This proves item (4).
\end{proof}

\begin{corollary}[The affine case of the Khan--Zeng half-space problem]\label{cor:affine-kz-solved}
For the affine Poisson vertex algebra, the half-space deformation quantization problem posed by Khan--Zeng is
unobstructed. The obstruction theory terminates at one loop, and that one-loop term is absorbed by the invariant
form shift $k\mapsto K_{\mathrm{eff}}(k,\hbar)$.
\end{corollary}

\begin{proof}
Theorem \ref{thm:affine-half-space-bv} gives a renormalized BV quantization to all orders. One-loop exactness
kills every possible higher-loop obstruction. The remaining one-loop functional is exactly the level-shift term,
which lies in the one-dimensional quantization torsor and hence is absorbed into the renormalized level.
\end{proof}

\begin{remark}[Physical normalization]
The coefficient $\delta_{\mathfrak g}$ is left abstract. Standard vertex-algebra and
Chern--Simons/WZW normalizations give
$\delta_{\mathfrak g}=-h^{\vee}$ on each simple summand, but the analytic input above proves only
proportionality to the invariant form, not the exact constant.
The half-space quantization theorem does not require pinning down that scalar.
\end{remark}

\section{The boundary obstruction line}

\begin{definition}[Boundary jet obstruction complex]\label{def:boundary-jet-obstruction}
Let
\[
\mathcal A_{\partial,\mathrm{aff}}
=
\mathbb C\bigl[J^a_{(m)},c^a_{(m)} \,;\, a=1,\dots,\dim\mathfrak g,\ m\ge 0\bigr]
\]
be the boundary jet algebra generated by the current jets
$J^a_{(m)}=\partial^m J^a$ and ghost jets $c^a_{(m)}=\partial^m c^a$.
Set
\[
\operatorname{Obs}^{\bullet}_{\partial,\mathrm{aff}}
=
\mathcal A_{\partial,\mathrm{aff}}/\partial \mathcal A_{\partial,\mathrm{aff}},
\]
graded by ghost number, and define the BRST differential by
\[
\delta J^a_{(m)}
=
K\,c^a_{(m+1)}
+
f^a{}_{bc}\sum_{r=0}^{m}\binom{m}{r}J^b_{(r)}c^c_{(m-r)},
\]
\[
\delta c^a_{(m)}
=
-\frac12 f^a{}_{bc}\sum_{r=0}^{m}\binom{m}{r}c^b_{(r)}c^c_{(m-r)}.
\]
Degree $0$ elements are local counterterms, degree $1$ elements are local boundary anomalies, and degree $2$
elements are obstructions to removing those anomalies.
\end{definition}

\begin{remark}[Current-preserving truncation]
The two-term complex $C^{\bullet}_{\mathrm{KM}}$ introduced below is the weight-one, singular-OPE truncation of
Definition \ref{def:boundary-jet-obstruction}. It is the correct deformation complex for the affine
current sector, i.e.\ for deformations that preserve a weight-one current generator and keep the singular OPE
linear in that current.
\end{remark}

\begin{construction}[The current-preserving boundary deformation complex]\label{cons:boundary-km-complex}
Let $J_x(z)$ denote the boundary current associated to $x\in \mathfrak g$.
A first-order current-preserving singular deformation of the affine OPE is a pair
\[
(\omega,\mu)\in \operatorname{Sym}^2(\mathfrak g^*)\oplus \operatorname{Hom}(\Lambda^2\mathfrak g,\mathfrak g)
\]
with deformed singular OPE
\[
J_x(z)J_y(w)
\sim
\frac{K\kappaChHodge(x,y)+\varepsilon\,\omega(x,y)}{(z-w)^2}
+
\frac{J_{[x,y]+\varepsilon\mu(x,y)}(w)}{z-w}.
\]
A first-order field redefinition is $R\in\operatorname{End}(\mathfrak g)$ acting by
$J_x\mapsto J_x+\varepsilon J_{Rx}$.
This yields a two-term complex
\[
C^0_{\mathrm{KM}} = \operatorname{End}(\mathfrak g),
\qquad
C^1_{\mathrm{KM}} = \operatorname{Sym}^2(\mathfrak g^*)\oplus \operatorname{Hom}(\Lambda^2\mathfrak g,\mathfrak g),
\]
with differential
\[
d_0R = (\omega_R,\mu_R),
\]
where
\[
\omega_R(x,y) = K\bigl(\kappaChHodge(Rx,y)+\kappaChHodge(x,Ry)\bigr),
\qquad
\mu_R(x,y)=R[x,y]-[Rx,y]-[x,Ry].
\]
The linearized Jacobi and invariance equations are
\[
\delta_{\mathrm{CE}}\mu = 0,
\qquad
\omega([x,y],z)-\omega(x,[y,z]) + K\kappaChHodge(\mu(x,y),z)-K\kappaChHodge(x,\mu(y,z))=0.
\]
\end{construction}

\begin{theorem}[The affine boundary deformation line]\label{thm:affine-boundary-line}
For simple $\mathfrak g$,
\[
H^1(C^{\bullet}_{\mathrm{KM}}) \cong \mathbb C\cdot \Theta,
\qquad
\Theta = [\,(\kappa,0)\,].
\]
Equivalently, every infinitesimal current-preserving singular deformation of the affine boundary OPE is
gauge-equivalent to a unique infinitesimal level shift.
\end{theorem}

\begin{proof}
Let $(\omega,\mu)$ be a cocycle. The linearized Jacobi identity says that $\mu$ is a Chevalley--Eilenberg
$2$-cocycle with values in the adjoint module. Since $\mathfrak g$ is simple, Whitehead's lemma gives
$H^2(\mathfrak g;\mathfrak g)=0$, so $\mu=\delta_{\mathrm{CE}}R$ for some $R\in\operatorname{End}(\mathfrak g)$.
Subtracting the coboundary $d_0R$ kills $\mu$, so the cohomology class is represented by $(\omega',0)$.
The second cocycle equation then reduces to invariance of $\omega'$, namely
\[
\omega'([x,y],z)=\omega'(x,[y,z]).
\]
Thus $\omega'$ is an invariant symmetric bilinear form on the simple Lie algebra $\mathfrak g$, hence a scalar
multiple of the Killing form. Therefore the cohomology is one-dimensional and generated by the class
$[(\kappa,0)]$.
\end{proof}

\begin{lemma}[Pole--weight balance for boundary currents]\label{lem:pole-weight-balance}
In the affine half-space theory, every singular contribution to the boundary current OPE
\[
J^a(z)J^b(w)\sim \sum_{m\ge 1} \frac{{\mathcal O}^{ab}_m(w)}{(z-w)^m}
\]
satisfies
\begin{equation}
\label{eq:pole-weight-balance}
m + \mathrm{wt}({\mathcal O}^{ab}_m)=2,
\end{equation}
where $\mathrm{wt}(J)=1$ and $\mathrm{wt}(\partial)=1$.
\end{lemma}

\begin{proof}
Each external current has weight $1$, so the input weight is $2$.
In the mixed BF/Chern--Simons presentation one assigns
$\mathrm{wt}(\beta)=1$, $\mathrm{wt}(\alpha)=0$, and $\mathrm{wt}(\partial)=1$.
Every bulk interaction then has total holomorphic weight $1$, while the reflected propagator is homogeneous
of degree $-1$ in the boundary separation. Renormalized local counterterms preserve this scaling because there is
no additional boundary tensor available. Hence any singular term in the OPE of two weight-one currents has
homogeneous total weight $2$. A pole of order $m$ contributes weight $m$, so the remaining operator must have
weight $2-m$.
\end{proof}

\begin{theorem}[Current-sector rigidity of the quantum boundary algebra]\label{thm:current-sector-rigidity}
Let $\mathfrak g$ be simple. The singular OPE of the quantum boundary currents has the exact affine form
\begin{equation}
\label{eq:quantum-affine-boundary-ope}
J^a(z)J^b(w)
\sim
\frac{K_{\mathrm{eff}}(k,\hbar)\,\kappa^{ab}}{(z-w)^2}
+
\frac{f^{ab}{}_c J^c(w)}{z-w}.
\end{equation}
In particular, no poles of order $\ge 3$, no $\partial J$ terms, and no composite singular fields such as
$\! :JJ: \!$ can appear in the singular part of the boundary current OPE.
\end{theorem}

\begin{proof}
By Lemma \ref{lem:pole-weight-balance}, only $m=2$ and $m=1$ can occur.
If $m=2$, then $\mathrm{wt}({\mathcal O}_2)=0$, so ${\mathcal O}_2$ is a scalar. By $\mathfrak g$-invariance,
it is a multiple of $\kappa^{ab}$.
If $m=1$, then $\mathrm{wt}({\mathcal O}_1)=1$, so ${\mathcal O}_1$ is a linear combination of currents.
Skew-symmetry in the simple-pole channel shows that the coefficient lies in
$\operatorname{Hom}_{\mathfrak g}(\Lambda^2\mathfrak g,\mathfrak g)$, which for simple $\mathfrak g$ is generated by the Lie bracket.
Thus the singular OPE is of the form
\[
J^a(z)J^b(w)
\sim
\frac{\Omega^{ab}}{(z-w)^2} + \frac{C\,f^{ab}{}_cJ^c(w)}{z-w}.
\]
By Theorem \ref{thm:affine-boundary-line}, after a current redefinition preserving the vacuum and translation,
$C$ is normalized to $1$ and $\Omega^{ab}=K_{\mathrm{eff}}\kappa^{ab}$.
Theorem \ref{thm:affine-half-space-bv} then identifies $K_{\mathrm{eff}}$ with the one-loop shifted level.
\end{proof}

\begin{corollary}[The boundary VOA]\label{cor:boundary-affine-voa}
The boundary vertex algebra of the affine half-space theory is the affine Kac--Moody vertex algebra
\[
\mathcal V_{\partial}(\mathfrak g,k) \cong V^{K_{\mathrm{eff}}(k,\hbar)}(\mathfrak g).
\]
\end{corollary}

\begin{proof}
Theorem \ref{thm:current-sector-rigidity} shows that the generating currents satisfy the universal affine
singular OPE with renormalized level $K_{\mathrm{eff}}$. The vertex algebra they generate is therefore the universal
affine Kac--Moody vertex algebra at that level.
\end{proof}

\begin{corollary}[Explicit loop cocycles]\label{cor:loop-cocycles-affine}
Let
\[
\omega_{\mathrm{KM}} := \oint \kappaChHodge(c,\partial J)\,dz
\]
be the Kac--Moody central cocycle in the boundary BRST complex. Then
\[
\mathfrak a^{(1)}_{\partial,\mathrm{aff}} = \delta_{\mathfrak g}\,\omega_{\mathrm{KM}},
\qquad
\mathfrak a^{(\ell)}_{\partial,\mathrm{aff}}=0\quad (\ell\ge 2).
\]
In particular, the two-loop anomaly cocycle vanishes identically.
\end{corollary}

\begin{proof}
Theorem \ref{thm:affine-boundary-line} shows that the current-preserving anomaly space is the line spanned by
$\omega_{\mathrm{KM}}$. Theorem \ref{thm:affine-half-space-bv} further shows that the quantum theory is one-loop exact.
Hence only the one-loop coefficient survives, and every higher-loop cocycle vanishes.
\end{proof}

\begin{corollary}[Homotopy-abelian formal line]\label{cor:homotopy-abelian-line}
After passing from the bare coordinate $k$ to the renormalized coordinate $K_{\mathrm{eff}}$, the
current-sector boundary deformation problem is formally smooth and homotopy-abelian:
its full deformation theory is the formal line parameterized by $K_{\mathrm{eff}}$, and all higher brackets vanish
after this reparametrization.
\end{corollary}

\begin{proof}
Theorem \ref{thm:affine-boundary-line} shows that the tangent space of the current-sector deformation theory is
the one-dimensional level-shift line. Theorem \ref{thm:affine-half-space-bv} shows that the complete quantum
theory is already represented by the single formal coordinate $K_{\mathrm{eff}}=k+\hbar\delta_{\mathfrak g}$, with no
higher-loop corrections. Thus, in the renormalized coordinate, every quantum correction has been absorbed and
the controlling $L_{\infty}[1]$-algebra is equivalent to an abelian one-dimensional complex.
\end{proof}

\section{The transverse boundary and the affine Koszul shadow}

\begin{corollary}[Affine Neumann/Dirichlet boundary pair]\label{cor:affine-koszul-shadow}
For the affine half-space theory, the Neumann boundary algebra is the affine vertex algebra
$V^{K_{\mathrm{eff}}}(\mathfrak g)$, while the transverse Dirichlet boundary algebra is the differential graded
commutative algebra
\[
C^{\bullet}(\mathfrak g[[z]])
\]
with Chevalley--Eilenberg differential.
Thus the affine half-space theory realizes the linear chiral Koszul pair predicted by the Poisson--vertex sigma
model.
\end{corollary}

\begin{proof}
The Neumann side is Corollary \ref{cor:boundary-affine-voa}. The Dirichlet side is classical and linear: for a
linear Poisson tensor, the surviving boundary ghosts form the Chevalley--Eilenberg algebra of $\mathfrak g[[z]]$.
The Neumann side is quantized by Theorem~\ref{thm:affine-half-space-bv}, so the linear Koszul pair
exists as a boundary pair for the affine half-space model.
\end{proof}

\section{Transport to Virasoro and \texorpdfstring{$W_3$}{W3} by Drinfeld--Sokolov reduction}

\begin{proposition}[Filtered Drinfeld--Sokolov compatibility]\label{prop:filtered-ds}
Let $F^{\bullet}$ be the Li filtration on the affine boundary VOA
$\mathcal V_{\partial}(\mathfrak g,k)\cong V^{K_{\mathrm{eff}}}(\mathfrak g)$.
The principal Drinfeld--Sokolov BRST differential preserves this filtration, and
\[
\operatorname{gr}\,H^0_{\mathrm{DS}}\bigl(\mathcal V_{\partial}(\mathfrak g,k)\bigr)
\cong
H^0_{\mathrm{DS,cl}}\bigl(\operatorname{gr}\mathcal V_{\partial}(\mathfrak g,k)\bigr).
\]
In particular, the associated graded of the reduced boundary algebra is the classical Drinfeld--Sokolov
reduction of the affine Poisson vertex algebra.
\end{proposition}

\begin{proof}
The Li filtration on the affine VOA is compatible with current weight and translation. The BRST current for
principal Drinfeld--Sokolov reduction is polynomial in affine currents and ghosts, hence preserves the induced
filtration on the BRST complex. Passing to associated graded replaces the quantum current OPE by its classical
Poisson--vertex limit, so the associated graded BRST differential is exactly the classical Drinfeld--Sokolov
differential. Taking zeroth cohomology gives the claim.
\end{proof}

\begin{theorem}[Boundary Drinfeld--Sokolov transport]\label{thm:boundary-ds-transport}
Let $\mathcal V_{\partial}(\mathfrak g,k)$ be the boundary VOA from
Corollary \ref{cor:boundary-affine-voa}.
Assume the effective level is non-critical on the relevant summand:
$K_{\mathrm{eff}}+2\neq 0$ for $\mathfrak{sl}_2$ and
$K_{\mathrm{eff}}+3\neq 0$ for $\mathfrak{sl}_3$.
Then principal Drinfeld--Sokolov reduction transports the single affine modulus
$K_{\mathrm{eff}}(k,\hbar)$ to the universal reduced families.

\begin{enumerate}
 \item For $\mathfrak g=\mathfrak{sl}_2$,
 \[
 H^0_{\mathrm{DS}}\bigl(\mathcal V_{\partial}(\mathfrak{sl}_2,k)\bigr)
 \cong
 \mathrm{Vir}_{\,c_{\mathrm{Vir}}(K_{\mathrm{eff}})},
 \qquad
 c_{\mathrm{Vir}}(K)=1-\frac{6(K+1)^2}{K+2}.
 \]
 \item For $\mathfrak g=\mathfrak{sl}_3$,
 \[
 H^0_{\mathrm{DS}}\bigl(\mathcal V_{\partial}(\mathfrak{sl}_3,k)\bigr)
 \cong
 \mathcal W_3\bigl(c_{W_3}(K_{\mathrm{eff}})\bigr),
 \qquad
 c_{W_3}(K)=2-\frac{24(K+2)^2}{K+3}.
 \]
\end{enumerate}
There are no additional reduced-sector quantum parameters on this universal DS surface.
\end{theorem}

\begin{proof}
By Corollary \ref{cor:boundary-affine-voa}, the boundary VOA is exactly the affine algebra

\begin{proposition}[Hamiltonian translation form of the PV sigma model]%
\label{prop:poisson-hamiltonian}
Let $V$ be a finite-type freely generated Poisson vertex algebra on generators
$a^1,\ldots,a^n$ with $\lambda$-brackets $\{a^i_\lambda a^j\}=
\sum_k \pi^{ij}_k(\boldsymbol a)\lambda^k$, and suppose that the holomorphic
translation operator $T\colon V\to V$ is inner in the sense that there exists
$H\in V$ with
\[
T(f) \;=\; \{H_\lambda\, f\}\big|_{\lambda=0}
\qquad\text{for all } f\in V.
\]
Then:
\begin{enumerate}
\item On the jet formal scheme $J_\infty M$, $M=\operatorname{Spec}\mathbb C[\boldsymbol a]$, the
$\lambda$-bracket is the Hamiltonian bidifferential
$\Pi_V(\partial;\boldsymbol a)$ of Construction~\ref{cons:kz-bv-fields-action}; its zero-mode part is the
ordinary Poisson bivector $\pi_0$ on $M$.
\item The element $H$ defines the Hamiltonian evolutionary vector field
$X_H(f)=\{H_\lambda f\}|_{\lambda=0}$, and the hypothesis says $X_H=T$.
On fields this is the infinitesimal holomorphic translation $\partial_z$, i.e. the formal additive action
$z\mapsto z+\epsilon$, not a dilation action.
\item The three-dimensional Poisson--vertex sigma model on
$\mathbb C_z\times\mathbb R_t$ with fields
$\boldsymbol a(z,t)$ is the Hamiltonian-translation extension of the AKSZ Poisson sigma model on the jet
target. In local Darboux coordinates its action decomposes as
\[
S_{\mathrm{BV}}
\;=\;
\underbrace{S_{\mathrm{Poisson}}}_{\text{Kontsevich}}
\;+\;
\underbrace{\int_{\mathbb C\times\mathbb R}
H(\boldsymbol a)\,\mathrm{d}\bar z\wedge\mathrm{d}t}_{\text{Hamiltonian translation term}},
\]
where $S_{\mathrm{Poisson}}$ is the AKSZ action of the Poisson sigma model on
$\mathbb C$ and the second term couples the Hamiltonian $H$ to the
holomorphic-topological volume form.
\end{enumerate}
\end{proposition}

\begin{proof}
For (1): the $\lambda$-bracket of a freely generated PVA is polynomial in
$\lambda$, so $\pi^{ij}(\lambda)=\sum_k\pi^{ij}_k(\boldsymbol a)\lambda^k$.
Sesquilinearity converts powers of $\lambda$ into powers of the jet derivative
$\partial_z$ and gives the Hamiltonian matrix $\Pi_V(\partial;\boldsymbol a)$.
At $\lambda=0$ the skew-symmetry and Jacobi identity specialize to the ordinary
Poisson identities for $\pi_0^{ij}(\boldsymbol a)=\pi^{ij}(\lambda)|_{\lambda=0}$.

For (2): the innerness condition $T = \{H_\lambda\,\cdot\}|_{\lambda=0}$ means
precisely that $T$ is the Hamiltonian evolutionary vector field of $H$ with respect to
$\pi_0$. On fields $\boldsymbol a(z,t)$, the operator $T=\partial_z$ generates
the formal additive translation flow $e^{\epsilon\partial_z}$.

For (3): the AKSZ construction for the Poisson sigma model on a surface
$\Sigma$ with target $(M,\pi)$ gives a BV action $S_{\mathrm{Poisson}}$.
Promoting $\Sigma = \mathbb C_z$ to the three-manifold
$\mathbb C_z\times\mathbb R_t$ while requiring fields to factorize
holomorphically in $z$ and associatively in $t$ adds exactly the coupling
$\int H\,\mathrm{d}\bar z\wedge\mathrm{d}t$ that enforces
$\bar\partial_z a^i = \{H,a^i\}_{\pi_0}\mathrm{d}t$ in the local-functional
quotient. This gives the Hamiltonian translation term in the KZ BV action.
\end{proof}

\begin{construction}[The equivariant Kontsevich integral on the half-space]%
\label{cons:equivariant-kontsevich-integral}
The half-space $X = \mathbb C\times\mathbb R_{\ge 0}$ admits a logarithmic
Fulton--MacPherson compactification
$\overline{\operatorname{Conf}}_n^{\partial}(X)$ whose codimension-one boundary
strata are classified by Construction~\ref{cons:half-space-colored-operad} into
four types. The equivariant Kontsevich integral assigns to each admissible
graph $\Gamma$ on $n$ vertices a weight
\[
w_\Gamma^{\mathrm{eq}}
\;=\;
\int_{\overline{\operatorname{Conf}}^{\partial}_{V(\Gamma)}(X)}
\bigwedge_{e\in E(\Gamma)} \omega_e^{\mathrm{refl}},
\]
where $\omega_e^{\mathrm{refl}}$ is the \emph{reflected} propagator form
(defined below). The boundary contributions decompose according to the four
strata:

\begin{enumerate}
\item \textbf{Bulk clusters.} A subset of interior vertices collapses away from
$\partial X$. The residue along this stratum is the standard Kontsevich weight
for the collapsed subgraph, giving the Poisson deformation quantization in the
interior. These strata reproduce the Kontsevich star product on
$(M,\pi)$.

\item \textbf{Boundary clusters.} A subset of boundary vertices collapses along
$\partial X = \mathbb C\times\{0\}$. The residue is a Kontsevich-type integral
on the half-plane $\mathbb H$, giving the associative structure on the boundary
algebra. In the affine case this recovers the vertex-algebra product on
$V_k(\mathfrak g)$.

\item \textbf{Mixed clusters.} Interior vertices collide with boundary vertices.
The residue computes the mixed Swiss-cheese composition:
bulk observables act on boundary observables. This is the open--closed map
of the Swiss-cheese operad $\mathrm{SC}^{\mathrm{ch,top}}$.

\item \textbf{Plus-diagonal and self-image faces.} An interior vertex
may approach an image charge through $z_i=z_j,\ t_i+t_j=0$, or a
single vertex may approach its own image at $t_i=0$. These faces have
no analogue as ordinary free fixed-locus divisors in the whole-space
Kontsevich integral. Their residues are the \emph{image-charge
corrections} that enforce the Neumann boundary condition, and they
produce the equivariant correction terms in the quantized action.
\end{enumerate}

On the dense open locus $t,t'>0$, the reflected propagator is the
signed method-of-images kernel:
\[
K^{\mathrm{refl}}_e(z,t;\,z',t')
\;=\;
K(z-z',t-t')+\varepsilon_e K(z-z',t+t'),
\]
where $K(z,t;\,z',t')$ is the bulk propagator of the
holomorphic-topological theory (meromorphic in $z-z'$, exponentially
decaying in $|t-t'|$). Extending this kernel to boundary faces is the
image-choice compactification problem; the image term is the source of
the plus-diagonal and self-image residues.
\end{construction}

\begin{theorem}[Image-charge propagator and reflected Arnold cancellation]

\label{eq:half-space-image-charge-propagator}

\label{eq:half-space-reflected-weight-composition}

\end{equation}

\label{eq:half-space-quadratic-reflected-obstruction}

nonlinear $\pi$, those faces contribute genuinely new terms at each

these contributions are compatible with the operadic structure.

\section{The jet-order inductive scheme}
\label{sec:jet-order-inductive}

The affine half-space theorem succeeds because the Lie--Poisson structure
is linear in the fields, which forces image-charge graphs at loop order
$\ge 2$ to vanish by degree counting. The general construction extends beyond the affine case starting at
the first jet-order correction to the Poisson kernel. An inductive
scheme reduces the general reflected weight
identity~\eqref{eq:reflected-weight-identity} to a tower of affine-type
problems, one at each jet order, with modified structure constants
incorporating lower-order corrections.

\subsection{The jet-order filtration on a Poisson vertex algebra}

\begin{definition}[Jet-order filtration]
\label{def:jet-order-filtration}
Let $V$ be a freely generated Poisson vertex algebra on generators
$a^1,\ldots,a^n$ with $\lambda$-bracket
$\{a^i_\lambda a^j\}=\sum_{k\ge 0}\pi^{ij}_k(\boldsymbol{a})\lambda^k$.
Write the structure functions as Taylor series in the fields:
\[
\pi^{ij}_k(\boldsymbol{a})
=
\underbrace{\pi^{ij}_{k,0}}_{\text{constant}}
+
\underbrace{\pi^{ij}_{k,1}(\boldsymbol{a})}_{\text{linear}}
+
\underbrace{\pi^{ij}_{k,2}(\boldsymbol{a})}_{\text{quadratic}}
+
\cdots
\]
where $\pi^{ij}_{k,m}$ is homogeneous of degree $m$ in $\boldsymbol{a}$.
Define the \emph{jet-order filtration} by
\[
J^{\le N}V
\;:=\;
\text{PVA with brackets truncated to }
\{a^i_\lambda a^j\}^{\le N}
=
\sum_{k\ge 0}\sum_{m=0}^{N}\pi^{ij}_{k,m}(\boldsymbol{a})\,\lambda^k.
\]
In particular:
\begin{itemize}
\item $J^{\le 0}V$: constant Poisson kernel (trivial brackets).
\item $J^{\le 1}V$: affine (Lie--Poisson) PVA. The linear structure
 functions $\pi^{ij}_{k,1}(\boldsymbol{a})=f^{ij}{}_{k,\ell}\,a^\ell$
 are the structure constants of a Lie algebra (with $\lambda$-dependence
 encoding differential-operator structure).
\item $J^{\le 2}V$: quadratic Poisson correction.
\end{itemize}
The induction below is used on the affine-based locus: the constant and
linear truncation $J^{\le 1}V$ is itself a PVA, and the higher
homogeneous terms are treated as extensions of that affine Lie conformal
algebra.
\end{definition}

\begin{lemma}[Jet-order Jacobi compatibility]
\label{lem:jet-jacobi}
Assume $V$ lies on the affine-based locus of Definition~\ref{def:jet-order-filtration}. Then
$J^{\le N}V$ satisfies the PVA Jacobi identity modulo $J^{>N}$
contributions. For each $N\ge 2$, the obstruction to solving the
Jacobi identity at the next homogeneous jet order is a class in the PVA
cohomology of $J^{\le 1}V$ with coefficients in the degree-$N$
jet-module.
\end{lemma}

\begin{proof}
The PVA Jacobi identity
$\{a^i_\lambda\{a^j_\mu a^k\}\}-\{a^j_\mu\{a^i_\lambda a^k\}\}
=\{\{a^i_\lambda a^j\}_{\lambda+\mu}a^k\}$
is polynomial in the structure functions. Restricting to jet order
$\le N$ on both sides, the Jacobi identity holds modulo terms of jet
order $>N$. At $N=1$, exactness is precisely the affine-based
hypothesis. For $N\ge 2$, the failure at jet order $N+1$ is a $3$-cocycle in the
Chevalley--Eilenberg complex of the affine PVA $J^{\le 1}V$ (viewed as
a Lie conformal algebra) with coefficients in the symmetric power
$\operatorname{Sym}^{N}(\boldsymbol{a})$.
\end{proof}

\subsection{Reflected weight identity by induction on jet order}

\begin{theorem}[Jet-order decomposition of the reflected weight identity]
\label{thm:jet-inductive-rwi}
Assume the reflected graph weights of Construction~\ref{cons:kz-renormalized-weights} are defined on each
finite jet--weight quotient. For an affine-based PVA $V$, the reflected
weight identity~\eqref{eq:reflected-weight-identity} admits the jet-order decomposition
\begin{equation}\label{eq:jet-rwi-decomposition}
\operatorname{RWI}(V)
=
\operatorname{RWI}(J^{\le 1}V)
+
\sum_{N=2}^{\infty}\delta_N(V),
\end{equation}
where $\operatorname{RWI}(J^{\le 1}V)$ is the reflected weight identity
for the affine truncation (which holds by
Theorem~\ref{thm:affine-half-space-bv}), and $\delta_N(V)$ is the
jet-order-$N$ correction.

Explicitly, $\delta_N(V)$ is a sum over admissible reflected graphs
$\Gamma$ whose vertices carry at least one interaction of jet order
$\ge N$ and whose total jet order is exactly~$N$.
\end{theorem}

\begin{proof}
The equivariant Kontsevich integral of
Construction~\ref{cons:equivariant-kontsevich-integral} is polynomial in
the Poisson structure functions. Expanding
$\pi^{ij}=\sum_{m\ge 0}\pi^{ij}_m$ and collecting terms by total
jet order gives the decomposition. At jet order $0$: the constant
Poisson kernel contributes no interaction vertices, so
$\operatorname{RWI}(J^{\le 0}V)=0$ trivially. At jet order $1$: the
linear Poisson kernel is the affine case, so
$\operatorname{RWI}(J^{\le 1}V)=0$ by
Theorem~\ref{thm:affine-half-space-bv}. The corrections $\delta_N$
for $N\ge 2$ are therefore the full content of the nonlinear reflected
weight identity.
\end{proof}

\begin{construction}[Jet-order-$N$ reflected graphs]
\label{cons:jet-N-graphs}
At jet order $N$, the contributing reflected graphs $\Gamma$ are:
\begin{enumerate}
\item Graphs with a single vertex of jet order $N$ and all other
 vertices of jet order $1$ (affine). These are the
 \emph{single-insertion graphs}.
\item Graphs with multiple vertices whose jet orders sum to $N$.
 These involve products of lower-jet-order insertions.
\end{enumerate}
The single-insertion graphs are the leading term. Their structure
is controlled by the Kontsevich integral with one ``nonlinear vertex''
$\pi^{ij}_{k,N}$ inserted into an otherwise affine background.
\end{construction}

\subsection{The quadratic correction: first non-affine theorem}

\begin{proposition}[Structure of the quadratic correction]
\label{prop:quadratic-structure}
At jet order~$2$, the correction $\delta_2(V)$ is a sum over a finite
set of reflected graph types. Each such graph $\Gamma$ has:
\begin{enumerate}
\item One quadratic vertex $v_*$ carrying the interaction
 $\pi^{ij}_{k,2}=q^{ij}_{k,\ell m}a^\ell a^m$ (valence~$4$: two
 outgoing derivative edges from the Poisson bivector, two incoming
 field-contraction edges from $a^\ell a^m$).
\item $p\ge 0$ affine vertices (valence~$3$: two outgoing, one
 incoming).
\item Boundary vertices receiving the outgoing edges.
\item At most one reflection-stratum contribution (from the
 image-charge term in $K^{\mathrm{refl}}$).
\end{enumerate}
The quadratic vertex has one extra incoming edge compared to an affine
vertex. In the equivariant Kontsevich integral of
Construction~\ref{cons:equivariant-kontsevich-integral}, this extra
edge shifts the form-degree/dimension balance of the configuration-space
weight integral by $+1$ relative to the affine case.
\end{proposition}

\begin{proof}
The equivariant Kontsevich integral is polynomial in the Poisson
structure functions. Extracting the homogeneous-degree-$2$ part in the
field variables selects exactly the graphs with one quadratic vertex
(contributing degree~$2$) plus any number of affine vertices
(contributing degree~$1$ each). The vertex valences follow from the
tensor structure of the Poisson kernel: $\pi^{ij}$ has two upper
(derivative) indices giving outgoing edges, and the polynomial
$q^{ij}_{k,\ell m}a^\ell a^m$ has two field indices giving incoming
contractions.
\end{proof}

\begin{corollary}[Quadratic reflected weight identity]
\label{cor:quadratic-rwi}
Assume the image-choice half-space Stokes package of Theorem~\ref{thm:doubling-rwi}. For every freely
generated PVA~$V$ of jet order $\le 2$, the
correction $\delta_2(V)$ vanishes:
$\operatorname{RWI}(J^{\le 2}V)=0$.
\end{corollary}

\begin{proof}
Immediate from Theorem~\ref{thm:doubling-rwi} once the
orientation-compatible image-choice compactification and reflected
Stokes pushforward required there have been constructed. No
renormalized BV quantization is used in this corollary.
\end{proof}

\begin{remark}[Three independent perspectives on the quadratic case]
The quadratic case has three independent checks:
(1)~the valence-shift mechanism (the quadratic vertex has one extra
edge compared to the affine vertex, shifting the form-degree/dimension
balance); (2)~the Dolgushev equivariant formality theorem (guaranteeing
existence of the equivariant star product for any smooth Poisson
structure); (3)~worked examples (every quadratic PVA arising as a
semiclassical limit of a known vertex algebra deformation-quantizes by
construction). The doubling principle isolates the analytic
input: an orientation-compatible image-choice FM compactification of
the half-space reflection calculus. Once that pushforward theorem is
available, the reflected identity is the image-choice expansion of the
standard whole-space Stokes identity.
\end{remark}

\subsection{The cubic obstruction and the localized residue identity}

\begin{construction}[Cubic localization of $\delta_3$]
\label{cons:cubic-localization}
At jet order $3$ (cubic Poisson correction), the single-insertion graphs
have a cubic vertex
$\pi^{ij}_{r,3}=c^{ij}_{r,\ell mn}a^\ell a^m a^n\lambda^r$
with valence $4+r$. The relevant one-loop graph $\Gamma_3^{(1)}$ is a
\emph{tadpole}: a single cubic vertex with one self-loop (propagator
connecting two legs of the same vertex) and one external leg.

The reflected weight $w_{\Gamma_3^{(1)}}^{\mathrm{eq}}$ reduces to a
residue integral over the self-image face where the loop propagator
connects a bulk vertex to its image:
\[
\delta_3(V)\big|_{\Gamma_3^{(1)}}
=
\int_{t>0}
\operatorname{Res}_{t=0}
\Bigl(
c^{ij}_{r,\ell mn}\,K(z,t;z,-t)\,K^{\mathrm{refl}}(z,t;z'_2,t'_2)
\Bigr)
\,\mathrm{d}z\,\mathrm{d}\bar z\,\mathrm{d}t.
\]
The propagator $K(z,t;z,-t)$ connecting a point to its reflection is the
\emph{self-image propagator}, whose behavior at $t\to 0$ controls
the convergence of this integral. The displayed expression is a residue
model for the self-image edge, not an ordinary unrenormalized value of
the naive product kernel at zero holomorphic separation.
\end{construction}

\begin{corollary}[Cubic reflected weight identity]
\label{cor:cubic-rwi}
Assume the image-choice half-space Stokes package of Theorem~\ref{thm:doubling-rwi}. The jet-order-$3$
correction $\delta_3(V)$ vanishes for every
freely generated PVA~$V$ satisfying the Jacobi identity. The
mechanism, revealed by the Jacobi-reflection decomposition below
(Theorem~\ref{thm:jacobi-reflection-decomposition}), is a cancellation
between the tadpole graph $\Gamma_3^{(1)}$ and the \emph{reflected
triangle graph}; the doubling theorem
(Theorem~\ref{thm:doubling-rwi}) proves this cancellation holds by
identifying it with the image-choice pushforward of a Stokes identity
on the doubled space.
\end{corollary}

\begin{proof}
Immediate from Theorem~\ref{thm:doubling-rwi} after imposing the same
orientation-compatible image-choice compactification and self-image
renormalization hypotheses.
\end{proof}

\subsection{The Jacobi-reflection decomposition}

The jet-order scheme reveals the precise role of the PVA Jacobi
identity in the half-space problem. The boundary strata of the
compactified half-space configuration space
$\overline{\operatorname{Conf}}^{\partial}_{V(\Gamma)}(X)$
split into four types
(Construction~\ref{cons:half-space-colored-operad}): bulk clusters,
boundary clusters, mixed clusters, and plus-diagonal/self-image faces. The
contributions of the first three types to the reflected weight identity
are algebraically determined; the fourth is the genuinely analytic
content.

\begin{theorem}[Jacobi-reflection decomposition]
\label{thm:jacobi-reflection-decomposition}
Assume the logarithmic half-space graph weights are defined on the finite jet--weight quotient containing
jet order $N$. The jet-order-$N$ correction $\delta_N(V)$ decomposes as
\begin{equation}\label{eq:jacobi-reflection-decomposition}
\delta_N(V) \;=\; \delta_N^{\mathrm{alg}}(V)
\;+\; \delta_N^{\mathrm{refl}}(V),
\end{equation}
where:
\begin{enumerate}
\item $\delta_N^{\mathrm{alg}}(V)$ is the sum of contributions from
 bulk, boundary, and mixed strata. It is completely determined by the
 PVA structure constants of $V$ (via the Jacobi identity on
 $\FM_k(\C)$, the $E_1$ associativity on $\Conf_k^{<}(\R)$, and the
 Swiss-cheese module axioms on the mixed strata).
\item $\delta_N^{\mathrm{refl}}(V)$ is the sum of contributions from
 plus-diagonal and self-image faces, involving the self-image propagator
 $K(z,t;\,z,-t)$.
\end{enumerate}
The reflected weight identity at jet order $N$ is the identity
\begin{equation}\label{eq:refl-equals-alg}
\delta_N^{\mathrm{refl}}(V) \;=\; -\,\delta_N^{\mathrm{alg}}(V).
\end{equation}
\end{theorem}

\begin{proof}
The Stokes identity on
$\overline{\operatorname{Conf}}^{\partial}_{V(\Gamma)}(X)$ expresses
$\delta_N(V)$ as a sum over codimension-one boundary strata. The
four stratum types of
Construction~\ref{cons:half-space-colored-operad} partition the
boundary into algebraic and reflection parts.

\emph{Bulk strata.}
A subset of interior vertices collapses away from $\partial X$.
The residue along such a stratum is a product of a subgraph
weight (computing an interior operation) and a quotient-graph weight.
The relations among these residues are exactly the $A_\infty$
identities on $\FM_k(\C)$, i.e., the PVA Jacobi identity at jet
order $N$ (Lemma~\ref{lem:jet-jacobi}).

\emph{Boundary strata.}
A subset of boundary vertices collapses along $\partial X$.
The residue factors through the $E_1$ operad structure on
$\Conf_k^{<}(\R)$, giving the associative composition of boundary
operations. These are algebraically determined by the boundary
algebra structure.

\emph{Mixed strata.}
Interior vertices collide with boundary vertices. The residue
computes the Swiss-cheese module action (bulk observables acting on
boundary observables), controlled by the module axioms.

\emph{Reflection strata.}
An interior vertex at $(z,t)$ collides with its image
$(z,-t)$. These strata have no algebraic counterpart: they
arise purely from the half-space geometry and the method of images.
Their contributions are computed by the self-image propagator
$K(z,t;\,z,-t)$ and its derivatives.

The decomposition~\eqref{eq:jacobi-reflection-decomposition} is the
partition of the boundary sum into these four types, grouped as
algebraic (first three) and reflection (fourth). The total Stokes
identity $\delta_N(V)=0$ is then
equivalent to~\eqref{eq:refl-equals-alg}.
\end{proof}

\begin{remark}[What the Jacobi identity controls and what it does not]
\label{rem:jacobi-controls}
The PVA Jacobi identity determines
$\delta_N^{\mathrm{alg}}(V)$ completely: it is a specific
computable expression in the structure constants
$\pi^{ij}_{k,m}(\boldsymbol a)$ of $V$. The reflection part
$\delta_N^{\mathrm{refl}}(V)$ is an integral over the self-image
propagator, an analytic quantity that depends on the specific BV
propagator of the half-space theory, not on the PVA structure
alone.

The reflected weight identity at jet order $N$ therefore asks:
\emph{does the analytic reflection integral match the algebraically
determined expression?} The Jacobi identity constrains the target
but does not compute the source.

For the affine case ($N=1$): both sides vanish. The algebraic side
vanishes by the $E_1$ compatibility of the affine bar-cobar
adjunction. The reflection side vanishes by degree counting (the
self-image propagator for linear Poisson structures has too many
edges).

For general $N$: neither side vanishes. The identity
$\delta_N^{\mathrm{refl}}=-\delta_N^{\mathrm{alg}}$ is a
non-trivial identity between an analytic integral and an algebraic
expression.
\end{remark}

\begin{theorem}[Algebraic determination of $\delta_N^{\mathrm{alg}}$]
\label{thm:algebraic-determination}
For a freely generated PVA $V$ satisfying the Jacobi identity, the
algebraic part $\delta_N^{\mathrm{alg}}(V)$ is a specific polynomial
expression in the structure constants $\pi^{ij}_{k,m}$ of $V$ for
$m\le N$ and the structure constants $f^{ij}_{k,\ell}$ of the affine
truncation $J^{\le 1}V$. Explicitly, $\delta_N^{\mathrm{alg}}(V)$
is a sum over trees and one-loop graphs on the configuration space
$\FM_k(\C)\times\Conf_k^{<}(\R)$ with:
\begin{itemize}
\item one vertex carrying the degree-$N$ Poisson kernel
 $\pi^{ij}_{k,N}$,
\item all other vertices carrying the affine kernel
 $\pi^{ij}_{k,1}$,
\item edges carrying the \emph{unreflected} propagator $K$
 (not $K^{\mathrm{refl}}$).
\end{itemize}
These are exactly the graphs that would appear in the
\emph{whole-space} Kontsevich integral with a single nonlinear
insertion.
\end{theorem}

\begin{proof}
The bulk, boundary, and mixed strata of
$\overline{\operatorname{Conf}}^{\partial}(X)$ are identified with
the boundary strata of the whole-space configuration space
$\overline{\operatorname{Conf}}(\C\times\R)$ (i.e., the standard FM
boundary structure without reflection). The residues along these
strata involve only the unreflected propagator $K$, since the
image-charge term $K(z,t;\,z',-t')$ contributes exclusively to
plus-diagonal and self-image faces. The jet-order-$N$ contribution from these
unreflected graphs, with one degree-$N$ insertion in an otherwise
affine background, is completely determined by the $A_\infty$
relations on $\FM_k(\C)$ (PVA Jacobi) and the $E_1$ relations on
$\Conf_k^{<}(\R)$.
\end{proof}

\begin{corollary}[The cubic identity as analytic obstruction under image-choice Stokes]
\label{cor:cubic-identity}
At jet order $3$, the reflected weight identity reduces to:
\begin{equation}\label{eq:cubic-identity}
\delta_3^{\mathrm{refl}}(V)
\;=\;
-\,\delta_3^{\mathrm{alg}}(V).
\end{equation}
The left side is the self-image propagator integral of
Construction~\ref{cons:cubic-localization} (a tadpole graph with the
self-image kernel $K(z,t;\,z,-t)$ on the loop). The right side is a
specific polynomial expression in the cubic structure constants
$c^{ij}_{r,\ell mn}$ and the affine structure constants
$f^{ij}_{k,\ell}$, computed by the unreflected one-loop Kontsevich
integral with a single cubic insertion.

The PVA Jacobi identity for $V$ determines
$\delta_3^{\mathrm{alg}}(V)$ uniquely. The cubic reflected weight
identity (Corollary~\ref{cor:cubic-rwi}) is equivalent to the
\emph{single analytic identity}~\eqref{eq:cubic-identity}.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{thm:jacobi-reflection-decomposition} at $N=3$.
The algebraic part $\delta_3^{\mathrm{alg}}$ is computed by
Theorem~\ref{thm:algebraic-determination} as a sum over trees and
one-loop graphs with one cubic vertex in an affine background. The
reflection part $\delta_3^{\mathrm{refl}}$ is the self-image integral
of Construction~\ref{cons:cubic-localization}. The reflected weight
identity is the equality of these two expressions.
\end{proof}

\subsection{The doubling theorem}

The Jacobi-reflection decomposition identifies the remaining analytic
content of the half-space theorem: the reflection integrals must match
the algebraically determined expressions. The doubling construction
embeds the half-space problem into the doubled whole-space, where the
standard Kontsevich--FM Stokes calculation applies after the
image-choice compactification has separated ordinary, plus, and
self-image faces.

\begin{construction}[The doubled configuration space]
\label{cons:doubled-configuration}
Let $X=\C\times\R_{\ge 0}$ be the half-space and
$X_{\mathrm{dbl}}=\C\times\R$ the doubled space. The reflection
$\sigma\colon X_{\mathrm{dbl}}\to X_{\mathrm{dbl}}$,
$\sigma(z,t)=(z,-t)$, has $X$ as a fundamental domain and
fixed-point locus $\C\times\{0\}\subset X_{\mathrm{dbl}}$.

On the dense open locus where all bulk normal coordinates satisfy
$t_i>0$, the \emph{method-of-images map}
\[
\iota\colon\Conf_n(X)
\hookrightarrow
\Conf_{2n}(X_{\mathrm{dbl}})^{\sigma}
\]
sends a configuration $(p_1,\ldots,p_n)$ with $p_i=(z_i,t_i)\in X$ to
the $\sigma$-invariant configuration
$(p_1,\ldots,p_n,\sigma(p_1),\ldots,\sigma(p_n))$. Each original
point is paired with its image $\sigma(p_i)=(z_i,-t_i)$. At boundary
faces this is not an embedding into the ordinary doubled configuration
space, because $p_i=\sigma(p_i)$ when $t_i=0$; the compactified
construction therefore uses the image-choice blowup package of
Assumption~\ref{assum:kz-analytic-sdr-package}, with plus-diagonals and
self-image faces separated before pushforward.
\end{construction}

\begin{proposition}[Reflected propagator as signed image pullback]
\label{prop:reflected-pullback}
Let $\omega_{ij}$ denote the propagator $1$-form on
$\Conf_2(X_{\mathrm{dbl}})$ for the HT theory on the
whole space. On the dense image locus, and after logarithmic extension
to the image-choice compactification, the reflected propagator form on
$\Conf_2(X)$ is
\begin{equation}\label{eq:refl-as-pullback}
\omega_{ij}^{\mathrm{refl}}
\;=\;
ev_{ij}^{-*}\omega
\;+\;
\epsilon_{ij}\,ev_{ij}^{+*}\omega,
\end{equation}
where $ev^{-}$ evaluates the ordinary difference
$(z_i-z_j,t_i-t_j)$, $ev^{+}$ evaluates the image difference
$(z_i-z_j,t_i+t_j)$, and $\epsilon_{ij}$ is the field-parity
reflection sign.
\end{proposition}

\begin{proof}
On the dense locus this is just the method-of-images formula
$K^{-}+\epsilon_{ij}K^{+}$. The ordinary term is evaluated on
$(z_i-z_j,t_i-t_j)$ and the image term on $(z_i-z_j,t_i+t_j)$. The
extension to boundary faces is not formal from the ordinary doubled
configuration space; it is precisely the log-extension and sign datum
included in Assumption~\ref{assum:kz-analytic-sdr-package}.
\end{proof}

\begin{theorem}[The reflected weight identity from the doubling
principle]
\label{thm:doubling-rwi}
Let the weight forms arise from a logarithmic $\SCchtop$-algebra
(Definition~\ref{def:log-SC-algebra}). Assume the reflected
half-space compactification is constructed with image-choice expansion,
field-parity image signs, self-image renormalization, and an
orientation-compatible pushforward from the doubled FM compactification
to the fundamental-domain compactification. Then the reflected weight
identity~\eqref{eq:reflected-weight-identity} holds for every
admissible graph $\Gamma$ on the half-space $X=\C\times\R_{\ge 0}$.
\end{theorem}

\begin{proof}
The proof is the formal Stokes reduction once the stated
compactification package is available.

\medskip\noindent\textbf{Step~1: Image-choice expansion.}
For each edge $e=(i,j)$ of $\Gamma$, the reflected form is a signed
sum
\[
\omega^{\mathrm{refl}}_e
=
\omega_{ij}+\varepsilon_e\,\omega_{i,\sigma(j)},
\]
where $\varepsilon_e$ is determined by the field parity and boundary
condition at the reflected endpoint. Hence
$\bigwedge_{e\in E(\Gamma)}\omega_e^{\mathrm{refl}}$ is the sum over
all image-choice graphs, not the form of a single doubled graph with
all image edges simultaneously present.

\medskip\noindent\textbf{Step~2: FM compactification of the doubled
space.}
By Definition~\ref{def:log-SC-algebra}, the propagator forms $\omega_{ij}$ extend to
logarithmic $1$-forms on the Fulton--MacPherson compactification
$\overline{\Conf}_{2n}(X_{\mathrm{dbl}})$. The collision divisors
$D_S$ (for subsets $S\subseteq\{1,\ldots,2n\}$) are the standard
FM boundary strata. The Arnold--Orlik--Solomon relations (Theorem~\ref{thm:FM-calculus}(c)) ensure
that codimension-$2$ corners cancel pairwise.

The doubled FM compactification has ordinary collision strata between
original points $(D_{\{i,j\}})$, between image points
$(D_{\{\sigma(i),\sigma(j)\}})$, and between an original point and an
image point $(D_{\{i,\sigma(j)\}})$. Their half-space images are not
ordinary fixed-locus strata: compactification and passage to the
fundamental domain do not commute at the boundary fixed locus. The
image-choice blowup separates the resulting plus-diagonal faces
$z_i=z_j,\ t_i+t_j=0$ and the self-image faces $t_i=0$ before
residue pushforward.

\medskip\noindent\textbf{Step~3: Stokes' theorem on the doubled
space.}
For each image-choice graph $\Gamma_\chi$, the wedge product
$\bigwedge_{e\in E(\Gamma_\chi)}\omega_e$ is a closed
logarithmic form of top degree on the interior of
$\overline{\Conf}_{2n}(X_{\mathrm{dbl}})$ (each factor is closed by
Definition~\ref{def:log-SC-algebra}, and the FM compactification has normal-crossing boundary
divisors). Stokes' theorem gives
\begin{equation}\label{eq:stokes-doubled}
0
\;=\;
\int_{\overline{\Conf}_{2n}(X_{\mathrm{dbl}})}
d\Bigl(\bigwedge_{e\in E(\Gamma_\chi)}\omega_e\Bigr)
\;=\;
\sum_{S}\int_{D_S}
\operatorname{Res}_{D_S}\Bigl(
\bigwedge_{e\in E(\Gamma_\chi)}\omega_e\Bigr).
\end{equation}
Summing over $\chi$ gives the whole-space weight identity for the
image-choice expansion.

\medskip\noindent\textbf{Step~4: Restriction to the
$\sigma$-invariant locus.}
The $\sigma$-action on $\overline{\Conf}_{2n}(X_{\mathrm{dbl}})$
preserves the FM boundary structure: $\sigma$ maps $D_S$ to
$D_{\sigma(S)}$. The assumed orientation-compatible pushforward to the
fundamental-domain compactification sends the summed
identity~\eqref{eq:stokes-doubled} to:
\begin{equation}\label{eq:stokes-restricted}
0
\;=\;
\sum_{S\;\sigma\text{-inv}}
\int_{D_S^{\sigma}}
\operatorname{Res}_{D_S}\Bigl(
\sum_\chi\bigwedge_{e\in E(\Gamma_\chi)}\omega_e\Bigr)\Bigg|_{\sigma}.
\end{equation}
Here $D_S^\sigma$ denotes the image-choice face in the
fundamental-domain compactification obtained from the paired doubled
corner, not the literal fixed locus of an ordinary doubled FM divisor.
These faces map to the four types of
Construction~\ref{cons:half-space-colored-operad}:
\begin{itemize}
\item $S\subseteq\{1,\ldots,n\}$ (original points only): $D_S^{\sigma}$
 is a \textbf{bulk collision stratum} in $X$.
\item $S\subseteq\{1,\ldots,n\}$ restricted to the boundary
 $t_i=0$: $D_S^{\sigma}$ is a \textbf{boundary collision stratum}.
\item $S$ contains both original and image points from the same pair
 $\{i,\sigma(i)\}$: $D_S^{\sigma}$ is a \textbf{mixed
 bulk-boundary stratum}.
\item $S=\{i,\sigma(j)\}$ with $i\neq j$, together with
 $S=\{i,\sigma(i)\}$ after self-image blowup: $D_S^\sigma$ is a
 \textbf{plus-diagonal or self-image face}.
\end{itemize}
By the image-choice expansion and Proposition~\ref{prop:reflected-pullback},
the pushed-forward form is
$\bigwedge_e\omega_e^{\mathrm{refl}}$ on $\Conf_n(X)$. Therefore
the fundamental-domain Stokes
identity~\eqref{eq:stokes-restricted} is exactly the reflected
weight identity~\eqref{eq:reflected-weight-identity}.
\end{proof}

\begin{construction}[Khan--Zeng BV fields and action]
\label{cons:kz-bv-fields-action}
Let
\[
V=\Sym\bigl(\C[\partial]\,a^1\oplus\cdots\oplus
\C[\partial]\,a^N\bigr)
\]
be a freely generated PVA with finite jet-order polynomial
$\lambda$-bracket
\[
\{a^i{}_\lambda a^j\}
=
\sum_{r=0}^{R} \Pi^{ij}_{r}
\bigl(a,\partial a,\ldots,\partial^{R}a\bigr)\lambda^r .
\]
Use the De Sole--Kac Hamiltonian-matrix convention: the operator
contracted with $\eta_i\eta_j$ is the transpose of the displayed
$\lambda$-bracket matrix,
\[
\Pi^{ij}_{V}(\partial_z;\Phi)
=
\sum_{r=0}^{R}\Pi^{ji}_{r}
\bigl(\Phi,\partial_z\Phi,\ldots,\partial_z^{R}\Phi\bigr)
\partial_z^r
\]
with $\partial_z$ acting on the field to its right. PVA
skew-symmetry is the formal skew-adjointness
$(\Pi_V)^*=-\Pi_V$, where
$(f\partial_z^r)^*=(-\partial_z)^r\circ f$ after integration by
parts in the local-functional quotient. On
$X=\C_z\times\R_{\ge 0,t}$ the BV field complex is the completed
shifted cotangent complex
\[
\cE_V
=
\Omega^{0,\bullet}(\C)
\widehat\otimes
\Omega^\bullet(\R_{\ge0})
\widehat\otimes
T^*[-1]J_\infty\Spec\C[a^1,\ldots,a^N],
\]
with Darboux coordinates $(\Phi^i,\eta_i)$ and BV form
\[
\omega_{\mathrm{BV}}
=
\int_X dz\,\sum_i \delta\Phi^i\,\delta\eta_i .
\]
The Khan--Zeng classical action is
\begin{equation}\label{eq:kz-bv-action-general}
S_{0,V}
=
\int_X dz\left(
\sum_i\eta_i(\dbar+d_t)\Phi^i
+
\frac12\sum_{i,j}\eta_i\,
\Pi^{ij}_{V}(\partial_z;\Phi)\,\eta_j
\right),
\end{equation}
Write
\[
I_{\mathrm{kin},V}=\int_X dz\,\sum_i\eta_i(\dbar+d_t)\Phi^i,
\qquad
I_{\mathrm{int},V}
=
\frac12\int_X dz\,
\sum_{i,j}\eta_i\,
\Pi^{ij}_{V}(\partial_z;\Phi)\,\eta_j .
\]
Thus $S_{0,V}=I_{\mathrm{kin},V}+I_{\mathrm{int},V}$, and the
free BRST differential is
$Q=\{I_{\mathrm{kin},V},{-}\}_{\mathrm{BV}}=\dbar+d_t$ on the
linearized fields. The Neumann boundary condition is $i^*\eta_i=0$.
The boundary field
complex is
\[
\cE_{\partial,V}:=
i^*\cE_V/(i^*\eta_1,\ldots,i^*\eta_N),
\]
so the boundary generators are the jets of the unrestricted
$\Phi^i$-fields.
\end{construction}

\begin{lemma}[PVA Jacobi is the classical master equation]
\label{lem:kz-cme-pva-jacobi}
For the action \eqref{eq:kz-bv-action-general},
\[
\frac12\{S_{0,V},S_{0,V}\}_{\mathrm{BV}}=0
\]
if and only if the $\lambda$-bracket of $V$ satisfies the PVA
skew-symmetry and Jacobi identities.
\end{lemma}

\begin{proof}
The BV bracket is computed in the local-functional quotient of
densities by total $\partial_z$, $\dbar$, and $d_t$ derivatives.
The kinetic term gives $Q=\dbar+d_t$ and $Q^2=0$. Therefore
\[
\frac12\{S_{0,V},S_{0,V}\}_{\mathrm{BV}}
 =
 QI_{\mathrm{int},V}
 +\frac12\{I_{\mathrm{int},V},I_{\mathrm{int},V}\}_{\mathrm{BV}} .
\]
The first summand is the mixed de Rham--Dolbeault total derivative of
the interaction density, hence vanishes after integration by parts,
compact support or the usual Costello support cutoff, and the boundary
condition $i^*\eta=0$. The sesquilinearity of the
$\lambda$-bracket is the separate statement that this local-functional
calculus is compatible with the $\partial_z$-jet variables.
The bracket of the interaction term with itself is the variational
Schouten bracket of the Hamiltonian bidifferential
$\Pi_V$:
\[
\frac12\{I_{\mathrm{int},V},I_{\mathrm{int},V}\}_{\mathrm{BV}}
=
\frac16\int_X dz\,
\sum_{i,j,k}
[\Pi_V,\Pi_V]^{ijk}_{\mathrm{var}}
(\partial_z;\Phi)\,\eta_i\eta_j\eta_k .
\]
On the free finite-jet algebra the odd monomials
$\eta_i\eta_j\eta_k$ detect the variational Schouten tensor in this
quotient. The De Sole--Kac Hamiltonian-operator criterion identifies
the two conditions
\[
(\Pi_V)^*=-\Pi_V,
\qquad
[\Pi_V,\Pi_V]_{\mathrm{var}}=0
\]
with, respectively, PVA skew-symmetry and the PVA Jacobi identity for
the corresponding $\lambda$-bracket. Thus the classical master
equation is precisely the PVA Hamiltonian condition. This local
equivalence does not use inner holomorphic translation; innerness
enters the topological enhancement and the half-space quantization
theorem.
\end{proof}

\begin{construction}[Reflected graph-weight ansatz]
\label{cons:kz-renormalized-weights}
Choose the product HT gauge for the kinetic operator
$Q=\dbar+d_t$. The singular holomorphic part of the gauge parametrix is
modeled by the logarithmic kernel
\[
K(z,t)=\frac{\Theta(t)}{2\pi z},
\qquad
QK=\delta_\C\boxtimes\delta_\R
\quad\text{modulo the smooth and boundary correction terms of the
chosen HT gauge}.
\]
The reflected half-space kernel is not a universal plus-image kernel.
For an edge $e$ its singular part has the form
\[
K^{\mathrm{refl}}_e(z,t;z',t')
=
K(z-z',t-t')+\varepsilon_e K(z-z',t+t'),
\]
where the sign $\varepsilon_e$ is fixed by the parity of the endpoint
field and the boundary condition; an $\eta$-endpoint satisfying
$i^*\eta=0$ carries odd image parity. Self-image edges are part of the
renormalization data and are not evaluated by the naive kernel at zero
holomorphic separation.
For an admissible graph $\Gamma$ with vertices labelled by the
finite polynomial tensors in $\Pi_V$, the scale-$L$ reflected weight is
defined, after choosing the half-space/raviolo compactification and
subtraction scheme, by the image-choice expansion
\[
W_{\Gamma,L}^{\mathrm{refl}}(V)
=
\int_{\overline{\Conf}^{\partial}_{\Gamma}(X)}
\bigwedge_{e\in E(\Gamma)} K^{\mathrm{refl}}_{e,L}\,
\prod_{v\in V(\Gamma)}\Pi_{V,v}.
\]
The integral is interpreted on each finite jet--weight quotient.
Theorem~\ref{thm:doubling-rwi} identifies its boundary residues with
the doubled whole-space Stokes residues only after the
orientation-compatible image-choice compactification stated there has
been constructed.
\end{construction}

\begin{assumption}[KZ finite-quotient analytic SDR package]
\label{assum:kz-analytic-sdr-package}
For the finite-type KZ input $V$, fix a complete filtered system of
jet--weight quotients
\[
\widehat C_V^\bullet=\varprojlim_{N,w}C_{V,N,w}^\bullet
\]
with surjective transition maps. On each quotient the renormalized
half-space BV algebra carries operators
$Q_{N,w}$, $\Delta_{N,w}$, the BV bracket, and the differential
$d_{V,N,w}=Q_{N,w}+[I_{\mathrm{int},V},{-}]_{\mathrm{BV}}$, and these
operators commute with truncation. The reflected graph weights of
Construction~\ref{cons:kz-renormalized-weights} extend logarithmically
to the half-space/raviolo compactification and satisfy the
orientation-compatible image-choice Stokes identity of
Theorem~\ref{thm:doubling-rwi}. Finally there are quotient-compatible
SDR data
\[
\mathrm{id}-H_{V,N,w}
=d_{V,N,w}h_{V,N,w}+h_{V,N,w}d_{V,N,w},
\qquad
h_{V,N,w}H_{V,N,w}=H_{V,N,w}h_{V,N,w}=h_{V,N,w}^2=0,
\]
preserving locality and the renormalization scheme, and the harmonic
projection of every QME obstruction defined below vanishes:
$H_{V,N,w}(\mathrm{Ob}_{\ell,N,w})=0$.
\end{assumption}

\begin{proposition}[Finite-quotient QME recursion]
\label{prop:kz-all-loop-counterterm-recursion}
Assume Assumption~\ref{assum:kz-analytic-sdr-package}.
Let $C_V^\bullet$ be the completed local
$\SCchtop$ convolution complex controlling the KZ half-space
BV theory:
\[
C_V^\bullet
=
\Hom^{\mathrm{col}}\!\bigl(
B(\SCchtop),
\End^{\mathrm{col}}_{\cE_V,\cE_{\partial,V}}\bigr)[[\hbar]],
\qquad
d_V=[I_{\mathrm{int},V},{-}]_{\mathrm{BV}}+Q .
\]
Put $I_{0,V}:=I_{\mathrm{int},V}$ for the recursive interaction
expansion. On each finite quotient of
Assumption~\ref{assum:kz-analytic-sdr-package}, define
\begin{equation}\label{eq:kz-qme-recursion}
I_{\ell,V,N,w}
=
-h_{V,N,w}\!\left(
\Delta_{N,w} I_{\ell-1,V,N,w}
+\frac12
\sum_{\substack{a+b=\ell\\ a,b\geq 1}}
\{I_{a,V,N,w},I_{b,V,N,w}\}_{\mathrm{BV}}
\right),
\qquad \ell\geq 1 .
\end{equation}
Then the compatible inverse limit
$I_V(\hbar)=I_{\mathrm{int},V}+\sum_{\ell\ge1}\hbar^\ell I_{\ell,V}$
satisfies
\[
\left(QI_V+\frac12\{I_V,I_V\}_{\mathrm{BV}}
+\hbar\Delta I_V\right)=0,
\qquad\text{equivalently}\qquad
\left(Q+\hbar\Delta\right)e^{I_V(\hbar)/\hbar}=0 .
\]
\end{proposition}

\begin{proof}
Set
\[
\mathrm{Ob}_{\ell}
=
\Delta I_{\ell-1,V}
+\frac12
\sum_{\substack{a+b=\ell\\ a,b\geq 1}}
\{I_{a,V},I_{b,V}\}_{\mathrm{BV}} .
\]
At order $\hbar^\ell$ the QME is
$d_V I_{\ell,V}+\mathrm{Ob}_\ell=0$. Its image on the quotient
$C_{V,N,w}^\bullet$ is a cocycle
\[
\mathrm{Ob}_{\ell,N,w}\in C_{V,N,w}^1,
\qquad
[\mathrm{Ob}_{\ell,N,w}]
\in H^1(C_{V,N,w}^\bullet,d_{V,N,w}),
\]
and this cohomology class is the obstruction to solving the
order-$\ell$ QME on that quotient. The identities
\[
Q^2=0,\qquad [Q,\Delta]=0,\qquad
\Delta\{F,G\}=\{\Delta F,G\}\pm\{F,\Delta G\}
\]
together with the BV Jacobi identity and the inductive QME through
order $\ell-1$ give $d_{V,N,w}\mathrm{Ob}_{\ell,N,w}=0$. By
Assumption~\ref{assum:kz-analytic-sdr-package}, the harmonic
projection of this cocycle vanishes:
$H_{V,N,w}(\mathrm{Ob}_{\ell,N,w})=0$.
Therefore
\[
\mathrm{Ob}_{\ell,N,w}
=d_{V,N,w}h_{V,N,w}\mathrm{Ob}_{\ell,N,w}
+h_{V,N,w}d_{V,N,w}\mathrm{Ob}_{\ell,N,w}
=d_{V,N,w}h_{V,N,w}\mathrm{Ob}_{\ell,N,w}.
\]
The choice \eqref{eq:kz-qme-recursion} solves the order-$\ell$
equation. The quotient-compatibility and surjectivity in
Assumption~\ref{assum:kz-analytic-sdr-package} give the complete
filtered inverse limit and no $\varprojlim^1$ obstruction appears.
\end{proof}

\begin{theorem}[General half-space BV quantization under the KZ analytic
package]
\label{thm:general-half-space-bv}
Let $V$ be a finite-type freely generated Poisson vertex algebra
whose $\lambda$-bracket is polynomial of finite jet order, and assume
that the holomorphic translation operator is inner. Assume
Assumption~\ref{assum:kz-analytic-sdr-package}. For the
Khan--Zeng Poisson--vertex sigma model on
$\C\times\R_{\ge 0}$ with Neumann boundary condition, the
construction above gives a renormalized perturbative BV quantization
to all orders in $\hbar$. The reflected propagator realizes the
logarithmic $\SCchtop$ weights, the effective action solves the
quantum master equation, and the boundary factorization algebra is a
vertex algebra deforming~$V$.
\end{theorem}

\begin{proof}
Construction~\ref{cons:kz-bv-fields-action} gives the BV field
complex, symplectic form, action, and Neumann boundary condition.
Lemma~\ref{lem:kz-cme-pva-jacobi} proves the classical master
equation from the PVA axioms.
Assumption~\ref{assum:kz-analytic-sdr-package} supplies the
renormalized reflected logarithmic graph weights, compactification,
finite-quotient SDR, and harmonic obstruction vanishing on the
half-space. Proposition~\ref{prop:kz-all-loop-counterterm-recursion}
constructs the counterterms and solves the QME to all orders.

The resulting graph operations satisfy the reflected weight identity
by the Stokes clause of
Assumption~\ref{assum:kz-analytic-sdr-package}. By
Theorem~\ref{thm:half-space-reduction}, that identity is exactly the
Stokes term in the boundary $A_\infty$/factorization relations. Hence
the boundary operations assemble into a vertex algebra. Its associated
graded at $\hbar=0$ is the original PVA because the tree-level
singular boundary bracket is the $\lambda$-bracket used in
\eqref{eq:kz-bv-action-general}.
\end{proof}

\begin{remark}[What the doubling proves]
The doubling argument proves a formal reduction of the reflected
weight identity to ordinary whole-space Stokes after the
image-choice compactification, field-parity signs, self-image
renormalization, and orientation-compatible residue pushforward have
been constructed. On the dense locus $t_i>0$ the reflected propagator
is the signed method-of-images pullback. At boundary faces the
involution has fixed points, and the plus-diagonal and self-image

collision divisors.

Thus doubling supplies the correct Stokes equation, not the missing
renormalized BV package. That package is exactly
Assumption~\ref{assum:kz-analytic-sdr-package}.
\end{remark}

\text{Dolgushev equivariant formality}

\section{Bridge-theorem inputs}
\label{sec:bridge-theorem-inputs}

The doubling theorem assumes the image-choice half-space
Stokes package. For polynomial finite-jet Khan--Zeng models, the
classical BV package is constructed above; the renormalized all-loop
QME solution additionally requires
Assumption~\ref{assum:kz-analytic-sdr-package}. For a named family
one therefore separates the proved classical Hamiltonian condition
from the analytic SDR and obstruction-vanishing input.

\begin{proposition}[The affine PV sigma model is a logarithmic $\SCchtop$-algebra]%
\label{prop:affine-is-log-SC}
Let $\fg$ be semisimple and let $k$ be the tuple of affine scalar
levels, equivalently the invariant form
$k\,\kappa\in\mathrm{Sym}^2(\fg^*)^\fg$. The three-dimensional
Poisson--vertex sigma model of Khan--Zeng attached to the affine
Poisson vertex algebra $V_{\mathrm{aff}}(\fg, k)$, on
$X = \C_z \times \R_t$, satisfies the four inputs of
Theorem~\textup{\ref{thm:physics-bridge}}:
\begin{enumerate}[label=\textup{(\alph*)}]
\item a BV datum $(\mathcal E,Q,\omega_{\mathrm{BV}},I,B)$ with HT gauge fixing and $Q_{\mathrm{BRST}}^2 = 0$;
\item a factorized logarithmic HT parametrix with kernel homotopy
$Q_{\boxtimes}P_{\epsilon<L}=K_L-K_\epsilon$;
\item one-loop finiteness;
\item product-polynomial interactions.
\end{enumerate}
Consequently, the BV--BRST complex of the affine PV sigma model is a
logarithmic $\SCchtop$-algebra in the sense of
Definition~\textup{\ref{def:log-SC-algebra}}.
\end{proposition}

\begin{proof}
The four inputs are as follows.

\smallskip\noindent
\emph{Input \textup{(a)}: BV data.}
The affine PV sigma model is the mixed BF/Chern--Simons theory
(Proposition~\ref{prop:affine-mixed-bf-dictionary}). The BV data for
mixed BF theory on $\C \times \R$ is standard: the field space is
$\cF = T^*[-1](\Omega^{0,\bullet}_\C \widehat\otimes
\Omega^\bullet_\R[1] \otimes \fg)$, the $(-1)$-symplectic pairing is
the integration pairing $\langle \alpha, \beta \rangle = \int_X
\kappaChHodge(\alpha \wedge \beta)$, and the action $S = S_{\mathrm{mix}} +
I_{\mathrm{CS}}^{k\kappa}$ satisfies $\{S, S\}_{\mathrm{BV}} = 0$ by the
Jacobi identity for $\fg$ and the Bianchi identity for the invariant-form
Chern--Simons functional.
The HT gauge fixing uses the operator
$Q = \dbar_\C + d_t$ with $Q^2 = 0$, and the gauge-fixed propagator
has singular part $G_{\mathrm{sing}} = K_\C \otimes H_\R$ with
$K_\C(z) = 1/(2\pi z)$ and $H_\R(t) = \Theta(t)$, up to the smooth
$Q$-exact correction allowed in Theorem~\ref{thm:physics-bridge}
(Setup~\ref{setup:affine-pva-setup} and Step~(i) of that theorem).

\smallskip\noindent
\emph{Input \textup{(b)}: Factorized parametrix.}
This is the chosen HT gauge parametrix just described. The claim is
about its singular logarithmic part, not about a pure tensor being the
unique Green kernel for $\dbar_\C+d_t$.

\smallskip\noindent
\emph{Input \textup{(c)}: One-loop finiteness.}
The affine theory is strictly stronger: it is one-loop
\emph{exact} by Theorem~\ref{thm:affine-half-space-bv}\textup{(2)}.
One-loop exactness implies one-loop finiteness \emph{a fortiori}.
Alternatively, the Garner--Raghavendran--Williams mechanism applies
directly: the holomorphic--topological gauge has $d' = 1$ (one real
topological direction), which is the regime where their convergence
theorem guarantees one-loop finiteness without UV divergences
\cite{GRW21}.

\smallskip\noindent
\emph{Input \textup{(d)}: Product-polynomial interactions.}
The interaction vertices of the affine action
\eqref{eq:affine-kz-action} are:
\[
V_{\mathrm{cubic}} = f_{abc}\,B^a A^b A^c,
\qquad
V_{\mathrm{CS}} = k_{ab}\,A^a \partial A^b.
\]
Both are polynomial in the fields (cubic and quadratic,
respectively). No higher-order or transcendental interactions
appear.

\smallskip
The four inputs are exactly the logarithmic
factorization, parametrix, finiteness, and product-polynomial clauses
of Definition~\ref{def:log-SC-algebra} in this affine mixed
BF/Chern--Simons model; equivalently, Theorem~\ref{thm:physics-bridge}
applies to this input.
\end{proof}

Since the affine PV sigma model is a logarithmic $\SCchtop$-algebra

\begin{proposition}[Drinfeld--Sokolov reduction preserves the $\SCchtop$ structure]%
\label{prop:ds-preserves-SC}
Let $\cA$ be a logarithmic $\SCchtop$-algebra, and let
$f \in \fg$ be a nilpotent element with BRST datum
$(\fg, f, \chi)$. Then the Drinfeld--Sokolov reduction
$\mathrm{DS}_f(\cA)$, defined as the BRST cohomology
\[
\mathrm{DS}_f(\cA) := H^\bullet(\cA \otimes \cC_f,\, Q_{\mathrm{BRST}}),
\]
inherits the structure of a logarithmic $\SCchtop$-algebra.
Concretely, the weight forms of $\mathrm{DS}_f(\cA)$ are the BRST
reductions of the weight forms of $\cA$:
\[
\omega_k^{\mathrm{DS}} = H^\bullet\bigl(\omega_k \otimes \cC_f^{\otimes k},\,
Q_{\mathrm{BRST}}^{\otimes k}\bigr),
\]
and these inherit the product decomposition
$\omega_k^{\mathrm{DS}} = (\omega_k^{\mathrm{DS}})^{\mathrm{hol}} \otimes
(\omega_k^{\mathrm{DS}})^{\mathrm{top}}$
because the BRST differential $Q_{\mathrm{BRST}}$ acts
algebraically on the fibers and commutes with the
holomorphic--topological decomposition of the base.
\end{proposition}

\begin{proof}
The BRST complex $(\cA \otimes \cC_f, Q_{\mathrm{BRST}})$ is a dg
algebra over the base $\C \times \R$. The BRST differential
$Q_{\mathrm{BRST}} = Q_\cA + d_{\mathrm{gh}} + \chi$ acts on
the fiber (i.e., on the algebra generators and the ghosts $\cC_f$)
without mixing holomorphic and topological directions: $Q_\cA$ is the
original BV--BRST differential (which preserves the
holomorphic--topological splitting by hypothesis), $d_{\mathrm{gh}}$
is the ghost differential (algebraic, independent of $z$ and $t$),
and $\chi$ is the character (a scalar, independent of $z$ and $t$).

Therefore, if $\omega_k = \omega_k^{\mathrm{hol}} \otimes
\omega_k^{\mathrm{top}}$ factors, then the BRST reduction acts
separately on the coefficient algebra while leaving the base
forms unchanged. The product decomposition is preserved.

One-loop finiteness of the reduced theory follows from
one-loop finiteness of the original theory: the BRST spectral
sequence converges at finitely many pages (the BRST complex has
bounded ghost degree), and each page inherits the one-loop
finiteness from $\cA$ because the BRST differential does not
introduce new UV divergences (it is algebraic, not analytic).

Polynomial interactions are preserved because the BRST complex
is polynomial in the ghost generators, and the DS reduction of a
polynomial interaction is polynomial.
\end{proof}

\begin{corollary}[Virasoro and $W_N$ inherit the logarithmic $\SCchtop$ shadow]%
\label{cor:virasoro-wn-are-SC}
The Virasoro vertex algebra and all $W_N$ algebras obtained by
principal Drinfeld--Sokolov reduction from the affine family are
logarithmic $\SCchtop$-algebras. This proves preservation of the
product-formal Swiss-cheese shadow under DS reduction. It does not by
itself prove the nonlinear KZ all-loop half-space BV quantization,
which requires Assumption~\ref{assum:kz-analytic-sdr-package}.
\end{corollary}

\begin{proof}
The affine PV sigma model is a logarithmic $\SCchtop$-algebra
(Proposition~\ref{prop:affine-is-log-SC}). Principal
Drinfeld--Sokolov reduction preserves this structure
(Proposition~\ref{prop:ds-preserves-SC}). Virasoro is the principal
DS reduction of $\widehat{\mathfrak{sl}}_2$; $W_N$ is the principal DS
reduction of $\widehat{\mathfrak{sl}}_N$.
\end{proof}

\section{Synthesis}

For the affine Khan--Zeng Poisson--vertex sigma model, the half-space
BV theorem is proved by the affine one-loop exact theorem. For
finite-type nonlinear Khan--Zeng PVAs, the classical BV action and the
CME=PVA-Jacobi equivalence are proved; the all-loop QME, reflected
weight identity, and boundary vertex algebra require the KZ analytic
SDR package of Assumption~\ref{assum:kz-analytic-sdr-package}:
\begin{gather*}
\text{PVA}
\;\xrightarrow{\text{PV sigma model}}\;
\text{half-space BV theory}
\;\xrightarrow{\text{image-choice Stokes + SDR}}\;
\text{reflected weight identity}\\
\xrightarrow{\text{recognition}}\;
\text{boundary vertex algebra}.
\end{gather*}
The doubling construction (Theorem~\ref{thm:doubling-rwi}) identifies
the nonlinear obstruction surface with the image-choice Stokes
pushforward. The jet-order scheme (Theorem~\ref{thm:jet-inductive-rwi}),
the Jacobi-reflection
decomposition (Theorem~\ref{thm:jacobi-reflection-decomposition}),
and the cubic identity (Corollary~\ref{cor:cubic-identity}) become
consequences only after that compactification and orientation
pushforward have been constructed.

The affine half-space theorem (Theorem~\ref{thm:affine-half-space-bv})
is the first case: for linear Poisson structures, the doubled
Kontsevich integral localizes on wheel graphs, giving one-loop
exactness and the level shift $K_{\mathrm{eff}}=k+\hbar\,\delta_\fg$.
For nonlinear Poisson structures, the doubled integral does not
localize on wheels, and higher-loop contributions are generically
nonzero, but the Stokes identity (and hence the reflected weight
identity) holds at all loop orders by the FM/raviolo compactification,
the closedness of the propagator form, and the SDR package of
Assumption~\ref{assum:kz-analytic-sdr-package}.

\begin{remark}[Jet order and Euler--Koszul depth]%
\label{rem:jet-order-ek}%
Each jet order beyond linear introduces generators of
conformal weight $> 1$ on the boundary, increasing the
Euler--Koszul class (Vol~I,
Euler--Koszul class formalism).
Jet order~$1$ (affine): boundary has
$W = \{1^{\dim\fg}\}$, $\operatorname{ek} = 0$.
Jet order~$2$ (quadratic Poisson): boundary acquires
weight-$2$ generators, $\operatorname{ek} \geq 1$.
\end{remark}

The half-space BV theorem resolves the \emph{mixed color} of the Swiss-cheese
operad. The companion resolution, the \emph{open color}, is the
complete strictification theorem of
Chapter~\ref{chap:dg-shifted-factorization}
(Theorem~\ref{thm:complete-strictification}), where root multiplicity
one forces the spectral Drinfeld class to vanish. The two resolutions
are structurally parallel: one uses the $\sigma$-equivariance of the
doubled space, the other uses the one-dimensionality of root spaces.
Both say: the obstruction has nowhere to live.

\label{sec:ahs-k3-boundary}

\begin{remark}[K3 boundary]
\label{rem:ahs-K3-boundary}
The affine half-space theorem is local on
$\C\times\R_{\ge 0}$. It does not identify the
$K3\times E$ Hall--Borcherds object, a bi-based Ran-space datum, or a
Bruinier Heegner coefficient. Those objects require the Vol~III
Hall--Drinfeld/Borcherds construction and the Bruinier projection; the
half-space result supplies only the local affine boundary model at a
chosen degenerating fibre.
\end{remark}
