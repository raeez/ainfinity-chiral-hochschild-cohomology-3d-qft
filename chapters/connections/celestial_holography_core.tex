\section{Modular twisted/celestial holography from first principles}
\label{sec:modular-twisted-celestial-first-principles}

The modular trace principle (Principle~\ref{princ:modular-trace}) compares the disk-level projections of $\Theta^{\mathrm{oc}}$ (boundary chiral algebras, spectral $R$-matrices) with a genus-filtered modular deformation problem.  The disk term is tree-level transfer data; the positive-genus terms are loop/$1/N$ corrections governed by obstruction classes.

\subsection{Disk-level transfer and the all-genus lifting problem}
\label{subsec:problem-restated-true-level}

Kaluza--Klein reduction of $6$d holomorphic theories produces $3$d holomorphic--topological theories whose boundary chiral algebras coincide with defect chiral algebras of the unreduced model \cite{ZengTwistedHolography}.
The bulk chiral algebra of the holomorphic twist is commutative with a shifted Poisson bracket, and boundary chiral algebras carry higher $A_\infty$-type operations \cite{CostelloDimofteGaiottoBoundary}.
Regularized Feynman diagrams satisfy quadratic identities controlling holomorphic--topological OPE data at all degrees \cite{GKW25}.
Chiral quadratic duality is governed by Maurer--Cartan elements in $A^!\otimes B$, with curvature entering through curved chiral DG structures \cite{GuiLiZengQuadraticDuality}.

These results identify the boundary vertex algebra with the genus-$0$ Maurer--Cartan datum. The all-genus problem is the lifting problem for this datum in a complete cyclic deformation complex.

\subsection{The abstract modular package}
\label{subsec:abstract-modular-package}

The formal structure of modular completion.

\begin{definition}[Complete genus-filtered cyclic deformation package]
\label{def:complete-genus-filtered-package}
A \emph{complete genus-filtered cyclic deformation package} consists of the following data.
\begin{enumerate}[label=\textup{(\roman*)}]
\item A complete $\h$-adically filtered graded vector space
\[
\mathfrak g^{\mathrm{mod}}=\prod_{g\ge 0}\h^g\mathfrak g^{(g)}.
\]
\item A family of multilinear operations
\[
\ell_k^{(g)}: \Sym^k(\mathfrak g^{\mathrm{mod}})\longrightarrow
\mathfrak g^{\mathrm{mod}}[2-k],
\qquad g\ge 0,\ k\ge 1,
\]
which together define a complete cyclic $L_\infty$-algebra.
\item A distinguished genus-$0$ Maurer--Cartan element
\[
\pi\in \mathfrak g^{(0),1}
\]
solving the genus-$0$ Maurer--Cartan equation
\[
\sum_{k\ge 1}\frac1{k!}\,\ell_k^{(0)}(\pi^{\otimes k})=0.
\]
\end{enumerate}
A \emph{modular lift} of $\pi$ is a formal series
\[
\Pi(\h)=\pi+\sum_{g\ge 1}\h^g\Theta^{(g)}
\]
satisfying the full Maurer--Cartan equation
\begin{equation}
\label{eq:full-modular-mc}
\sum_{r\ge 0}\h^r\sum_{k\ge 1}\frac1{k!}\,
\ell_k^{(r)}\!\bigl(\Pi(\h)^{\otimes k}\bigr)=0.
\end{equation}
\end{definition}

\begin{remark}
For HT boundary theories, $\mathfrak g^{\mathrm{mod}}$ is the completed cyclic
boundary deformation complex tensored with a genus package such as
$R\Gamma(\ov{\M}_{\bullet,\bullet},\Q)$, or equivalently the graph/Feynman transform of the
genus-$0$ deformation theory. Definition
\ref{def:complete-genus-filtered-package} separates the formal deformation theory from the
physical realization.
\end{remark}

\begin{theorem}[Modular twisted/celestial transfer criterion]
\label{thm:modular-twisted-celestial-transfer}
Let $(T,\mathfrak b)$ be a twisted or celestial holographic model whose disk-level boundary
algebra $\A_\partial=\A_\partial(T,\mathfrak b)$ is identified, for a transverse boundary
condition, with the defect chiral algebra of the unreduced theory. Assume the following transfer
data are fixed:
\begin{enumerate}[label=\textup{(\roman*)}]
\item a complete genus-filtered cyclic deformation package
\[
\mathfrak g^{\mathrm{mod}}_\partial=\prod_{g\ge 0}\h^g\mathfrak g^{(g)}_\partial
\]
for the completed cyclic boundary deformation complex;
\item a distinguished genus-$0$ Maurer--Cartan element
\[
\pi_\partial\in \mathfrak g_\partial^{(0),1}
\]
encoding exactly the disk-level boundary OPE and the tree-level transferred $A_\infty$-module
structure;
\item a filtered bulk--boundary Koszul/Verdier $L_\infty$ comparison whose genus-$0$ value
identifies the bulk and boundary Maurer--Cartan elements up to gauge.
\end{enumerate}
Then an all-genus boundary transfer extending the disk-level dictionary is the same datum as a
modular lift
\[
\Pi_\partial(\h)=\pi_\partial+\sum_{g\ge 1}\h^g\Theta_\partial^{(g)}
\]
of $\pi_\partial$.  The coefficient $\Theta_\partial^{(g)}$ is a genus-$g$ loop/$1/N$ correction
inside the chosen package and is not determined by the tree-level transfer alone.  At order $G$
the obstruction is
\[
\Obs_G(\Pi_{<G})\in H^2(\mathfrak g_\partial^{(G)},d_{\pi_\partial});
\]
after applying the bulk--boundary comparison, the discrepancy between bulk and boundary lifts is
measured by the cone classes of Theorem~\ref{thm:relative-bulk-boundary-reconstruction}.
\end{theorem}

\subsection{Order-by-order obstruction theory}
\label{subsec:order-by-order-obstruction-theory}

Higher-genus corrections live as obstruction classes in the $d_\pi$-cohomology of the genus-filtered package.

\begin{definition}[Twisted differential and linearized genus-$0$ complex]
\label{def:twisted-differential}
Given $\pi\in\mathfrak g^{(0),1}$ as in Definition
\ref{def:complete-genus-filtered-package}, define
\begin{equation}
\label{eq:twisted-differential}
d_\pi(x):=\sum_{m\ge 0}\frac1{m!}\,\ell_{m+1}^{(0)}(\pi^{\otimes m},x).
\end{equation}
This is the linearization of the genus-$0$ Maurer--Cartan equation at $\pi$.
\end{definition}

\begin{theorem}[The first modular obstruction]
\label{thm:first-modular-obstruction}
Let $\mathfrak g^{\mathrm{mod}}$ be a complete genus-filtered cyclic deformation package and let
$\pi\in \mathfrak g^{(0),1}$ be its genus-$0$ Maurer--Cartan element. Define the genus-$1$
source term
\begin{equation}
\label{eq:omega-one}
\omega_1(\pi):=\sum_{k\ge 1}\frac1{k!}\,\ell_k^{(1)}(\pi^{\otimes k})
\in \mathfrak g^{(1),2}.
\end{equation}
Then:
\begin{enumerate}[label=\textup{(\alph*)}]
\item $d_\pi^2=0$;
\item $d_\pi\omega_1(\pi)=0$;
\item there is therefore a canonical cohomology class
\[
\Obs_1(\pi):=[\omega_1(\pi)]
\in H^2\bigl(\mathfrak g^{(1)},d_\pi\bigr);
\]
\item a genus-$1$ correction $\Theta^{(1)}\in\mathfrak g^{(1),1}$ exists modulo gauge if and
only if $\Obs_1(\pi)=0$;
\item if such genus-$1$ corrections exist, their gauge-equivalence classes form a torsor for
$H^1(\mathfrak g^{(1)},d_\pi)$.
\end{enumerate}
\end{theorem}

\begin{proof}
The linearization of the Maurer--Cartan equation at a Maurer--Cartan point always squares to
zero; this is the standard $L_\infty$ identity for the twisted unary bracket, proving $d_\pi^2=0$.
Now insert the ansatz
\[
\Pi(\h)=\pi+\h\Theta^{(1)}+O(\h^2)
\]
into \eqref{eq:full-modular-mc}. The coefficient of $\h^0$ is the genus-$0$ Maurer--Cartan
equation for $\pi$, while the coefficient of $\h^1$ is exactly
\[
d_\pi\Theta^{(1)}+\omega_1(\pi)=0.
\]
Applying $d_\pi$ and using $d_\pi^2=0$ gives $d_\pi\omega_1(\pi)=0$, so
$\omega_1(\pi)$ defines a degree-$2$ cohomology class. The displayed equation also shows that
a genus-$1$ lift exists precisely when this cocycle is exact. Finally, if
$\Theta^{(1)}$ and $\widetilde\Theta^{(1)}$ are two solutions then their difference is
$d_\pi$-closed, and a genus-$1$ gauge transformation changes the difference by a
$d_\pi$-exact term. This yields the torsor statement.
\end{proof}

\begin{definition}[Higher source terms]
\label{def:higher-source-terms}
Let
\[
\Pi_{<G}:=\pi+\sum_{1\le h<G}\h^h\Theta^{(h)}
\]
be a partial lift satisfying the Maurer--Cartan equation modulo $\h^G$. The
\emph{$G$-th source term} is the coefficient
\begin{equation}
\label{eq:higher-source-term}
\omega_G(\Pi_{<G})
:=
[\h^G]\,
\sum_{r\ge 0}\sum_{k\ge 1}\frac1{k!}\,
\h^r\ell_k^{(r)}\!\bigl(\Pi_{<G}^{\otimes k}\bigr)
\in \mathfrak g^{(G),2}.
\end{equation}
\end{definition}

\begin{theorem}[All-genus obstruction tower]
\label{thm:all-genus-obstruction-tower}
For every $G\ge 1$, the source term $\omega_G(\Pi_{<G})$ is $d_\pi$-closed. Hence there is a
canonical obstruction class
\[
\Obs_G(\Pi_{<G})
:=
[\omega_G(\Pi_{<G})]
\in H^2\bigl(\mathfrak g^{(G)},d_\pi\bigr).
\]
In detail:
\begin{enumerate}[label=\textup{(\roman*)}]
\item the partial lift $\Pi_{<G}$ extends to order $G$ if and only if
$\Obs_G(\Pi_{<G})=0$;
\item if extensions exist, their gauge-equivalence classes form a torsor for
$H^1(\mathfrak g^{(G)},d_\pi)$;
\item if $H^2(\mathfrak g^{(G)},d_\pi)=0$ for all $G\ge 1$, then $\pi$ admits a full modular
lift;
\item if, in addition, $H^1(\mathfrak g^{(G)},d_\pi)=0$ for all $G\ge 1$, then the modular lift
is unique up to gauge.
\end{enumerate}
\end{theorem}

\begin{proof}
Let
\[
E(\Pi):=\sum_{r\ge 0}\sum_{k\ge 1}\frac1{k!}\,\h^r\ell_k^{(r)}(\Pi^{\otimes k})
\]
be the Maurer--Cartan functional. Since $\Pi_{<G}$ solves the MC equation modulo $\h^G$, one
has
\[
E(\Pi_{<G})=\h^G\omega_G(\Pi_{<G})+O(\h^{G+1}).
\]
For any $x\in\mathfrak g^{(G),1}$,
\[
E(\Pi_{<G}+\h^Gx)=\h^G\bigl(\omega_G(\Pi_{<G})+d_\pi x\bigr)+O(\h^{G+1}),
\]
because the only terms linear in $x$ of total order $G$ arise from genus-$0$ brackets with all
other entries equal to $\pi$. Thus extension to order $G$ is equivalent to solving
\[
d_\pi x+\omega_G(\Pi_{<G})=0,
\]
which proves part \textup{(i)}.

To prove closure, apply the linearized Maurer--Cartan operator at $\Pi_{<G}$ to
$E(\Pi_{<G})$. The $L_\infty$ identities imply the Bianchi-type identity
\[
d_{\Pi_{<G}}E(\Pi_{<G})=0.
\]
Reducing modulo $\h^{G+1}$ and using $d_{\Pi_{<G}}\equiv d_\pi \pmod{\h}$ yields
$d_\pi\omega_G(\Pi_{<G})=0$.

If $x$ and $x'$ are two order-$G$ lifts, then $x-x'$ is $d_\pi$-closed; order-$G$ gauge
transformations shift the difference by a $d_\pi$-exact term. This gives part \textup{(ii)}.
The remaining statements follow by induction on $G$.
\end{proof}

\begin{corollary}[Existence and uniqueness by vanishing]
\label{cor:existence-uniqueness-vanishing}
If
\[
H^2\bigl(\mathfrak g^{(G)},d_\pi\bigr)=0\qquad \forall G\ge 1,
\]
then every genus-$0$ Maurer--Cartan class lifts to all genera. If also
\[
H^1\bigl(\mathfrak g^{(G)},d_\pi\bigr)=0\qquad \forall G\ge 1,
\]
then the lift is unique up to gauge.
\end{corollary}

\subsection{Bulk--boundary comparison as a cone problem}
\label{subsec:bulk-boundary-cone-problem}

The cone of the linearized bulk--boundary map controls the relative lifting problem.

\begin{definition}[Linearized bulk--boundary map]
\label{def:linearized-bulk-boundary-map}
Let
\[
F:\mathfrak g^{\mathrm{mod}}_{\mathrm{bulk}}\longrightarrow \mathfrak g^{\mathrm{mod}}_{\partial}
\]
be a filtered $L_\infty$-morphism, and let
$\pi_{\mathrm{bulk}}\in\mathfrak g_{\mathrm{bulk}}^{(0),1}$,
$\pi_\partial\in\mathfrak g_\partial^{(0),1}$
be genus-$0$ Maurer--Cartan elements such that $F_*(\pi_{\mathrm{bulk}})$ is gauge-equivalent to
$\pi_\partial$. The \emph{linearized bulk--boundary map} is the induced cochain map
\[
F_\pi:(\mathfrak g^{\mathrm{mod}}_{\mathrm{bulk}},d_{\pi_{\mathrm{bulk}}})
\longrightarrow
(\mathfrak g^{\mathrm{mod}}_{\partial},d_{\pi_\partial}).
\]
\end{definition}

\begin{theorem}[Relative bulk--boundary reconstruction]
\label{thm:relative-bulk-boundary-reconstruction}
Assume the hypotheses of Definition \ref{def:linearized-bulk-boundary-map}. If
\[
H^1\bigl(\Cone(F_\pi)\bigr)=H^2\bigl(\Cone(F_\pi)\bigr)=0,
\]
then the genus-by-genus modular lift problem for $\pi_{\mathrm{bulk}}$ is equivalent, up to gauge,
to the genus-by-genus modular lift problem for $\pi_\partial$. In particular:
\begin{enumerate}[label=\textup{(\roman*)}]
\item every boundary modular lift uniquely determines a compatible bulk modular lift;
\item every bulk modular lift uniquely determines a compatible boundary modular lift;
\item all extension obstructions and ambiguities are measured by the same cone cohomology.
\end{enumerate}
\end{theorem}

\begin{proof}
Compatible pairs of bulk and boundary lifts form the Maurer--Cartan space of the homotopy
fiber of $F$ over $\pi_\partial$. The tangent complex of that homotopy fiber at the point
$(\pi_{\mathrm{bulk}},\pi_\partial)$ is $\Cone(F_\pi)[-1]$. Standard obstruction theory for formal
moduli problems therefore places the extension obstruction in
$H^2(\Cone(F_\pi))$ and the ambiguity in $H^1(\Cone(F_\pi))$. When both groups vanish,
compatible lifts exist and are unique at each order in $\h$, hence by induction through the
entire genus filtration.
\end{proof}

\begin{corollary}[Boundary determination of the modular bulk]
\label{cor:boundary-determines-bulk}
If the linearized bulk--boundary map is a quasi-isomorphism in degrees $1$ and $2$, then the
full modular bulk deformation problem is controlled by the boundary algebra. Equivalently,
under this hypothesis the holographic dictionary is reconstructible from the boundary modular
lift alone.
\end{corollary}

\subsection{The boundary modular class}
\label{subsec:boundary-modular-class}

The genus-$1$ source term has a canonical graph-theoretic form: the one-wheel sum on transferred tree-level vertices. Its cohomology class is the first modular invariant of the boundary algebra.

\begin{definition}[Loop-generated modular model]
\label{def:loop-generated-model}
A modular deformation package is \emph{loop-generated} if its first positive-genus operations
are generated entirely by connected stable graphs of first Betti number $1$ whose vertices are
decorated by genus-$0$ operations. Equivalently, there are no primitive genus-$1$ vertices:
all genus-$1$ information is produced by a single loop with tree-level vertex decorations.
\end{definition}

\begin{definition}[One-wheel sum and boundary modular class]
\label{def:one-wheel-sum-and-class}
Let $(A,m_\bullet)$ be a cyclic minimal $A_\infty$-algebra, and suppose its modular
deformation package is loop-generated. Let $\mathsf{Wheel}_1$ denote the set of connected
stable graphs of first Betti number $1$, modulo isomorphism. For
$\Gamma\in\mathsf{Wheel}_1$, let $B_\Gamma(m_\bullet)$ be the graph contraction obtained by
placing the transferred vertices $m_n$ at the vertices of $\Gamma$ and contracting internal edges
using the cyclic pairing. Define the completed one-wheel sum
\begin{equation}
\label{eq:one-wheel-sum}
W_1(A):=
\sum_{\Gamma\in\mathsf{Wheel}_1}^{\mathrm{comp}}
\frac{w_\Gamma}{|\Aut(\Gamma)|}
B_\Gamma(m_\bullet)
\in \mathfrak g_A^{(1),2},
\end{equation}
where $w_\Gamma$ denotes the regularized weight of the wheel type. The cohomology class
\begin{equation}
\label{eq:boundary-modular-class}
\mathfrak W(A):=[W_1(A)]\in H^2(\mathfrak g_A^{(1)},d_{\pi_A})
\end{equation}
is called the \emph{boundary modular class} of $A$.
\end{definition}

\begin{theorem}[One-wheel reduction]
\label{thm:one-wheel-reduction}
Assume the modular deformation package of $(A,m_\bullet)$ is loop-generated. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $W_1(A)$ is $d_{\pi_A}$-closed;
\item its cohomology class agrees with the first obstruction class,
\[
\mathfrak W(A)=\Obs_1(A,\pi_A);
\]
\item $\mathfrak W(A)$ depends only on the cyclic $A_\infty$ quasi-isomorphism class of $A$.
\end{enumerate}
\end{theorem}

\begin{proof}
In a loop-generated model, the coefficient of $\h$ in the Maurer--Cartan functional is a sum
over connected stable graphs of first Betti number $1$. Each such graph has a unique loop.
Collapse every maximal tree attached to this loop by cyclic homological perturbation. Each
collapsed tree becomes a single effective vertex labeled by one of the transferred operations
$m_n$. The resulting expression is exactly the completed one-wheel sum \eqref{eq:one-wheel-sum}.
Therefore the order-$\h$ Maurer--Cartan equation reads
\[
d_{\pi_A}\Theta_A^{(1)}+W_1(A)=0,
\]
so $W_1(A)$ is closed and its cohomology class is the obstruction class.

Finally, cyclic $A_\infty$ quasi-isomorphisms induce quasi-isomorphisms of the corresponding
cyclic deformation complexes; the obstruction class of the Maurer--Cartan lifting problem is
therefore invariant under cyclic $A_\infty$ quasi-isomorphism.
\end{proof}

\begin{corollary}[Vanishing criterion and additivity]
\label{cor:vanishing-criterion-and-additivity}
\begin{enumerate}[label=\textup{(\roman*)}]
\item A genus-$1$ modular lift exists if and only if $\mathfrak W(A)=0$.
\item If also $H^1(\mathfrak g_A^{(1)},d_{\pi_A})=0$, then the genus-$1$ lift is unique up to
gauge.
\item If $A=A_1\oplus A_2$ is an orthogonal direct sum of cyclic minimal $A_\infty$-algebras,
then
\[
\mathfrak W(A)=\mathfrak W(A_1)\oplus\mathfrak W(A_2).
\]
\end{enumerate}
\end{corollary}

\begin{proof}
Parts \textup{(i)} and \textup{(ii)} are immediate from Theorems
\ref{thm:first-modular-obstruction} and \ref{thm:one-wheel-reduction}. For
\textup{(iii)}, the cyclic pairing and all transferred products split block-diagonally, so every
one-wheel contraction is supported on exactly one summand.
\end{proof}

\subsection{The holomorphic BF model on $S^3$}
\label{subsec:holomorphic-bf-model-on-s3}

Specialise to Zeng's holomorphic BF model on $S^3$ \cite{ZengTwistedHolography}.

\begin{definition}[The CR-cohomological boundary algebra]
\label{def:cr-cohomological-boundary-algebra}
Let
\[
A:=\Hb = H_b^{0,0}(S^3)\oplus H_b^{0,1}(S^3).
\]
By the harmonic decomposition of the tangential Cauchy--Riemann complex and the computation
of CR cohomology, one has
\[
H_b^{0,0}(S^3)\cong \C[w_1,w_2],
\qquad
H_b^{0,1}(S^3)\cong \C[\ov w_1,\ov w_2]\,\epsilon,
\]
where $\epsilon$ is the $SU(2)$-invariant generator of $\Omega_b^{0,1}(S^3)$
\cite{ZengTwistedHolography}. Set
\[
A^0:=\C[w_1,w_2],
\qquad
A^1:=\C[\ov w_1,\ov w_2]\,\epsilon.
\]
\end{definition}

\begin{definition}[Monomial bases and trace pairing]
\label{def:monomial-bases-and-trace-pairing}
For $p,q,r,s\ge 0$, define
\[
a_{p,q}:=w_1^p w_2^q\in A^0,
\qquad
b_{r,s}:=\frac{(r+s+1)!}{r!\,s!}\,\ov w_1^r\ov w_2^s\,\epsilon\in A^1.
\]
The cyclic trace pairing is normalized by
\[
\Tr(a_{p,q}b_{r,s})=\delta_{pr}\delta_{qs}.
\]
\end{definition}

\begin{remark}[The cyclic SDR]
\label{rem:cyclic-sdr}
Zeng constructs a special deformation retract
\[
h\,(\Omega_b^{0,\bullet}(S^3),\delbar_b)
\underset{i}{\overset{p}{\rightleftarrows}}
(\HCR,0)
\]
with $h\circ i=0$, $p\circ h=0$, $h^2=0$, and $\id-ip=\delbar_b h+h\delbar_b$.
The retract is compatible with the cyclic structure, and therefore transfers the
commutative dg algebra structure on $\Omega_b^{0,\bullet}(S^3)$ to a cyclic $A_\infty$ (indeed
$C_\infty$) structure $m_\bullet=\{m_n\}_{n\ge 2}$ on $A$ \cite{ZengTwistedHolography}.
This is the transferred boundary algebra governing the KK interactions.
\end{remark}

\subsection{One-vertex calculus}
\label{subsec:concrete-one-vertex-calculus}

Cyclic traces of the transferred operations carry the first modular data; their one-vertex sectors satisfy rigid arithmetic selection rules.

\begin{proposition}[The $m_2$ shift formula]
\label{prop:m2-shift-formula}
For all $p,q,r,s\ge 0$,
\[
m_2(a_{p,q},b_{r,s})=
\begin{cases}
 b_{r-p,s-q},& p\le r,\ q\le s,\\[4pt]
 0,& \text{otherwise.}
\end{cases}
\]
\end{proposition}

\begin{proof}
Zeng's explicit formula for the transferred binary product is
\[
m_2\!\left(w_1^p w_2^q,
\frac{(r+s+1)!}{r!s!}\,\ov w_1^r \ov w_2^s\epsilon\right)
=
\frac{(r+s-p-q+1)!}{(r-p)!(s-q)!}\,\ov w_1^{r-p}\ov w_2^{s-q}\epsilon.
\]
Recognizing the right-hand side as $b_{r-p,s-q}$ proves the claim.
\end{proof}

\begin{definition}[One-vertex traced sectors]
\label{def:one-vertex-traced-sectors}
For $n\ge 2$ and $\beta_1,\dots,\beta_{n-1}\in A^1$, define the traced one-vertex sector
\begin{equation}
\label{eq:one-vertex-traced-sector}
Q_n(\beta_1,\dots,\beta_{n-1})
:=
\sum_{p,q\ge 0}
\Tr\!\bigl(a_{p,q}\,m_n(a_{p,q},\beta_1,\dots,\beta_{n-1})\bigr).
\end{equation}
For $n=2$, write
\[
\theta_{r,s}:=Q_2(b_{r,s}).
\]
\end{definition}

\begin{theorem}[The tadpole projector]
\label{thm:tadpole-projector}
The coefficients $\theta_{r,s}$ satisfy
\[
\theta_{r,s}=
\begin{cases}
1,& r,s\ \text{both even},\\
0,& \text{otherwise}.
\end{cases}
\]
Equivalently,
\[
W^{(2)}_{\mathrm{alg}}:=\sum_{r,s\ge 0}\theta_{r,s}b_{r,s}
=\sum_{m,n\ge 0} b_{2m,2n}.
\]
In particular, the quadratic one-vertex wheel is the projector onto the even--even KK lattice.
\end{theorem}

\begin{proof}
By Proposition \ref{prop:m2-shift-formula},
\[
\theta_{r,s}
=
\sum_{p,q\ge 0}\Tr\bigl(a_{p,q} b_{r-p,s-q}\bigr)
=
\sum_{p,q\ge 0}\delta_{p,r-p}\delta_{q,s-q}.
\]
This has a solution if and only if $2p=r$ and $2q=s$, i.e. if and only if $r$ and $s$ are both
even. When a solution exists it is unique, so the sum equals $1$.
\end{proof}

\begin{corollary}[Lowest-mode tadpole]
\label{cor:lowest-mode-tadpole}
Let
\[
L:=\Span\{b_{0,0},b_{1,0},b_{0,1}\}\subset A^1.
\]
Then the tadpole projector restricts to
\[
W^{(2)}_{\mathrm{alg}}\big|_L=b_{0,0}.
\]
If one restores the geometric genus-$1$ weight $w_2$, the physical tadpole truncation on the
lowest modes is $w_2\,b_{0,0}$.
\end{corollary}

\begin{theorem}[One-vertex trace selection rule]
\label{thm:one-vertex-trace-selection-rule}
Let $n\ge 2$, and let $\beta_i=b_{r_i,s_i}\in A^1$ for $1\le i\le n-1$. Set
\[
R:=\sum_{i=1}^{n-1} r_i,
\qquad
S:=\sum_{i=1}^{n-1} s_i.
\]
Then for every $p,q\ge 0$ one has
\[
m_n(a_{p,q},b_{r_1,s_1},\dots,b_{r_{n-1},s_{n-1}})
\in
\C\cdot b_{R-p+n-2,\,S-q+n-2}.
\]
Consequently,
\[
Q_n(b_{r_1,s_1},\dots,b_{r_{n-1},s_{n-1}})\neq 0
\]
can occur only if
\begin{equation}
\label{eq:selection-rule}
2p=R+n-2,
\qquad
2q=S+n-2.
\end{equation}
In particular, for fixed external labels there is at most one internal KK mode contributing to
the one-vertex trace.
\end{theorem}

\begin{proof}
Zeng's explicit formula for $m_n$ in the orthonormal $SU(2)$ basis shows that the output of
\[
m_n\bigl(e_{m_0}^{(j_0)},\ov e_{m_1}^{(\ov j_1)}\epsilon,\dots,
\ov e_{m_{n-1}}^{(\ov j_{n-1})}\epsilon\bigr)
\]
lies in the unique representation
\[
\ov e_{M_{n-1}}^{(\ov J_{n-1}-j_0+n-2)}\epsilon,
\qquad
\ov J_{n-1}=\sum_{i=1}^{n-1}\ov j_i,
\qquad
M_{n-1}=m_0+\sum_{i=1}^{n-1}m_i.
\]
Translate this to monomial labels by
\[
j_0=\frac{p+q}{2},\qquad m_0=\frac{p-q}{2},
\qquad
\ov j_i=\frac{r_i+s_i}{2},\qquad m_i=\frac{s_i-r_i}{2}.
\]
Writing the output as $b_{u,v}$ means
\[
\frac{u+v}{2}=\ov J_{n-1}-j_0+n-2,
\qquad
\frac{v-u}{2}=M_{n-1}.
\]
Solving for $(u,v)$ gives
\[
u=R-p+n-2,
\qquad
v=S-q+n-2.
\]
Tracing against $a_{p,q}$ forces $(u,v)=(p,q)$, which is exactly
\eqref{eq:selection-rule}. Therefore there is at most one internal mode.
\end{proof}

\begin{theorem}[Low-mode parity law]
\label{thm:low-mode-parity-law}
Restrict the one-vertex sectors to the lowest-mode space
\[
L=\Span\{b_{0,0},b_{1,0},b_{0,1}\}.
\]
For a tensor in $L^{\otimes(n-1)}$, let $N_{10}$ denote the number of $b_{1,0}$ inputs and
$N_{01}$ the number of $b_{0,1}$ inputs. Then
\[
Q_n\big|_{L^{\otimes(n-1)}}\neq 0
\qquad\Longrightarrow\qquad
N_{10}\equiv N_{01}\equiv n\pmod 2.
\]
\end{theorem}

\begin{proof}
On $L$, one has $R=N_{10}$ and $S=N_{01}$. By Theorem
\ref{thm:one-vertex-trace-selection-rule}, nonvanishing requires
$R+n-2$ and $S+n-2$ to be even. Since $n-2\equiv n\pmod 2$, the statement follows.
\end{proof}

\begin{corollary}[Cubic support on the lowest modes]
\label{cor:cubic-support-lowest-modes}
For $n=3$, the only nonzero low-mode one-vertex channels are the mixed ordered pairs
\[
(b_{1,0},b_{0,1})
\qquad\text{and}\qquad
(b_{0,1},b_{1,0}).
\]
No ordered pair involving $b_{0,0}$ contributes to $Q_3|_{L^{\otimes 2}}$.
\end{corollary}

\begin{proof}
For $n=3$, Theorem \ref{thm:low-mode-parity-law} requires
$N_{10}\equiv N_{01}\equiv 1\pmod 2$. Since there are only two inputs, the only possibility is
one copy of $b_{1,0}$ and one copy of $b_{0,1}$.
\end{proof}

\begin{proposition}[Vanishing of the symmetric cubic one-vertex sector]
\label{prop:vanishing-symmetric-cubic-one-vertex-sector}
The symmetric cubic one-vertex sector vanishes on the lowest modes. Equivalently, the
trilinear form $Q_3|_{L^{\otimes 2}}$ factors through an alternating $2$-form
\[
\omega_{\mathrm{pm}}\in \Lambda^2 L^*,
\]
the \emph{low-mode pre-modular two-form}.
\end{proposition}

\begin{proof}
By Corollary \ref{cor:cubic-support-lowest-modes}, the only nonzero channels are the two mixed
ordered pairs. Zeng's antisymmetry relation for the cubic coefficients shows that swapping the
two $A^1$ inputs reverses the coefficient. Therefore the symmetric combination cancels, and the
result factors through $\Lambda^2L^*$.
\end{proof}

\begin{theorem}[Quartic support theorem on the lowest modes]
\label{thm:quartic-support-theorem-lowest-modes}
For $n=4$, the one-vertex low-mode sector is supported on exactly three unordered channels:
\[
(b_{0,0},b_{0,0},b_{0,0}),
\qquad
(b_{1,0},b_{1,0},b_{0,0}),
\qquad
(b_{0,1},b_{0,1},b_{0,0}).
\]
Equivalently, on $\Sym^3(L)$ one has a unique decomposition
\begin{equation}
\label{eq:q4-normal-form}
Q_4\big|_{\Sym^3(L)}
=
Q_{000}\,b_{0,0}^3
+
Q_{110}\,b_{1,0}^2b_{0,0}
+
Q_{011}\,b_{0,1}^2b_{0,0},
\end{equation}
for uniquely defined scalars $Q_{000},Q_{110},Q_{011}\in\C$.
\end{theorem}

\begin{proof}
For $n=4$, Theorem \ref{thm:low-mode-parity-law} gives
$N_{10}\equiv N_{01}\equiv 0\pmod 2$. Since there are three inputs,
\[
N_{10}+N_{01}+N_{00}=3,
\]
so the only possible triples are
\[
(N_{10},N_{01},N_{00})=(0,0,3),\ (2,0,1),\ (0,2,1).
\]
These are exactly the three unordered channels above.
\end{proof}

\begin{corollary}[The charged quartic sector is two-dimensional]
\label{cor:charged-quartic-sector-two-dimensional}
After removing the neutral channel $b_{0,0}^3$, the first charged one-vertex quartic sector on
low modes lies in the $2$-dimensional space
\[
\Span\{b_{1,0}^2b_{0,0},\ b_{0,1}^2b_{0,0}\}.
\]
Thus the first charged quartic modular datum is encoded by the pair $(Q_{110},Q_{011})$.
\end{corollary}

\begin{remark}[Tiny internal support]
\label{rem:tiny-internal-support}
The trace selection rule determines the internal test mode uniquely. For the three quartic
channels of Theorem \ref{thm:quartic-support-theorem-lowest-modes}, one finds
\[
Q_{000}: \ a_{1,1},
\qquad
Q_{110}: \ a_{2,1},
\qquad
Q_{011}: \ a_{1,2}.
\]
The charged quartic coefficients are therefore algebraically finite, without regularization. By contrast, generic loop sectors of the KK theory require infinite KK sums \cite{ZengTwistedHolography}.
\end{remark}

\subsection{The first two-vertex bubble and the exceptional finite locus}
\label{subsec:first-two-vertex-bubble-exceptional-finite-locus}

The first loop-shaped graph is the two-vertex cubic--cubic wheel. On the lowest-mode sector it remains a finite algebraic contraction.

\begin{theorem}[Support of the first mixed cubic--cubic bubble]
\label{thm:support-first-mixed-cubic-cubic-bubble}
Consider the first two-vertex wheel built from two cubic vertices. Restricted to the lowest-mode
subspace $L$, its symmetric external content is supported on a single monomial:
\[
b_{1,0}^2 b_{0,1}^2.
\]
No monomial involving $b_{0,0}$ occurs in this first cubic--cubic bubble sector.
\end{theorem}

\begin{proof}
By Corollary \ref{cor:cubic-support-lowest-modes}, each cubic vertex is nonzero only if its two
$A^1$ inputs are one copy of $b_{1,0}$ and one copy of $b_{0,1}$. Therefore every nonzero
cubic--cubic wheel contributes exactly one $b_{1,0}$ and one $b_{0,1}$ from each vertex. The
total external monomial is thus forced to be $b_{1,0}^2b_{0,1}^2$.
\end{proof}

\begin{theorem}[Exceptional finiteness of the first mixed bubble]
\label{thm:exceptional-finiteness-first-mixed-bubble}
The first mixed cubic--cubic bubble on the lowest modes is finite before regularization. More
precisely, for the mixed cubic sector one has
\[
m_3(a_{p,q},b_{1,0},b_{0,1})\in \C\cdot b_{2-p,2-q},
\]
and therefore only the finite range
\[
0\le p,q\le 2
\]
can contribute to the first mixed bubble coefficient. In particular, the first mixed bubble is a
finite algebraic contraction of finitely many cubic structure constants.
\end{theorem}

\begin{proof}
Apply Theorem \ref{thm:one-vertex-trace-selection-rule} with $n=3$ and external labels
$(r_1,s_1)=(1,0)$, $(r_2,s_2)=(0,1)$. Then $R=S=1$, so the cubic output is
\[
b_{R-p+1,S-q+1}=b_{2-p,2-q}.
\]
For this to make sense in $A^1$, one must have $2-p\ge 0$ and $2-q\ge 0$, hence
$0\le p,q\le 2$. Thus the internal mode sum is finite.
\end{proof}

\begin{definition}[The low-mode charged modular jet]
\label{def:low-mode-charged-modular-jet}
The \emph{low-mode charged modular jet} of the holomorphic BF boundary algebra is the triple
\begin{equation}
\label{eq:charged-modular-jet}
\mathfrak J_{\mathrm{ch}}^{\mathrm{low}}:=(Q_{110},Q_{011},B_{\mathrm{mix}}),
\end{equation}
where $Q_{110}$ and $Q_{011}$ are the charged one-vertex quartic coefficients of
\eqref{eq:q4-normal-form}, and $B_{\mathrm{mix}}$ is the coefficient of
$b_{1,0}^2b_{0,1}^2$ in the first cubic--cubic bubble sector.
\end{definition}

\begin{theorem}[Low-mode modular normal form]
\label{thm:low-mode-modular-normal-form}
Up to and including the first cubic--cubic bubble order, the symmetric modular shadow of the
holomorphic BF boundary algebra on the lowest-mode sector has the form
\begin{equation}
\label{eq:low-mode-modular-normal-form}
W_{\mathrm{low}}
=
w_2\,b_{0,0}
+
Q_{000}\,b_{0,0}^3
+
Q_{110}\,b_{1,0}^2b_{0,0}
+
Q_{011}\,b_{0,1}^2b_{0,0}
+
B_{\mathrm{mix}}\,b_{1,0}^2b_{0,1}^2
+
\textup{(higher sectors)}.
\end{equation}
No other symmetric low-mode monomial can appear at this order.
\end{theorem}

\begin{proof}
The first term is the tadpole contribution from Corollary
\ref{cor:lowest-mode-tadpole}. The one-vertex cubic symmetric contribution vanishes by
Proposition \ref{prop:vanishing-symmetric-cubic-one-vertex-sector}. The one-vertex quartic
contribution is exhausted by Theorem \ref{thm:quartic-support-theorem-lowest-modes}. The
first two-vertex bubble contribution is exhausted by Theorem
\ref{thm:support-first-mixed-cubic-cubic-bubble}. No further symmetric low-mode monomial can
occur before higher-valence or higher-vertex sectors enter.
\end{proof}

\begin{remark}[Finite charged reduction]
\label{rem:what-has-been-proved}
Theorem \ref{thm:low-mode-modular-normal-form} reduces the first charged low-mode modular
calculation to the three finite contractions
$Q_{110},Q_{011},B_{\mathrm{mix}}$. These are the only charged low-mode symmetric modular
coefficients at the tadpole, one-vertex quartic, and first cubic--cubic bubble orders. The
infinite KK/modular calculation is therefore replaced, at this order, by the finite charged jet
\(\mathfrak J_{\mathrm{ch}}^{\mathrm{low}}\).
\end{remark}

\section{Celestial OPE from the bar complex: gluon and graviton sectors}
\label{sec:celestial-ope-bar-complex}

The tree-level celestial transform sends $4$-dimensional scattering amplitudes to
$2$-dimensional celestial CFT correlators by Mellin transform. The boundary chiral algebras
of the holomorphic-topological framework (affine Kac--Moody for gauge theory, Virasoro for
gravity) encode internal collision-residue data through their bar complexes. These residues
are not, by themselves, the physical celestial OPE coefficients. The physical coefficients
also contain the Mellin convolution of the collinear splitting functions, and this is
separate comparison data. The $d\log$ absorption mechanism still gives the internal bar
pole bookkeeping: the bar kernel $d\log(z_1 - z_2)$ absorbs one pole order, so a local
singularity $\sim z^{-N}$ produces an $r$-matrix pole $\sim z^{-(N-1)}$. Loop and quantum
corrections require the genus-filtered comparison maps of
Proposition~\ref{prop:mellin-shadow-dictionary}.

\subsection{Gluon celestial OPE from $\barBch(\widehat{\fg}_k)$}
\label{subsec:ch-core-gluon-celestial-ope}

The holomorphic-topological twist of $4$d $\mathcal{N} = 2$ Yang--Mills with gauge algebra $\fg$ on $\mathbb{C} \times \mathbb{R}_{\ge 0}$ produces a boundary Kac--Moody algebra $\widehat{\fg}_k$. The positive-helicity gluon maps to the affine current $J^a(z)$. The $\lambda$-bracket $\{J^a {}_\lambda J^b\} = f^{ab}{}_c\, J^c + k\,\kappa^{ab}\,\lambda$ Laplace-transforms to $r^{\mathrm{Lap}}(z) = f^{ab}{}_c\, J^c / z + k\,\kappa^{ab}/z^2$. After $d\log$ absorption, the $z^{-1}$ structure-constant term drops to $z^0$ and the $z^{-2}$ level term becomes $z^{-1}$:
\[
r^{\mathrm{KM}}(z) \;=\; \frac{k\,\Omega}{z}
\;=\; \frac{k\,\kappa^{ab}\,t_a \otimes t_b}{z},
\]
where $\Omega$ is the Casimir tensor.

\begin{theorem}[Tree-level gluon: level residue versus celestial OPE coefficient]
\label{thm:ch-core-celestial-gluon-ope}
Let $\mathcal{O}^{+,a}_{\Delta}(z)$ denote the celestial primary of a positive-helicity gluon
of colour~$a$ and conformal dimension~$\Delta$. The affine bar collision residue after
$d\log$ absorption is the level channel
\[
r^{\mathrm{KM}}_{\mathrm{lev}}(z)=\frac{k\,\kappa^{ab}\,t_a\otimes t_b}{z}.
\]
This tensor is symmetric in the two colour labels. It is not naturally identified with the
antisymmetric structure constant \(f^{ab}{}_{c}\), and no colour-linear normalization of
\(k\Omega\) produces the tree-level gluon OPE coefficient. The physical tree celestial OPE
has the separate Mellin-splitting coefficient. Suppressing the conventional overall
Yang--Mills coupling and phase, its leading singular part is
\[
\mathcal{O}^{+,a}_{\Delta_1}(z_1)\,\mathcal{O}^{+,b}_{\Delta_2}(z_2)
\;\sim\;
\frac{B(\Delta_1-1,\Delta_2-1)f^{ab}{}_c}{z_1 - z_2}\;
\mathcal{O}^{+,c}_{\Delta_1 + \Delta_2 - 1}(z_2)
\;+\; \textup{regular},
\]
as in Fan--Pasterski--Shao and Pate--Raclariu--Strominger--Yuan
\cite{FanPasterskiShaoCelOPE,PateRaclariuStromingerYuan19}; in their convention the
suppressed prefactor is the usual \(-ig_{\mathrm{YM}}\). Thus the bar level residue and the
physical celestial OPE coefficient are distinct pieces of the comparison problem. The bar
residue controls the pole filtration and level channel; the physical coefficient requires
the Mellin transform of the collinear splitting function.
\end{theorem}

\begin{proof}
The affine $\lambda$-bracket has two colour structures. The structure-constant term
\(f^{ab}{}_{c}J^c/z\) is antisymmetric and drops to degree \(z^0\) after $d\log$
absorption; the level term \(k\kappa^{ab}/z^2\) is symmetric and becomes the simple-pole bar
residue \(k\kappa^{ab}/z\). Symmetry already forbids a natural colour-linear identification
of \(k\kappa^{ab}\) with \(f^{ab}{}_{c}\). The physical celestial OPE is obtained instead by
Mellin-transforming the collinear gluon splitting amplitude; this convolution gives
\(B(\Delta_1-1,\Delta_2-1)\), while the colour factor remains \(f^{ab}{}_{c}\). The only
comparison visible from the bar residue alone is the pole-filtration/dimension-shift datum,
not the physical OPE coefficient. Positive genus corrections are separate classes
\(\Obs^{\mathrm{Mell}}_{g,1}\in H^2(\Cone(\mu_{g,1}))\).
\end{proof}

\begin{remark}[Comparison with Fan--Pasterski--Shao and PRSY]
\label{rem:ch-core-fft}
The celestial gluon OPE has a beta-function coefficient and an antisymmetric colour factor.
The affine level residue is the symmetric Casimir channel \(k\kappa^{ab}\). The two data
meet only after a comparison map that contains the Mellin splitting calculation; the bar
complex alone does not extract \(f^{ab}{}_{c}\) from \(k\Omega\).
\end{remark}


\subsection{Graviton celestial OPE from $\barBch(\mathrm{Vir}_c)$}
\label{subsec:ch-core-graviton-celestial-ope}

The holomorphic-topological twist of $3$d gravity on $\mathbb{C} \times \mathbb{R}_{\ge 0}$ produces a boundary Virasoro algebra $\mathrm{Vir}_c$. The positive-helicity graviton maps to the stress tensor~$T(z)$. The Virasoro $\lambda$-bracket $\{T{}_\lambda T\} = (\partial + 2\lambda)\,T + \frac{c}{12}\,\lambda^3$ Laplace-transforms to $r^{\mathrm{Lap}}(z) = \partial T/z + 2T/z^2 + (c/2)/z^4$. After $d\log$ absorption, the $\partial T$ term drops, producing the two-pole collision $r$-matrix:
\[
r^{\mathrm{Vir}}(z) \;=\; \frac{c/2}{z^3} \;+\; \frac{2T}{z}.
\]

\begin{theorem}[Tree-level graviton: Virasoro residue versus celestial OPE]
\label{thm:ch-core-celestial-graviton-ope}
For \(\barBch(\mathrm{Vir}_c)\), the internal Virasoro bar collision residue after $d\log$
absorption is
\[
r^{\mathrm{Vir}}(z)=\frac{c/2}{z^3}+\frac{2T}{z}.
\]
This is an internal chiral-algebra residue. It is not the physical tree-level celestial
graviton OPE. Suppressing the conventional gravitational coupling, the positive-helicity
tree celestial graviton OPE has leading singular part
\[
G^+_{\Delta_1}(z_1,\bar z_1)\,G^+_{\Delta_2}(z_2,\bar z_2)
\;\sim\;
\frac{\bar z_{12}}{z_{12}}\,
B(\Delta_1-1,\Delta_2-1)\,
G^+_{\Delta_1+\Delta_2}(z_2,\bar z_2)
\;+\; \textup{regular}.
\]
In the PRSY normalization this displayed term is multiplied by
\(-\kappa_{\mathrm{grav}}/2\) \cite{PateRaclariuStromingerYuan19}. No tree-level physical
celestial OPE contains the Virasoro central triple-pole term \((c/2)z_{12}^{-3}\) or an output dimension
\(\Delta_1+\Delta_2-3\). Such terms are internal Virasoro collision residues; turning them
into physical celestial OPE data requires extra anomaly and comparison input not present in
the tree PRSY OPE.
\end{theorem}

\begin{proof}
The Virasoro $\lambda$-bracket
\(\{T{}_\lambda T\}=(\partial+2\lambda)T+\frac{c}{12}\lambda^3\) Laplace-transforms to
\(\partial T/z+2T/z^2+(c/2)/z^4\). After $d\log$ absorption, the \(\partial T\) term drops,
the \(2T\) term becomes a simple pole, and the central term becomes a triple pole. This
proves the displayed internal residue. The physical tree graviton OPE is obtained by
Mellin-transforming the gravitational collinear splitting function, and the PRSY formula
has the antiholomorphic ratio \(\bar z_{12}/z_{12}\), the beta function
\(B(\Delta_1-1,\Delta_2-1)\), and output dimension \(\Delta_1+\Delta_2\). These data are not
produced by the Virasoro central residue. Positive genus corrections are controlled by the classes
\(\Obs^{\mathrm{Mell}}_{g,r}\in H^2(\Cone(\mu_{g,r}))\).
\end{proof}


\subsection{Helicity-chirality}
\label{subsec:ch-core-helicity-chirality}

The bar complex carries the holomorphic helicity sector. The anti-holomorphic sector enters through a Verdier/Koszul comparison datum.

\begin{theorem}[Helicity splitting via the Swiss-cheese decomposition]
\label{thm:ch-core-helicity-splitting}
Let $\mathcal A$ be the boundary chiral algebra of an HT theory. Assume a Verdier-helicity
comparison
\[
\nu\colon \barBch(\mathcal A^!)\xrightarrow{\;\sim\;}
\barBch(\mathcal A)^{\mathrm{anti\text{-}hol}}
\]
identifying the Koszul dual collision complex with the anti-holomorphic helicity sector, and
assume a tree-level Mellin propagator comparison whose two-negative-helicity component is the
Parke--Taylor pairing on ordered configurations. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item The bar complex $\barBch(\mathcal{A})$ is built from logarithmic forms on $\overline{\mathrm{FM}}_n(\mathbb{C})$, which are holomorphic. It encodes exactly the self-dual (positive-helicity) sector.
\item The Koszul dual bar complex $\barBch(\mathcal{A}^!)$ encodes the anti-self-dual (negative-helicity) sector: Koszul duality $\mathcal{A} \mapsto \mathcal{A}^!$ exchanges holomorphic and anti-holomorphic collision data.
\item Mixed-helicity amplitudes arise from the pairing between
$\barBch(\mathcal{A})$ and $\barBch(\mathcal{A}^!)$ mediated by the
universal twisting morphism
$\tau_{\mathcal A}\colon \barBch(\mathcal{A}) \to \mathcal{A}$
and the Verdier pairing
$\mathcal{A}\otimes\mathcal{A}^!\to\omega_{\operatorname{Ran}}$.
Physically, this pairing is the bulk propagator
$K(z,\bar z,t) \sim \Theta(t)/(2\pi z)$ connecting holomorphic and
anti-holomorphic insertions.
\item All-positive and single-negative tree amplitudes vanish: the weight form $\omega_n^{(+,\ldots,+)}$ has the wrong form-type to pair nontrivially with the volume form on $\overline{\mathrm{FM}}_n(\mathbb{C})$.
\item MHV amplitudes (two negative helicities) are the simplest non-vanishing pairing between $\barBch(\mathcal{A})$ and $\barBch(\mathcal{A}^!)$: the Parke--Taylor formula encodes the structure of this minimal pairing.
\end{enumerate}
In the two-colour language: the holomorphic direction ($\mathbb{C}$) encodes positive helicity via the bar differential; the topological direction ($\mathbb{R}$) encodes negative helicity via the Koszul dual. Mixed-helicity amplitudes are bulk--boundary couplings in the chiral derived centre pair $(C^\bullet_{\mathrm{ch}}(\mathcal{A},\mathcal{A}),\, \mathcal{A})$. The directional restriction of $\mathrm{SC}^{\mathrm{ch,top}}$ is the statement that the open-colour positive-helicity sector receives action from the closed-colour bulk but no opposite arrow.

Helicity is chirality; helicity splitting is the two-colour decomposition of the $E_1$ bar coalgebra.
\end{theorem}

\begin{proof}
The bar differential $d_{\barBch} = d_Q + d_{\mathrm{res}} + d_{A_\infty}$ is built from collision residues of the holomorphic OPE on $\overline{\mathrm{FM}}_n(\mathbb{C})$. The logarithmic forms $\Omega^\bullet_{\log}(\overline{\mathrm{FM}}_n(\mathbb{C}))$ are generated by $d\log(z_i - z_j)$ and carry no $d\bar z$ factors: they span a pure Hodge type~$(p,0)$ within the mixed Hodge structure on $H^\bullet(\overline{\mathrm{FM}}_n(\mathbb{C}))$. Every element of $\barBch(\mathcal{A})$ is therefore a purely holomorphic object: it sees exactly the self-dual sector and nothing else.
The comparison $\nu$ identifies the Verdier dual collision complex with the anti-holomorphic
sector, proving \textup{(ii)} inside the specified comparison datum. The twisting morphism
\(\tau\) is the bar universal coupling; the Mellin propagator comparison identifies its
tree-level image with \(K(z,\bar z,t)\), giving \textup{(iii)}. Purely holomorphic forms have
zero pairing with the missing anti-holomorphic degree, and one anti-holomorphic insertion still
does not supply the required \((0,n-3)\)-degree; this proves \textup{(iv)} by type. With two
anti-holomorphic insertions the assumed Parke--Taylor comparison gives the MHV pairing; in
arity \(4\) this is Theorem~\ref{thm:cbt-mhv-bar-integral}.
\end{proof}

\begin{remark}[Genus $g \geq 1$ persistence of helicity splitting]
\label{rem:ch-core-helicity-splitting-genus}

\begin{remark}[Colour-ordered amplitudes and the two-colour architecture]
\label{rem:ch-core-colour-ordered-two-colour}

 R_\sigma \cdot \mathcal{M}_n^{\mathrm{ord}}(\sigma),
\]
where $R_\sigma$ is the monodromy of the $R$-matrix along the

This is the celestial manifestation of the two-colour
architecture developed in Part~\ref{part:e1-core}: the ordered
bar complex $\Barchord(\cA)$ carries the colour-ordered

$\prod_{i=1}^n \langle i,\, i{+}1 \rangle^{-1}$ is the
$d\log$ residue on $\Conf_n^{<}(\C)$, the holomorphic colour
of the ordered bar differential, while the colour-sum
over~$\Sigma_n/\mathrm{cyc}$ is the topological colour's descent.
\end{remark}

\subsection{The filtered celestial transfer theorem}
\label{subsec:filtered-celestial-transfer}

The $d\log$ part of the Mellin transform identifies the associated graded of the HT-twisted soft expansion with the shadow tower of the celestial chiral algebra. Exact equality of filtered objects is the vanishing of the corresponding comparison cone.

The bulk side is the $4$d HT-twist programme of
Costello--Li--Paquette~\cite{CL16,
CostelloPaquetteTwistedSugraStates}; the boundary side is the
celestial OPE formalism of Pasterski--Shao--Strominger
\cite{PasterskiShaoStromingerFlatSpaceHolog,StromingerLectures2018};
the external soft-theorem comparison uses Hamada--Shiu
\cite{HamadaShiuSubsubSoft} and the celestial soft-current algebra
\cite{GuevaraHimwichPateStrominger21,StromingerWInfinityCelestial}.

\begin{definition}[Celestial chiral algebra of an HT-twist]
\label{def:celestial-chiral-algebra-ht-twist}
Let $T_4$ be a $4$-dimensional QFT (gauge, gravity, matter, or any
specified mixture of these) admitting a holomorphic--topological twist $T_4^{\mathrm{HT}}$
in the sense of Costello--Li. The \emph{celestial chiral algebra}
$\cA^{\mathrm{cel}}(T_4)$ is the boundary chiral algebra of the
$3$-dimensional holomorphic--topological reduction of $T_4^{\mathrm{HT}}$
on a null slab $S^2 \times \R_{\geq 0}$, extracted as the kernel of the
BV--BRST differential on observables supported on the celestial
sphere $S^2$ at $\mathscr{I}^+$.
\end{definition}

\begin{theorem}[Filtered celestial transfer]
\label{thm:universal-celestial-holography}
\index{celestial holography!filtered transfer theorem|textbf}
Let $T_4$ be a $4$d QFT admitting an HT-twist
$T_4^{\mathrm{HT}}$ and a transverse boundary condition whose boundary
QME anomaly vanishes. Assume:
\begin{enumerate}[label=\textup{(\alph*)}]
\item the celestial gluing class
\[
\gamma_{\mathrm{cel}}(T_4)\in
H^1\!\bigl(\PP^1_{\mathrm{cel}},
\mathcal{Aut}_{\mathrm{ch}}(\cA^{\mathrm{cel}})\bigr)
\]
vanishes, so that the boundary chiral algebra globalizes from the
affine celestial chart to $\PP^1_{\mathrm{cel}}$;
\item a genus-filtered shadow tower
\(\Theta_{\cA^{\mathrm{cel}}}=\sum_{g\ge0}\hbar^g\Theta^{(g)}\)
is fixed on \(\cA^{\mathrm{cel}}\);
\item for each genus \(g\) and shadow depth \(r\) under consideration there is a Mellin--shadow
cochain map
\[
\mu_{g,r}\colon
\gr^r H^\bullet\!\bigl(\Barch(\cA^{\mathrm{cel}})\bigr)^{(g)}
\longrightarrow
\gr^r\mathrm{Soft}_{\mathrm{HT}}^{(g)}(T_4),
\]
whose associated graded is the \(d\log\) extraction map.  The map \(\mu_{0,r}\) is tree-level
transfer data; the maps \(\mu_{g,r}\) for \(g\ge1\) include the loop/quantum correction data.
\end{enumerate}
Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $\cA^{\mathrm{cel}}(T_4)$ is a well-defined vertex-/chiral algebra
on $S^2$, constructed as the BV--BRST cohomology of the HT-reduced
boundary observables.
\item The derived chiral centre
$Z_{\mathrm{ch}}^{\mathrm{der}}(\cA^{\mathrm{cel}})
 := C^{\bullet}_{\mathrm{ch}}(\cA^{\mathrm{cel}},\cA^{\mathrm{cel}})$
carries an $E_2$-chiral algebra structure (chiral Deligne), and the
pair
$(\cA^{\mathrm{cel}},\, Z_{\mathrm{ch}}^{\mathrm{der}}(\cA^{\mathrm{cel}}))$
is canonically an algebra over the Swiss-cheese operad
$\mathrm{SC}^{\mathrm{ch,top}}$, with the bulk $Z^{\mathrm{der}}_{\mathrm{ch}}$
acting on the boundary $\cA^{\mathrm{cel}}$ via the universal brace.
\item The shadow obstruction tower
$\Theta_{\cA^{\mathrm{cel}}} = \sum_{g\geq 0}\hbar^g \Theta^{(g)}$ of
$\cA^{\mathrm{cel}}$ determines the associated graded of the
HT-twisted conformally-soft sector via the Mellin--shadow comparison
maps:
\begin{itemize}
\item $r=2$: leading Weinberg soft theorem from $\Theta^{(0)}$ at
 shadow depth~$1$.
\item $r=3$: sub-leading Cachazo--Strominger soft theorem from
 $\Theta^{(0)}$ at shadow depth~$2$.
\item $r\geq 4$: higher soft-current comparison from
 $\Theta^{(0)}$ at shadow depth~$r-1$
 (Theorem~\ref{thm:higher-soft-shadow}).
\end{itemize}
\item The exact filtered soft theorem
at $(g,r)$ holds if and only if the obstruction class
\[
\Obs^{\mathrm{Mell}}_{g,r}(T_4)
\in
H^2\!\bigl(\Cone(\mu_{g,r})\bigr)
\]
vanishes; the ambiguity of the filtered lift is
$H^1(\Cone(\mu_{g,r}))$.
\item The shadow class of $\cA^{\mathrm{cel}}$ (in the characteristic-datum
G/L/C/M classification of Part~\ref{part:examples}) records the algebraic regime of the
HT soft sector:
\begin{itemize}
\item Class $\mathbf G$ (gauge): $m_k^{\mathrm{SC}}=0$ for $k\geq 3$, the
 bar complex is Koszul; this is the pure-gauge tree-level pattern.
\item Class $\mathbf L$ (logarithmic): double-pole closure, quadratic
 BMS extension; this is the logarithmic/topologically finite pattern.
\item Class $\mathbf C$ (Clifford): quartic-pole stratum, cubic/quartic
 BMS extension (Theorem~\ref{thm:class-C-quartic-bms}); this records
 a rigid secondary anomaly.
\item Class $\mathbf M$ (modular): unbounded shadow depth, infinite
 tower of soft theorems; this is the Virasoro/\(\mathfrak{w}_{1+\infty}\)
 gravitational soft pattern.
\end{itemize}
\end{enumerate}
\end{theorem}

\begin{proof}
(i) Existence of $\cA^{\mathrm{cel}}$: under the stated anomaly
hypothesis, the HT-twist of $T_4$ produces a local BV theory on $\R^4$
holomorphic in one
complex direction and topological in the transverse direction; reduction
along the null slab yields a $3$d HT theory whose boundary chiral
algebra is well-defined by Costello--Gaiotto~\cite{CostelloGaiottoBoundaryVOA}
and Costello--Li~\cite{CL16}. Well-definedness of the
BV--BRST cohomology on $S^2$ follows from anomaly cancellation in the
twist and from $\gamma_{\mathrm{cel}}(T_4)=0$.
(ii) The $\mathrm{SC}^{\mathrm{ch,top}}$-structure on
$(\cA^{\mathrm{cel}}, Z^{\mathrm{der}}_{\mathrm{ch}}(\cA^{\mathrm{cel}}))$
is the two-coloured bulk--boundary action produced by the chiral
higher Deligne theorem (Theorem~\ref{thm:chiral-higher-deligne})
together with the Vol~II Swiss-cheese construction
(Theorem~\ref{V1-thm:sc-chtop-bulk-boundary}). The closed colour is
the $E_2$-chiral structure on the derived chiral centre, the open
colour is the $E_1$-chiral structure on $\cA^{\mathrm{cel}}$, and the
mixed sector is the universal brace action; the directional
restriction of $\mathrm{SC}^{\mathrm{ch,top}}$ is therefore retained.
(iii) The Mellin--shadow maps translate each shadow-tower coefficient
into the associated graded of a Mellin-space soft pole. Exact filtered equality is standard
obstruction theory for the cone of $\mu_{g,r}$, giving the displayed
$H^2/H^1$ obstruction--ambiguity pair.
(iv) The G/L/C/M classification is that of Part~\ref{part:examples};
class $\mathbf C$ is Theorem~\ref{thm:class-C-quartic-bms}. The
extremal case class~$\mathbf M$ is the infinite shadow depth of the
$\mathfrak{w}_{1+\infty}$ bar complex of Remark~\ref{rem:ch-core-soft-bar-cohomology}.
\end{proof}

\subsection{The Mellin--shadow dictionary}
\label{subsec:mellin-shadow-dictionary}

The filtered comparison sends bar-cohomology data on the celestial chiral algebra to Mellin-space poles of HT-twisted scattering amplitudes. Its associated graded is the $d\log$ extraction map; the filtered statement is the vanishing of the comparison cone.

\begin{proposition}[Mellin--shadow dictionary]
\label{prop:mellin-shadow-dictionary}
\index{Mellin transform!shadow dictionary}
\index{celestial holography!Mellin dictionary}
Let $T_4$ be an HT-twistable $4$d QFT and
$\cA^{\mathrm{cel}} = \cA^{\mathrm{cel}}(T_4)$ its celestial chiral
algebra (Definition~\ref{def:celestial-chiral-algebra-ht-twist}).
Let $\mathcal{M}_n(\omega_i,z_i,\bar z_i)$ be the $n$-point
momentum-space amplitude of $T_4$ in the HT-twisted sector, with
$\omega_i$ the energies and $(z_i,\bar z_i)$ the celestial
stereographic coordinates of the external momenta. Define the
Mellin transform
\[
\widetilde{\mathcal{M}}_n(\Delta_i,z_i,\bar z_i)
 := \prod_{i=1}^n \int_0^\infty \omega_i^{\Delta_i-1}\,d\omega_i\cdot
 \mathcal{M}_n(\omega_i,z_i,\bar z_i).
\]
Fix an integer \(k\ge0\) and a genus \(g\). Assume the perturbative Mellin transform at
genus \(g\) is defined and that a cochain map
\[
\mu_{g,k}\colon
\gr^kH^\bullet(\Barch(\cA^{\mathrm{cel}}))^{(g)}
\longrightarrow
\gr^k\mathrm{Soft}^{(g)}_{\mathrm{HT}}(T_4)
\]
has associated graded equal to the \(d\log\) extraction map. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item The tree-level term \(\widetilde{\mathcal{M}}_n^{(0)}\) is the $n$-point correlation
function of primary operators $\mathcal{O}_{\Delta_i}(z_i,\bar z_i)$
of the celestial chiral algebra $\cA^{\mathrm{cel}}$ in the
HT-twisted perturbative sector
(PSS~\cite{PasterskiShaoStromingerFlatSpaceHolog}).
\item Conformally soft poles $\Delta_i \to 1 - k$ for integer $k\geq 0$
have associated graded identified with bar-cohomology classes
$[\xi_k] \in H^k(\Barch(\cA^{\mathrm{cel}}))$ of bar degree $k$.
The genus-\(g\) filtered residue formula
\[
\operatorname{Res}_{\Delta_i = 1-k}\,
\widetilde{\mathcal{M}}^{(g)}_n(\Delta_i,z_i,\bar z_i)
\;=\;
\bigl\langle \xi_k(z_i,\bar z_i)\cdot
 \prod_{j\neq i}\mathcal{O}_{\Delta_j}(z_j,\bar z_j)
\bigr\rangle_{\cA^{\mathrm{cel}}}^{(g)}
\]
holds exactly when
\[
\Obs^{\mathrm{Mell}}_{g,k}(T_4)
\in H^2\!\left(
\Cone\!\left(
\mu_{g,k}\colon
\gr^kH^\bullet(\Barch(\cA^{\mathrm{cel}}))^{(g)}
\to
\gr^k\mathrm{Soft}^{(g)}_{\mathrm{HT}}(T_4)
\right)\right)
\]
vanishes; otherwise the displayed identity is the associated-graded
comparison.
\item Under these comparison maps, the shadow obstruction tower
$\Theta_{\cA^{\mathrm{cel}}}^{(g)}$ of genus $g$ is the
$g$-loop associated graded of the soft pole structure, and the
perturbative genus expansion $Z^{\mathrm{sh}} = \exp(\sum F_g\,\hbar^{2g})$
(Remark~\ref{rem:ch-core-w-shadow}) is the genus expansion of the
Mellin amplitude after the cone obstructions vanish in every genus.
\end{enumerate}
The correspondence is compatible with the $\mathrm{SC}^{\mathrm{ch,top}}$-structure
of Theorem~\ref{thm:universal-celestial-holography}(ii): bulk braces
correspond to soft-soft OPEs, boundary products correspond to
Mellin convolutions.
\end{proposition}

\begin{proof}
\textup{(i)} is the content of \cite[\S 2]{StromingerLectures2018} combined with
the identification of the HT-twisted boundary algebra as the boundary
chiral algebra of Costello--Gaiotto. \textup{(ii)} is a direct computation: at
the Mellin pole $\Delta_i = 1-k$, the integrand $\omega_i^{-k}d\omega_i$
extracts the degree-$k$ piece of the low-energy expansion of
$\mathcal{M}_n$, which equals the shadow-dual bar-cohomology class
$[\xi_k]$ on the associated graded by the $\mathrm{d}\log$-extraction
mechanism (Theorem~\ref{thm:ch-core-helicity-splitting}(i) and
Remark~\ref{rem:ch-core-helicity-splitting-genus}). The exact
filtered comparison is the lifting problem for \(\mu_{g,k}\), with
obstruction and ambiguity in the displayed cone cohomology. \textup{(iii)}
follows from the genus filtration on the bar complex together with the convergence criterion
for \(Z^{\mathrm{sh}}\). Compatibility with
$\mathrm{SC}^{\mathrm{ch,top}}$ is the celestial incarnation of the
chiral Deligne theorem.
\end{proof}

\subsection{Higher-$r$ soft theorems as shadow-tower classes}
\label{subsec:higher-r-soft}

Weinberg's soft graviton theorem at \(r=2\) and
Cachazo--Strominger's subleading theorem at \(r=3\) occupy the first
two shadow depths. Hamada--Shiu~\cite{HamadaShiuSubsubSoft} and the
celestial soft-current algebra
\cite{GuevaraHimwichPateStrominger21,StromingerWInfinityCelestial}
provide the external higher-soft comparison surface; the shadow tower
supplies the associated graded and the comparison cone whose vanishing
produces the exact filtered identity at each depth.

\begin{theorem}[Higher-soft theorems from shadow depth]
\label{thm:higher-soft-shadow}
\index{soft graviton theorem!higher-order}
\index{Hamada--Shiu soft theorem}
\index{celestial soft-current algebra}
Let $T_4$ be HT-twistable and $\cA^{\mathrm{cel}} = \cA^{\mathrm{cel}}(T_4)$.
Assume $\cA^{\mathrm{cel}}$ has shadow obstruction tower with finite
depth $r_{\max}$ at each genus
(e.g.\ class~$\mathbf L$ or~$\mathbf C$) or controlled asymptotic
growth (class~$\mathbf M$, with absolute convergence of
$Z^{\mathrm{sh}}$, Remark~\ref{rem:ch-core-w-shadow}). Assume also that the tree-level
Mellin--shadow maps \(\mu_{0,r}\) exist for the depths under consideration. Then for every
integer $r\geq 2$:
\begin{enumerate}[label=\textup{(\roman*)}]
\item The sub${}^{r-2}$-leading soft-graviton theorem of $T_4$
has associated graded residue
$\operatorname{Res}_{\Delta_s = 2-(r-2)}
 \widetilde{\mathcal{M}}_n = \langle \xi_{r-1} \cdots\rangle$
for $\xi_{r-1}\in H^{r-1}(\Barch(\cA^{\mathrm{cel}}))$, the shadow
class of depth $r-1$.
\item The coefficient in the \(r\geq4\) shadow-side statement is
controlled at tree level by the genus-$0$ shadow coefficient
$\Theta^{(0)}_{r-1}$.
\item The HT-twisted conformally-soft hierarchy is obtained from the
bar cohomology by filtered lifting:
\[
H^\bullet(\Barch(\cA^{\mathrm{cel}}))
\cong
\gr\!\left(
\bigoplus_{r\geq 0}\,\text{sub}^{r}\text{-leading soft data}
\right),
\]
and exact equality holds at depth $r$ precisely when
\[
\Obs^{\mathrm{soft}}_{r}(T_4)
\in H^2\!\bigl(\Cone(\mu_{0,r})\bigr)
\]
vanishes.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) Follows from Proposition~\ref{prop:mellin-shadow-dictionary}(ii)
applied to a soft graviton of energy $\omega_s\to 0$: the Mellin pole
$\Delta_s = 2 - (r-2) = 4-r$ is the $(r-2)$-th subleading
term in the soft expansion of a graviton insertion, and its residue
pairs on the associated graded with the bar class of depth $r-1$.
(ii) At $r=2$ (Weinberg) and $r=3$ (Cachazo--Strominger), the match is
Theorem~\ref{thm:ch-core-helicity-splitting}. At $r\geq 4$, the
coefficient extracted by the shadow residue is computed level-by-level
from the OPE of $T(z)$ with itself (for gravitons) or $J^a(z)$ with
itself (for gluons). The resulting operator is compared with the
Hamada--Shiu kinematic soft operator and the celestial soft-current
algebra, while the scalar coefficient is determined internally by the
vertex-algebra OPE coefficients of \(\cA^{\mathrm{cel}}\) at depth
\(r-1\), via the \(\mathrm d\log\) extraction. The filtered lift is
governed by
$H^\bullet(\Cone(\mu_{0,r}))$.
(iii) By the Koszul bigrading on $\Barch(\cA^{\mathrm{cel}})$ (Part~II)
each bar-cohomology class has a shadow depth $r$; the obstruction
class records exactly whether the associated-graded soft datum lifts
to an equality of filtered celestial residues.
\end{proof}

\begin{remark}[Obstruction-free soft sectors]
\label{rem:higher-soft-obstruction-free}
If $H^2(\Cone(\mu_{0,r}))=0$ for every $r$, then
Theorem~\ref{thm:higher-soft-shadow} gives the exact all-depth
tree-level soft hierarchy in the HT-twisted perturbative sector. For classes
$\mathbf G,\mathbf L,\mathbf C$ this is the original-complex
criterion; for class $\mathbf M$ it is the weight-completed criterion.
At genus \(g\ge1\), the same sentence uses \(\mu_{g,r}\) and the corresponding
\(\Obs^{\mathrm{Mell}}_{g,r}\).
\end{remark}

\subsection{Class $\mathbf C$ and the quartic BMS extension}
\label{subsec:class-C-quartic-bms}

The BMS algebra $\mathfrak{bms}_4$ is the semidirect product of super-translations with super-rotations (Barnich--Troessaert~\cite{BarnichTroessaertBMS}). Its Virasoro extension is the $\mathfrak{w}_{1+\infty}$ tower of Strominger~\cite{StromingerWInfinityCelestial}. Class~$\mathbf C$ supplies a third operation: a quartic BMS extension at shadow depth $r=4$.

\begin{theorem}[Class $\mathbf C$ quartic BMS extension]
\label{thm:class-C-quartic-bms}
\index{BMS algebra!quartic extension}
\index{celestial holography!class C}
Let $\cA^{\mathrm{cel}}$ be a celestial chiral algebra of class~$\mathbf C$,
i.e.\ with double-pole / quartic-pole closure and Clifford secondary
structure (Theorem~\ref{V1-thm:g9-clifford-structure}). Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item The shadow stratum of depth $r=4$ carries a non-trivial quartic
operation $\mu_4^{\mathrm{BMS}}$ on the BMS generators, producing a
\emph{quartic BMS extension} $\mathfrak{bms}_4^{\mathbf C}$ whose
central term is controlled by the secondary anomaly $u = \eta^2$
of the anomaly-completed transgression.
\item The quartic BMS bracket
$[T_{\ell_1},T_{\ell_2},T_{\ell_3},T_{\ell_4}]_{\mathrm{qBMS}}$
is the quartic $L_\infty$-bracket on the soft-graviton modes at
celestial conformal dimension $\Delta = 1-3$ (shadow depth $4$).
\item On the $c=13$ self-dual line, $\mathfrak{bms}_4^{\mathbf C}$
reduces to the Clifford-completed exterior algebra at
$u = \kappa\cdot\omega_g$, giving the Clifford-type BMS extension
of Theorem~\ref{V1-thm:g9-clifford-structure}.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) Class~$\mathbf C$ has a quartic collision residue
(the pole-order dichotomy): the bar differential has
a non-trivial $m_4$ at residue-pole order $4$. This $m_4$ promotes
the shadow stratum at depth $r=4$ to a genuine quartic operation on
the BMS generators. The secondary anomaly $u = \eta^2$
(\S~\ref{subsec:tholog-coda})
is precisely the obstruction cocycle that centralizes $\mu_4^{\mathrm{BMS}}$.
(ii) follows from the Swiss-cheese Deligne theorem applied to the
class~$\mathbf C$ shadow stratum at depth $4$; the $L_\infty$
structure is the transferred bracket of
\S~\ref{sec:ordered-bar-explicit}. (iii) is
Theorem~\ref{V1-thm:g9-clifford-structure} on the self-dual line.
\end{proof}

\begin{remark}[Physical content]
\label{rem:class-C-physics}
Class~$\mathbf C$ is the algebraic signature of theories carrying a
rigid secondary anomaly (gravitational anomaly-completed
chiral algebras, certain $\mathcal W$-algebras). The quartic BMS
extension is the celestial form of this secondary datum:
it is the lowest extension on which the $u$-degeneration of the Clifford algebra
acts non-trivially, hence the lowest extension separating
the $c=26$ critical-string dichotomy
(\S~\ref{sec:critical-string-clifford-degeneration} in
\texttt{thqg\_critical\_string\_dichotomy.tex}).
\end{remark}
