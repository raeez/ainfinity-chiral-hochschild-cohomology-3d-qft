\begin{abstract}
\noindent\itshape
The BV/BRST quantization of a three-dimensional holomorphic--topological
field theory on $\bC_z\times\bR_t$ is the chiral bar construction of its
boundary chiral algebra. Holomorphic collisions on $\bC$ generate the
bar differential; topological point-splittings on $\bR$ generate the
coproduct; the two-coloured Swiss-cheese chiral--topological operad
$\SCchtop$ is the universal home of the bulk--boundary structure. Under
this identification the quantum master equation is the Maurer--Cartan
equation, each $A_\infty$ identity is Stokes' theorem on a
Fulton--MacPherson compactification, and the conformal anomaly is bar
curvature: $Q_{\mathrm{BRST}}^2 = \frac{c-26}{12}c_0$ corresponds to
$d_{\mathrm{fib}}^2=\kappa_{\mathrm{tot}}\,\omega_1$ with
$\kappa_{\mathrm{tot}}=(c-26)/2$. The functorial capstone assigns to a
chirally Koszul boundary algebra $\cA$ a holomorphic--topological
theory with $\Obs^\partial\simeq\cA$ and
$\Obs^{\mathrm{bulk}}\simeq\Zder(\cA)$; its value at $\cA=\Vir_c$ under
the Brown--Henneaux chart $c=3\ell/2G_N$ is the boundary reading of pure
3d gravity. This companion states the load-bearing definitions,
theorems with their exact hypothesis packages, the physics dictionary,
and the genuinely open problems, and places the volume in the
five-garden programme.
\end{abstract}

\section{The seed}\label{sec:seed}

A three-dimensional field theory on $\bC_z\times\bR_t$, holomorphic
along $\bC$ and topological along $\bR$, has two inequivalent ways of
splitting points. Operators separated along $\bC$ collide against the
logarithmic kernel $\eta_{ij}=d\log(z_i-z_j)$; the residues of these
collisions against operator-product modes assemble into a differential.
Operators separated along $\bR$ merely order themselves; their fusion
assembles into a coproduct. The first structure is the chiral bar
differential of Beilinson--Drinfeld chiral algebra theory, realised on
the Fulton--MacPherson compactifications $\FM_k(\bC)$; the second is the
deconcatenation coproduct on the tensor coalgebra
$T^c(s^{-1}\bar\cA)$. A three-dimensional holomorphic--topological
theory is precisely a coherent interlocking of the two, and the operad
that governs the interlocking --- closed colour $\FM_k(\bC)$, open
colour ordered configurations $\Conf^{<}_m(\bR)$, mixed colour the
Voronov half-plane models $\FM_{k,m}(\bH)$, with the directional
restriction $\SCchtop(\ldots,\mathsf{op},\ldots;\mathsf{cl})=\varnothing$
imposed as a zero chain complex --- is the two-coloured Swiss-cheese
chiral--topological operad $\SCchtop$. Dunn additivity does not apply to
it; the decoupled product $\FM_k(\bC)\times E_1(m)$ is only the
depth-zero associated graded; its coupled homotopy-Koszulity is a named
conjecture.

Under this identification the standard apparatus of quantization
becomes the standard apparatus of homotopical algebra, term by term.
The quantum master equation
$\hbar\Delta_{\mathrm{BV}}S_{\mathrm{eff}}
+\tfrac12\{S_{\mathrm{eff}},S_{\mathrm{eff}}\}_{\mathrm{BV}}=0$
is the Maurer--Cartan equation
$D_\cA\Theta_\cA+\tfrac12[\Theta_\cA,\Theta_\cA]=0$ of the bar
construction; at fixed diagrammatic weight it is the planar $A_\infty$
identity, and the $A_\infty$ identity in turn is Stokes' theorem for
logarithmic forms on $\FM_k(\bC)$, with the Arnold--Orlik--Solomon
relations cancelling the corner strata. The conformal anomaly is bar
curvature: the Virasoro ghost system of weights $(2,-1)$ has
$c_{\mathrm{gh}}=-26$, and
\[
Q_{\mathrm{BRST}}^2=\frac{c-26}{12}\,c_0
\qquad\longleftrightarrow\qquad
d_{\mathrm{fib}}^2=\kappa_{\mathrm{tot}}\cdot\omega_1,
\qquad
\kappa_{\mathrm{tot}}=\kch(\cA)+\kch(bc)=\frac{c}{2}-13 .
\]
Anomaly cancellation $c=26$ and flatness $\kappa_{\mathrm{tot}}=0$ are
the same equation. Nothing in this dictionary is a metaphor; each entry
is either a proved comparison or a comparison problem with a name, and
this companion keeps the two apart.

\section{The objects}\label{sec:objects}

\begin{definition}[$A_\infty$-chiral algebra]
An $A_\infty$-chiral algebra on a curve is a cochain complex $(A,Q)$
with translation $\partial$, unit $\mathbf 1$, and operations
$m_k\colon A^{\otimes k}\to A((\lambda_1))\cdots((\lambda_{k-1}))$ of
degree $2-k$, the $\lambda_i$ being spectral parameters Fourier-dual to
coordinate differences, packaged as a square-zero degree-one
coderivation $D_A$ on $\barB(A)=T^c(s^{-1}\bar A)$. A curved variant
absorbs $m_1^2\neq0$ over moduli into $m_0$. The algebra arises as a
chart: $A=A_b=\operatorname{RHom}_{\cC}(b,b)$ for a vacuum object $b$ of
an open factorization category $\cC$; the Morita class of $\cC$, not
the quasi-isomorphism class of $A_b$, is the invariant.
\end{definition}

\begin{definition}[The bicoloured primitive]
The primitive object of the volume is not an algebra but the nine-tuple
$(\cC|_{(X,D,\tau)},\ \cD^{\mathrm{cl}}|_X;\ b,\ A_b,\
F^{\mathrm{cl}};\ \beta_{1/2},\ \mathrm{tr}^{\mathrm{cl}}_X;\ \Tr_\cC)$:
open factorization dg-category on a tangential log curve, closed
factorization category, boundary vacuum, chart algebra, closed vacuum
factorization algebra, half-braiding, closed-colour trace, open-colour
cyclic trace. Modularity is a property of the open category --- cyclic
trace plus clutching --- whose closed shadow has modular consequences;
it is never a property of the closed algebra alone.
\end{definition}

Five objects are never conflated: the chart algebra $A_b$; the bar
coalgebra $\barB(A_b)$, which classifies twisting and coupling; its
cohomology $A^{\mathrm i}$; the Koszul dual $A^!$, formed by Verdier
duality on the Ran space; and the derived chiral centre
$\Zder(A_b)\simeq\ChirHoch^\bullet(A_b,A_b)$, which is the bulk.
$\Omega\barB(A_b)\simeq A_b$ is bar--cobar \emph{inversion}, not Koszul
duality; the bulk is reached by Hochschild cochains, not by the bar.

\begin{definition}[BV datum]
A 3d holomorphic--topological BV theory on $\bR_t\times\bC_z$ has field
complex
$\cE=\Omega^\bullet(\bR_t)\widehat\otimes\Omega^{0,\bullet}(\bC_z)
\otimes\mathfrak a[1]$,
differential $Q=d_t+\bar\partial_z+d_{\mathfrak a}$, odd symplectic
pairing $\omega_{\mathrm{BV}}(\alpha,\beta)=\int dz\wedge
\langle\alpha,\beta\rangle_{\mathfrak a}$, cubic interaction, a
$Q$-compatible Lagrangian boundary condition, and heat-kernel
propagator $P_{\epsilon<L}=\int_\epsilon^LQ^*K_s\,ds$ whose singular
part is the Cauchy--step kernel $\Theta(t)/2\pi z$.
\end{definition}

Two further objects carry the higher structure. The \emph{modular
Swiss-cheese operad} extends the closed colour to all genera,
$C_*(\FM_k(\Sigma_g))$, with mixed sector the holomorphic--topological
screens; its genus-zero restriction is $\SCchtop$. The \emph{relative
Feynman transform} $\mathrm{FT}_{\mathrm{rel}}(C)$ is the completed
stable-graph coalgebra with bicomplex $D_0$ (genus-preserving bar) and
$D_1$ (modular self-sewing), $(D_0+D_1)^2=0$; it is the exact recipient
of all-genus partition data.

\section{The theorems}\label{sec:theorems}

Statuses are the manuscript's own: \emph{proved}, \emph{proved with a
fixed associator}, \emph{conditional} on a named package, or
\emph{conjecture}. The volume's discipline is that each theorem carries
its licensing data in the statement; this companion preserves exactly
that.

\begin{theorem}[Physics bridge; conditional]\label{thm:bridge}
A 3d holomorphic--topological QFT on $\bR_t\times\bC_z$ with
\textup{(a)} BV datum in product gauge, \textup{(b)} factorized
parametrix $P_{\mathrm{sing}}=\sum_\alpha K^{\mathrm{hol}}_\alpha(z)
\boxtimes H^{\mathrm{top}}_\alpha(t)$ modulo smooth exact terms,
\textup{(c)} one-loop finiteness in the sense of
Gwilliam--Rabinovich--Williams, and \textup{(d)} product-polynomial
interactions, has BV--BRST observables forming a logarithmic
$\SCchtop$-algebra. Hypothesis \textup{(b)} is essential and not
automatic: the naive product of Green kernels fails it.
\end{theorem}

\begin{theorem}[Stokes--Arnold calculus; proved]\label{thm:fm}
On the manifold-with-corners $\FM_k(\bC)$, closed interior logarithmic
forms with operadic boundary factorization obey
$\int_\Gamma d\omega=\sum_{|S|\ge2}\epsilon_{D_S}
\int_{\Gamma\cap D_S}\Rez_{D_S}(\omega)$, with computed orientation
signs; the resulting identities on residues are exactly the
$A_\infty$ identities, the Arnold--Orlik--Solomon relations cancelling
all corner contributions.
\end{theorem}

\begin{theorem}[BV equals bar, stratified by class]\label{thm:bvbar}
The identification of the BV complex with the chiral bar complex,
$Q_{\mathrm{BV}}=\{S,-\}\leftrightarrow d_{\mathrm{bar}}$, holds:
unconditionally at chain level for class $\mathsf G$
\textup{(Heisenberg, lattice)} and class $\mathsf L$ \textup{(affine
Kac--Moody at non-critical level)}; for class $\mathsf C$
\textup{($\beta\gamma$, $bc$)} conditionally on vanishing of the
harmonic-decoupling obstructions; for class $\mathsf M$
\textup{(Virasoro, $\cW_N$)} it is \emph{false} on the raw bounded
direct-sum complex and holds in the weight-completed ambient. On
cohomology the identification holds for all four classes through the
Drinfeld--Sokolov--Hochschild bridge. The interacting obstruction is
the harmonic part of the Hodge-decomposed BV propagator
$P_{\mathrm{BV}}=d\log E+[d_{\mathrm{fib}},h]+P_{\mathrm{harm}}$.
\end{theorem}

\begin{theorem}[Genus-zero BRST--bar; conditional]\label{thm:brst}
For a PBW conformal algebra $\cA$ with $c=26$ there is a
quasi-isomorphism
$(\cA^{\mathrm{rel}}_{\mathrm{tot}},Q_{\mathrm{BRST}})
\xrightarrow{\sim}(\barB^{\mathrm{ch}}_0(\cA_{\mathrm{tot}}),
d_{\mathrm{bar}})$ by conformal-weight filtration, conditional on
vanishing filtered lifting obstructions. Unconditionally,
$\kappa_{\mathrm{tot}}=0$ if and only if
$c_{\mathrm{matter}}+c_{\mathrm{ghost}}=0$: both squares are controlled
by $c-26$. The semi-infinite variant
$\barB^{\mathrm{ch}}(\widehat\fg_k)\simeq
C^{\infty/2+\bullet}(\widehat\fg_k,V_k)$ holds for all
$k\neq-h^\vee$ with no anomaly condition --- the semi-infinite wedge
absorbs it --- and extends to $\cW$-algebras by Drinfeld--Sokolov
reduction.
\end{theorem}

\begin{theorem}[The $\SCchtop$ heptagon; proved with associator]
\label{thm:heptagon}
Seven presentations of $\SCchtop$ --- coloured operadic; Koszul-dual
via distributive laws; factorization on the Ran space; BV/BRST
(coderived); convolution $L_\infty$; Drinfeld centre of the
factorization module category; derived-algebraic-geometric
(one-shifted symplectic moduli) --- are pairwise
$\infty$-quasi-isomorphic after fixing a Drinfeld associator
$\Phi\in\mathrm{GRT}_1(\bQ)$, and the identification is
associator-independent on cohomology. Chain-level strictness holds on
the evaluation-generated Koszul locus. The arity-$(2,2)$ bicoloured
coherence polytope is required; the pentagon alone is insufficient.
The Swiss-cheese operad is \emph{not} formal \textup{(Livernet)}, so
the edge that would follow from formality instead routes through the
coupled homotopy-Koszulity of $\SCchtop$ --- a named conjecture, proved
only at the level of associated graded.
\end{theorem}

\begin{theorem}[Chiral higher Deligne]\label{thm:deligne}
For $\cA$ on the Koszul locus: \textup{(1)} the derived chiral centre
$\Zder(\cA)=\ChirHoch^\bullet(\cA,\cA)$ carries the canonical chiral
brace \textup{($E_2$-chiral)} action \textup{(proved with associator)};
\textup{(2)} conditional on a signed two-coloured contracting homotopy
for $\Omega((\SCchtop)^!)$ --- a genuinely new two-coloured lift of the
chiral Positselski theorem, not yet established --- the pair
$(\Zder(\cA),\cA)$ is the universal $\SCchtop$ bulk--boundary datum;
\textup{(3)} with explicit topologization data the mixed structure
refines to $E_3$-topological, associator-dependent at chain level and
independent on cohomology. Chiral Hochschild cohomology commutes with
Drinfeld--Sokolov reduction:
$\ChirHoch^\bullet(\cW_k(\fg))\simeq
H^\bullet_{\mathrm{DS}}(\ChirHoch^\bullet(V_k(\fg)))$
\textup{(weight-completed)}.
\end{theorem}

\begin{theorem}[PVA descent; proved]\label{thm:pva}
The $Q$-cohomology of a logarithmic $\SCchtop$-algebra carries a
$(-1)$-shifted Poisson vertex algebra structure:
$[a]\cdot[b]=[\mu(a,b)]$ and
$\{[a]_\lambda[b]\}=[\sum_n\frac{\lambda^n}{n!}a_{(n)}b]$, with
commutativity from the exchange cylinder \textup{(the two line
orderings joined through the bulk)}, Jacobi from the three-face Stokes
argument on $\FM_3(\bC)$, Leibniz from the mixed sector. Vanishing of
the higher $m_{k\ge3}$ on cohomology is \emph{conditional} on
compatible topological null-homotopies; contractibility of
$\Conf^{<}_k(\bR)$ alone is not sufficient. Quantization of the PVA is
a separate conditional problem \textup{(analytic
Knizhnik--Zamolodchikov strong deformation retract, Stokes data,
reflected weights, and the lift $T=[Q_{\mathrm{tot}},G]$)}.
\end{theorem}

\begin{theorem}[Raviolo restriction; proved]\label{thm:raviolo}
Time-slice restriction of a $\SCchtop$-algebra at $t_0$ glues two
formal disks along the punctured disk and produces a dg raviolo vertex
algebra, the state--field map recovered from the singular part of $m_2$
by the divided-power Borel transform
$\lambda^n/n!\leftrightarrow z^{-n-1}$; raviolo cohomology carries the
Poisson vertex structure.
\end{theorem}

\begin{theorem}[Higher genus; conditional tower]\label{thm:genus}
Dunn additivity fails for $\SCchtop$; the genus obstruction is the
curvature $d_{\mathrm{fib}}^2=\kch(\cA)\,\omega_g$. For chirally Koszul
$\cA$ at non-critical level and $g\ge2$, granted modular-bootstrap
acyclicity $H^2(\fg^{\cA}_{\mathrm{mod}},d_{\Theta^{(0)}})=0$ and the
degree-two comparison criterion, the twisting cochain extends genus by
genus and the modular bar complex acquires a twisted-tensor
representative; Fay's trisecant identity supplies the genus-$g$ Arnold
relations for the prime-form propagator. The complete modular bar
object \emph{is} the relative Feynman transform
\textup{(proved, and explicitly formal: the stable-graph skeleton does
not determine global periods or monodromy)}.
\end{theorem}

\begin{theorem}[Unified chiral quantum group; proved on its verified
surface]\label{thm:ucqg}
For simple $\fg$, a good grading from an $\fsl_2$-triple, level
$k\neq-h^\vee$ and shift $\mu$, there is a chiral bialgebra datum
$Q^{k,f,\mu}_{\fg}$ with shifted-parameter coassociative spectral
coproduct, spectral $R$-matrix satisfying the Yang--Baxter equation
with quasi-triangularity $R(z)\Delta^{\mathrm{op}}_z=\Delta_zR(z)$,
Koszul dual realising bar--cobar inversion, and Drinfeld--Sokolov
specialisation $Q^{k,f_{\mathrm{prin}},0}_{\fg}=\cW_k(\fg)$. The
verified surface is the locus where five named obstruction classes
vanish. Specialisation fibres include the Yangian, affine and shifted
Yangians, truncated BFN Coulomb-branch algebras, finite and affine
$\cW$-algebras. The super-chiral Yangian satisfies
$(Y^s_\hbar(\fg))^!\simeq Y^s_{-\hbar}(\fg)^\theta$ with the
Chevalley--Kac involution, at all orders via the unitarity functional
equation $R_s^{-1}(u)=R_s(-u-\hbar h^\vee_s)$.
\end{theorem}

\begin{theorem}[Ordered associative chiral Koszul duality; proved at
genus zero]\label{thm:ordered}
For a strongly admissible augmented associative chiral algebra $A$ with
bar coalgebra $C=\barB^{\mathrm{ch}}(A)$: the ordered chiral shuffle
gives $\barB(A\otimes^{\mathrm{ch}}B)\simeq\barB(A)\widehat\otimes
\barB(B)$; opposite algebras give co-opposite coalgebras; there is a
PBW-complete equivalence of bimodule and bicomodule derived categories;
the diagonal bimodule corresponds to the coalgebra $C_\Delta$ --- the
diagonal is the universal boundary condition --- and is the unit for
derived cotensor; Hochschild and coHochschild theories are dual; and
when $A$ is $d$-Calabi--Yau the ordered pair-of-pants and trace
operations carry a $d$-shifted ordered Frobenius structure.
\end{theorem}

\begin{theorem}[Fractional ghosts; proved]\label{thm:frghost}
Non-principal Drinfeld--Sokolov reductions carry half-integral Kazhdan
gradings; a Kummer branched cover of the curve of degree $d_f$
integralises the grading, the Sugawara antighost identity
$[Q,G']=T$ holds on the cover and Galois-descends. Consequently the
derived chiral centres of the Bershadsky--Polyakov algebra
$\cW^{(2)}_3$, of minimal $\cW^k(\fsl_4,f_{\min})$, and of \emph{all}
good-graded nilpotent reductions carry $E_3$-topological structure ---
the topologisation ladder extends unconditionally beyond principal
gradings.
\end{theorem}

\begin{theorem}[Universal Holography; coherence theorem on realization
data]\label{thm:holography}
There is a functor
$\Phi_{\mathrm{hol}}\colon\mathsf{ChirAlg}^{\omega,\mathrm{BL},
\mathrm{adm},\Xi}_X\to\mathsf{HT\text{-}QFT}^{\Xi}_{X\times\bR}$

\end{theorem}

\section{The dictionary and the constants}\label{sec:dictionary}
